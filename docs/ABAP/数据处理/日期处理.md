## 日期交叉工具类

方法列表：

- `MERGE_DATE_RANGES` 合并日期
- `GET_VALID_DATE_RANGE ` 获取有效时间间隔
- `SPLIT_TIME_RANGE_BY_DAY ` 将时间段按天拆分
- `STD_TIME_TO_SECONDS` 转换时间为秒数
- `CALCULATE_DAY_INTERSECTION` 获取交叉日期（非跨天）
- `CALCULATE_DAY_INTERSECTION_NEW` 获取交叉日期（可处理跨天）

```abap
class zcl_hr_daye_util definition
  public
  final
  create public .

  public section.

    types:
      begin of ty_date_range,
        begda type dats,
        endda type dats,
      end of ty_date_range .
    types:
      tt_date_range type sorted table of ty_date_range with non-unique key begda endda .
    types:
      begin of ty_time_range,
        date  type dats,
        beguz type tims,
        enduz type tims,
      end of ty_time_range .
    types:
      tt_time_range type sorted table of ty_time_range with non-unique key date beguz .

    class-methods merge_date_ranges
      importing
        !it_data1        type tt_date_range
        !it_data2        type tt_date_range
      returning
        value(rt_result) type tt_date_range .
    class-methods get_valid_date_range
      importing
        !it_p0000        type p0000_tab
        !it_p0001        type p0001_tab
      returning
        value(rt_result) type tt_date_range .
    class-methods split_time_range_by_day
      importing
        !is_time         type ty_time_range
      returning
        value(rt_result) type tt_time_range .
    class-methods std_time_to_seconds
      importing
        !iv_date         type dats
        !iv_time         type tims
      returning
        value(rv_result) type int4 .
    class-methods calculate_day_intersection
      importing
        !is_segment1      type ty_time_range
        !is_segment2      type ty_time_range
      exporting
        !et_intersect     type tt_time_range
        !et_non_intersect type tt_time_range .
    class-methods calculate_day_intersection_new
      importing
        !is_time1          type ty_time_range
        !is_time2          type ty_time_range
      exporting
        !et_intersection   type tt_time_range
        !et_nointersection type tt_time_range .
  protected section.
  private section.
ENDCLASS.
```

```abap
CLASS ZCL_HR_DAYE_UTIL IMPLEMENTATION.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>CALCULATE_DAY_INTERSECTION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_SEGMENT1                    TYPE        TY_TIME_RANGE
* | [--->] IS_SEGMENT2                    TYPE        TY_TIME_RANGE
* | [<---] ET_INTERSECT                   TYPE        TT_TIME_RANGE
* | [<---] ET_NON_INTERSECT               TYPE        TT_TIME_RANGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method calculate_day_intersection.

    clear: et_intersect, et_non_intersect.

    data(lv_start1) = is_segment1-beguz.
    data(lv_end1) = is_segment1-enduz.
    data(lv_start2) = is_segment2-beguz.
    data(lv_end2) = is_segment2-enduz.

    " 计算交集
    data(lv_intersect_start) = cond tims(
      when lv_start1 > lv_start2 then lv_start1
      else lv_start2 ).
    data(lv_intersect_end) = cond tims(
      when lv_end1 < lv_end2 then lv_end1
      else lv_end2 ).

    if lv_intersect_start <= lv_intersect_end.
      append value #(
        date  = is_segment1-date
        beguz = lv_intersect_start
        enduz = lv_intersect_end ) to et_intersect.
    endif.

    " 计算非交集部分（可能有两个区间）
    if lv_start1 < lv_intersect_start.
      append value #(
        date  = is_segment1-date
        beguz = lv_start1
        enduz = lv_intersect_start ) to et_non_intersect.
    endif.

    if lv_end1 > lv_intersect_end.
      append value #(
        date  = is_segment1-date
        beguz = lv_intersect_end
        enduz = lv_end1 ) to et_non_intersect.
    endif.
  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>CALCULATE_DAY_INTERSECTION_NEW
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_TIME1                       TYPE        TY_TIME_RANGE
* | [--->] IS_TIME2                       TYPE        TY_TIME_RANGE
* | [<---] ET_INTERSECTION                TYPE        TT_TIME_RANGE
* | [<---] ET_NOINTERSECTION              TYPE        TT_TIME_RANGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method calculate_day_intersection_new.

    clear et_intersection.
    clear et_nointersection.

    " 将两个时间段按天拆分
    data(lt_segments1) = split_time_range_by_day( is_time1 ).

    data(lt_segments2) = split_time_range_by_day( is_time2 ).

    " 计算每天的交叉部分
    loop at lt_segments1 into data(ls_segment1).
      read table lt_segments2 into data(ls_segment2) with key date = ls_segment1-date.
      if sy-subrc = 0.
        calculate_day_intersection(
          exporting
            is_segment1      = ls_segment1
            is_segment2      = ls_segment2
          importing
            et_intersect     = data(lt_intersect)
            et_non_intersect = data(lt_non_intersect) ).
        append lines of lt_intersect to et_intersection.
        append lines of lt_non_intersect to et_nointersection.
      else.
        " 如果没有对应日期的段，整个段都是非交叉
        append ls_segment1 to et_nointersection.
      endif.
    endloop.


  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>GET_VALID_DATE_RANGE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IT_P0000                       TYPE        P0000_TAB
* | [--->] IT_P0001                       TYPE        P0001_TAB
* | [<-()] RT_RESULT                      TYPE        TT_DATE_RANGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method get_valid_date_range.

    data: lt_p0000 type sorted table of p0000 with unique key begda,
          lt_p0001 type sorted table of p0001 with unique key begda.

    "--------------------------> 踢出离职数据
    loop at it_p0000 assigning field-symbol(<ls_p0000>).

      if lt_p0000 is initial.
        data(ls_p0000) = <ls_p0000>.
        insert ls_p0000 into table lt_p0000.
        continue.
      endif.

      if ls_p0000-stat2 = <ls_p0000>-stat2 and ls_p0000-stat2 = '3'.
        ls_p0000-endda = <ls_p0000>-endda.
        modify table lt_p0000 from ls_p0000.
      else.
        ls_p0000 = <ls_p0000>.
        insert ls_p0000 into table lt_p0000.
      endif.

    endloop.

    delete lt_p0000 where stat2 <> '3'.

    "--------------------------> 踢出不符合条件的PA0001
    lt_p0001 = it_p0001.


    "--------------------------> 最终有效数据
    data: lt_data1 type tt_date_range,
          lt_data2 type tt_date_range.

    move-corresponding lt_p0000 to lt_data1.
    move-corresponding lt_p0001 to lt_data2.

    rt_result = merge_date_ranges(
      it_data1 = lt_data1
      it_data2 = lt_data2
    ).

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>MERGE_DATE_RANGES
* +-------------------------------------------------------------------------------------------------+
* | [--->] IT_DATA1                       TYPE        TT_DATE_RANGE
* | [--->] IT_DATA2                       TYPE        TT_DATE_RANGE
* | [<-()] RT_RESULT                      TYPE        TT_DATE_RANGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method merge_date_ranges.

    data: lt_range  type tt_date_range,
          lt_result type table of ty_date_range.

    loop at it_data1 into data(ls_data1).
      loop at it_data2 into data(ls_data2).
        data(ls_range) = ls_data1.
        ls_range-begda = nmax( val1 = ls_data1-begda val2 = ls_data2-begda ).
        ls_range-endda = nmin( val1 = ls_data1-endda val2 = ls_data2-endda ).
        if ls_range-begda <= ls_range-endda.
          insert ls_range into table lt_range.
        endif.
      endloop.
    endloop.

    data: lv_endda type dats.
    loop at lt_range into ls_range.

      if lt_result is initial.
        append ls_range to lt_result.
        continue.
      endif.

      data(ls_last) = lt_result[ lines( lt_result ) ].

      lv_endda = ls_last-endda + 1.

      if ls_range-begda = lv_endda.
        lt_result[ lines( lt_result ) ]-endda = ls_range-endda.
      else.
        append ls_range to lt_result.
      endif.

    endloop.

    insert lines of lt_result into table rt_result.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>SPLIT_TIME_RANGE_BY_DAY
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_TIME                        TYPE        TY_TIME_RANGE
* | [<-()] RT_RESULT                      TYPE        TT_TIME_RANGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method split_time_range_by_day.

    data: lv_start_epoch type i,
          lv_end_epoch   type i,
          lv_days_diff   type i.

    " 标准化时间为秒数
    lv_start_epoch = std_time_to_seconds(
      iv_date = is_time-date
      iv_time = is_time-beguz ).

    lv_end_epoch = std_time_to_seconds(
      iv_date = is_time-date
      iv_time = is_time-enduz ).

    " 处理跨天情况
    if lv_end_epoch <= lv_start_epoch.
      lv_end_epoch = lv_end_epoch + 86400. " 增加一天秒数
    endif.

    lv_days_diff = lv_end_epoch div 86400 - lv_start_epoch div 86400.

    " 按天拆分时间段
    do lv_days_diff + 1 times.
      data(lv_current_day) = is_time-date + ( sy-index - 1 ).
      data(lv_day_start) = cond tims(
        when sy-index = 1 then is_time-beguz
        else '000000' ).
      data(lv_day_end) = cond tims(
        when sy-index = lv_days_diff + 1 then is_time-enduz
        else '240000' ).

      append value #(
        date  = lv_current_day
        beguz = lv_day_start
        enduz = lv_day_end ) to rt_result.

    enddo.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_HR_DAYE_UTIL=>STD_TIME_TO_SECONDS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_DATE                        TYPE        DATS
* | [--->] IV_TIME                        TYPE        TIMS
* | [<-()] RV_RESULT                      TYPE        INT4
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method std_time_to_seconds.

    " 将日期时间转换为秒数（相对基准日期）
    constants cv_date_first type dats value '19700101'.

    data(lv_days) = iv_date - cv_date_first.

    rv_result = lv_days * 86400 +
                iv_time+0(2) * 3600 +
                iv_time+2(2) * 60 +
                iv_time+4(2).


  endmethod.
ENDCLASS.
```
