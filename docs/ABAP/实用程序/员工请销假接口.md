* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->CONV_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_DATA_IN                     TYPE        TY_DATA_IN
* | [<-->] CS_DATA                        TYPE        TY_DATA
* | [<-->] CS_OUT                         TYPE        TY_RESULT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method conv_data.

    if is_data_in-pernr is initial.
      cs_out-code = gc_error.
      cs_out-message = |人员编号必输|.
      return.
    endif.

    if not is_exists_pernr( is_data_in-pernr ).
      cs_out-code = gc_error.
      cs_out-message = |人员编号不存在|.
      return.
    endif.

    if is_data_in-leave_type is initial.
      cs_out-code = gc_error.
      cs_out-message = |假勤类型必输|.
      return.
    endif.

    if is_data_in-leave_list is initial.
      cs_out-code = gc_error.
      cs_out-message = |无明细数据|.
      return.
    endif.

    cs_data = value #( pernr    = is_data_in-pernr
                       nocommit = is_data_in-nocommit
    ).

    if is_data_in-leave_type = '10' or is_data_in-leave_type = '20'.

      data(lt_leave_list) = is_data_in-leave_list.

      splite_date(
        changing
          ct_data = lt_leave_list
          cs_out  = cs_out
      ).

      handling_daily_data(
        exporting
          iv_pernr = is_data_in-pernr
          iv_type  = is_data_in-leave_type
          it_data  = lt_leave_list
        importing
          et_leave = gt_leave_date
          et_work  = gt_work_date
        changing
          cs_out   = cs_out
      ).

      check cs_out-code <> gc_error.
      loop at gt_work_date into data(ls_work_date).
        if ls_work_date-action = 'M'.
          append initial line to cs_data-it2001 assigning field-symbol(<ls_it2001_mod>).
          <ls_it2001_mod> = value #( pernr = is_data_in-pernr
                                     begda = ls_work_date-date
                                     endda = ls_work_date-date
                                     subty = ls_work_date-awart
                                     awart = ls_work_date-awart
                                     beguz = normalize_time( ls_work_date-beguz )
                                     enduz = normalize_time( ls_work_date-enduz ) ).
          move-corresponding ls_work_date-p2001 to <ls_it2001_mod>-mod_pskey.
          <ls_it2001_mod>-mod_pskey-infty = '2001'.
        else.
          ls_work_date-p2001-infty = '2001'.
          append_data_to_pnnnn(
            exporting
              is_data = ls_work_date-p2001
            changing
              cs_data = cs_data-delete_tab
          ).
        endif.
      endloop.

      loop at gt_leave_date into data(ls_leave_date).
        if ls_leave_date-beguz <> ls_leave_date-enduz.
          append initial line to cs_data-it2001 assigning field-symbol(<ls_it2001>).
          <ls_it2001> = value #( pernr = is_data_in-pernr
                                 begda = ls_leave_date-date
                                 endda = ls_leave_date-date
                                 subty = ls_leave_date-awart
                                 awart = ls_leave_date-awart
                                 beguz = normalize_time( ls_leave_date-beguz )
                                 enduz = normalize_time( ls_leave_date-enduz )
          ).
        endif.
      endloop.

    else.
      cs_out-code = gc_error.
      cs_out-message = |假勤类型未知|.
      return.
    endif.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->SPLITE_DATE
* +-------------------------------------------------------------------------------------------------+
* | [<-->] CT_DATA                        TYPE        TY_DATA_IN-LEAVE_LIST
* | [<-->] CS_OUT                         TYPE        TY_RESULT(optional)
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method splite_date.

    data(lt_date_tmp) = ct_data.

    refresh ct_data.

    loop at lt_date_tmp into data(ls_date_tmp).

      if ls_date_tmp-awart is initial.
        cs_out-code = gc_error.
        cs_out-message = |缺勤类型不能为空|.
        return.
      endif.

      if ls_date_tmp-begda is initial.
        cs_out-code = gc_error.
        cs_out-message = |缺勤开始日期不能为空|.
        return.
      endif.

      if ls_date_tmp-endda is initial.
        cs_out-code = gc_error.
        cs_out-message = |缺勤结束日期不能为空|.
        return.
      endif.

      if ls_date_tmp-begda > ls_date_tmp-endda.
        cs_out-code = gc_error.
        cs_out-message = |结束日期不能小于开始日期|.
        return.
      elseif ls_date_tmp-begda = ls_date_tmp-endda.

        append ls_date_tmp to ct_data.

      elseif ls_date_tmp-begda < ls_date_tmp-endda.

        data(lv_date) = ls_date_tmp-begda.

        do.

          if lv_date > ls_date_tmp-endda.
            exit.
          endif.

          data(ls_data) = value ty_leave( ).

          if lv_date = ls_date_tmp-begda.
            ls_data = value ty_leave(
                awart = ls_date_tmp-awart
                begda = lv_date
                endda = lv_date
                beguz = ls_date_tmp-beguz
                enduz = '240000'
            ).
          elseif lv_date = ls_date_tmp-endda.
            ls_data = value ty_leave(
               awart = ls_date_tmp-awart
               begda = lv_date
               endda = lv_date
               beguz = '000000'
               enduz = ls_date_tmp-enduz
            ).
          else.
            ls_data = value ty_leave(
              awart = ls_date_tmp-awart
              begda = lv_date
              endda = lv_date
              beguz = '000000'
              enduz = '240000'
            ).
          endif.

          if ls_data is not initial.
            append ls_data to ct_data.
          endif.

          lv_date = lv_date + 1.
        enddo.

      else.
        cs_out-code = gc_error.
        cs_out-message = |日期未知|.
        return.
      endif.

    endloop.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->HANDLING_DAILY_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_TYPE                        TYPE        ANY
* | [--->] IV_PERNR                       TYPE        ANY
* | [--->] IT_DATA                        TYPE        TY_DATA_IN-LEAVE_LIST
* | [<---] ET_LEAVE                       TYPE        TT_LEAVE_DATE
* | [<---] ET_WORK                        TYPE        TT_WORK_DATE
* | [<-->] CS_OUT                         TYPE        TY_RESULT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method handling_daily_data.

    refresh et_leave.
    refresh et_work.

    check cs_out-code <> gc_error.

    case iv_type.
      when '10'.
        loop at it_data into data(ls_data).

          if ls_data-beguz is initial.
            ls_data-beguz = '000000'.
          endif.

          if ls_data-enduz is initial.
            ls_data-enduz = '240000'.
          endif.

          et_leave = value #( base et_leave ( date  = ls_data-begda
                                              awart = ls_data-awart
                                              beguz = normalize_time( ls_data-beguz )
                                              enduz = normalize_time( ls_data-enduz ) ) ).
        endloop.

      when '20'.

        data: lt_cancel type tt_leave,
              lt_pa2001 type tt_pa2001.

        lt_cancel = it_data.

        loop at lt_cancel into ls_data.

          if ls_data-beguz is initial.
            ls_data-beguz = '000000'.
          endif.

          if ls_data-enduz is initial.
            ls_data-enduz = '240000'.
          endif.

          read_pa2001(
            exporting
              iv_pernr  = iv_pernr
              is_data   = ls_data
            importing
              et_pa2001 = data(lt_pa2001_tmp)
              et_leave  = data(lt_leave_tmp)
          ).

          append lines of lt_pa2001_tmp to lt_pa2001.

        endloop.

        sort lt_pa2001.
        delete adjacent duplicates from lt_pa2001 comparing all fields.
        loop at lt_pa2001 into data(ls_pa2001).

          data: lt_original_leave type tt_leave,
                lt_remain_leave   type tt_leave_date.
          clear: lt_original_leave,
                 lt_remain_leave.

          append value #(
            awart = ls_pa2001-awart
            begda = ls_pa2001-begda
            endda = ls_pa2001-endda
            beguz = ls_pa2001-beguz
            enduz = ls_pa2001-enduz
          ) to lt_original_leave.

          splite_date(
            changing
              ct_data = lt_original_leave
              cs_out  = cs_out
          ).

          check cs_out-code <> gc_error.

          data(lv_has_cancel_overlap) = abap_false.

          loop at lt_original_leave into data(ls_origin_leave).

            if ls_origin_leave-beguz is initial or ls_origin_leave-beguz = space.
              ls_origin_leave-beguz = '000000'.
            endif.

            if ls_origin_leave-enduz is initial or ls_origin_leave-enduz = space.
              ls_origin_leave-enduz = '240000'.
            endif.

            data(lt_day_segment) = value tt_leave_date(
              ( date  = ls_origin_leave-begda
                awart = ls_origin_leave-awart
                beguz = normalize_time( ls_origin_leave-beguz )
                enduz = normalize_time( ls_origin_leave-enduz ) ) ).

            loop at lt_cancel into data(ls_cancel)
              where begda = ls_origin_leave-begda
                and awart = ls_origin_leave-awart.

              data(lt_next_segment) = value tt_leave_date( ).

              loop at lt_day_segment into data(ls_segment).
                data(lv_cancel_beguz) = ls_cancel-beguz.
                data(lv_cancel_enduz) = ls_cancel-enduz.

                if lv_cancel_beguz is initial or lv_cancel_beguz = space.
                  lv_cancel_beguz = '000000'.
                endif.

                if lv_cancel_enduz is initial or lv_cancel_enduz = space.
                  lv_cancel_enduz = '240000'.
                endif.

                lv_cancel_beguz = normalize_time( lv_cancel_beguz ).
                lv_cancel_enduz = normalize_time( lv_cancel_enduz ).

                if lv_cancel_enduz <= ls_segment-beguz
                  or lv_cancel_beguz >= ls_segment-enduz.
                  append ls_segment to lt_next_segment.
                  continue.
                endif.

                lv_has_cancel_overlap = abap_true.

                if ls_segment-beguz < lv_cancel_beguz.
                  append value #(
                    date  = ls_segment-date
                    awart = ls_segment-awart
                    beguz = ls_segment-beguz
                    enduz = lv_cancel_beguz
                  ) to lt_next_segment.
                endif.

                if ls_segment-enduz > lv_cancel_enduz.
                  append value #(
                    date  = ls_segment-date
                    awart = ls_segment-awart
                    beguz = lv_cancel_enduz
                    enduz = ls_segment-enduz
                  ) to lt_next_segment.
                endif.
              endloop.

              lt_day_segment = lt_next_segment.

              if lt_day_segment is initial.
                exit.
              endif.
            endloop.

            append lines of lt_day_segment to lt_remain_leave.
          endloop.

          if lv_has_cancel_overlap is initial.
            continue.
          endif.

          handling_leave_date(
            exporting
              iv_pernr = iv_pernr
            changing
              ct_leave = lt_remain_leave
          ).

          " 销假后没有剩余请假时才删除原记录。
          if lt_remain_leave is initial.
            append initial line to et_work assigning field-symbol(<fs_work_del_full>).
            <fs_work_del_full>-action = 'D'.
            move-corresponding ls_pa2001 to <fs_work_del_full>-p2001.
            continue.
          endif.

          data ls_remain_leave type ty_leave_date.

          if lines( lt_remain_leave ) = 1.
            read table lt_remain_leave into ls_remain_leave index 1.

            append initial line to et_work assigning field-symbol(<fs_work_mod>).
            <fs_work_mod>-action = 'M'.
            <fs_work_mod>-date = ls_remain_leave-date.
            <fs_work_mod>-awart = ls_remain_leave-awart.
            <fs_work_mod>-beguz = ls_remain_leave-beguz.
            <fs_work_mod>-enduz = ls_remain_leave-enduz.
            move-corresponding ls_pa2001 to <fs_work_mod>-p2001.
          else.
            read table lt_remain_leave into ls_remain_leave index 1.

            " 多段剩余请假时，原记录先改成第一段，其余段再新增。
            append initial line to et_work assigning field-symbol(<fs_work_mod_split>).
            <fs_work_mod_split>-action = 'M'.
            <fs_work_mod_split>-date = ls_remain_leave-date.
            <fs_work_mod_split>-awart = ls_remain_leave-awart.
            <fs_work_mod_split>-beguz = ls_remain_leave-beguz.
            <fs_work_mod_split>-enduz = ls_remain_leave-enduz.
            move-corresponding ls_pa2001 to <fs_work_mod_split>-p2001.

            delete lt_remain_leave index 1.
            append lines of lt_remain_leave to et_leave.
          endif.
        endloop.

    endcase.

    if iv_type <> '20'.
      handling_leave_date(
        exporting
          iv_pernr = iv_pernr
        changing
          ct_leave = et_leave
      ).
    endif.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->HANDLING_LEAVE_DATE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_PERNR                       TYPE        ANY
* | [<-->] CT_LEAVE                       TYPE        TT_LEAVE_DATE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method handling_leave_date.

    data(lt_leave) = ct_leave.

    refresh ct_leave.
    loop at lt_leave into data(ls_leave).
      data(lv_date) = ls_leave-date.

      if ls_leave-awart in get_awart_for_natural_day( ).
        append ls_leave to ct_leave.
        continue.
      endif.

      data lv_prev_date type d.
      lv_prev_date = lv_date - 1.
      data(lt_psp) = read_timedata_psp(
        iv_pernr = iv_pernr
        iv_date  = lv_prev_date
      ).
      data(lt_psp_current) = read_timedata_psp(
        iv_pernr = iv_pernr
        iv_date  = lv_date
      ).
      append lines of lt_psp_current to lt_psp.

      loop at lt_psp into data(ls_psp) where tprog <> 'OFF'.
        if ls_psp-begzt is initial
          and ls_psp-endzt is initial.
          continue.
        endif.

        data(lv_shift_beg) = ls_psp-begzt.
        data(lv_shift_end) = ls_psp-endzt.
        data(lv_shift_cross_day) = xsdbool( ls_psp-endzt <= ls_psp-begzt ).

        if lv_shift_cross_day = abap_true.
          if ls_psp-datum = lv_prev_date.
            lv_shift_beg = '000000'.
            lv_shift_end = ls_psp-endzt.
          elseif ls_psp-datum = lv_date.
            lv_shift_beg = ls_psp-begzt.
            lv_shift_end = '240000'.
          else.
            continue.
          endif.
        elseif ls_psp-datum <> lv_date.
          continue.
        endif.

        if lv_shift_end <= lv_shift_beg.
          continue.
        endif.

        data(lv_shift_beg_norm) = normalize_time( lv_shift_beg ).
        data(lv_shift_end_norm) = normalize_time( lv_shift_end ).

        calculate_day_intersection(
          exporting
            is_segment1      = value #( date = ls_leave-date start = normalize_time( ls_leave-beguz ) end = normalize_time( ls_leave-enduz ) )
            is_segment2      = value #( date = lv_date       start = lv_shift_beg_norm                end = lv_shift_end_norm )
          importing
            et_intersect     = data(lt_time1)
            et_non_intersect = data(lt_time2)
        ).

        loop at lt_time1 into data(ls_time).
          if ls_time-start <> ls_time-end.
            data(lv_leave_beg_norm) = normalize_time( ls_time-start ).
            data(lv_leave_end_norm) = normalize_time( ls_time-end ).

            if ls_time-date = lv_date
              and lv_shift_cross_day = abap_false
              and lv_leave_beg_norm = lv_shift_beg_norm
              and lv_leave_end_norm = lv_shift_end_norm.

              data(lv_split_sec) =
                ( time_to_seconds( lv_leave_beg_norm )
                + time_to_seconds( lv_leave_end_norm ) ) div 2.
              data(lv_split_time) = seconds_to_time( lv_split_sec ).

              if lv_split_time > lv_leave_beg_norm
                and lv_split_time < lv_leave_end_norm.
                append value #(
                  date  = ls_time-date
                  beguz = lv_leave_beg_norm
                  enduz = lv_split_time
                  awart = ls_leave-awart ) to ct_leave.
                append value #(
                  date  = ls_time-date
                  beguz = lv_split_time
                  enduz = lv_leave_end_norm
                  awart = ls_leave-awart ) to ct_leave.
              else.
                append value #(
                  date  = ls_time-date
                  beguz = lv_leave_beg_norm
                  enduz = lv_leave_end_norm
                  awart = ls_leave-awart ) to ct_leave.
              endif.
            else.
              append value #(
                date  = ls_time-date
                beguz = lv_leave_beg_norm
                enduz = lv_leave_end_norm
                awart = ls_leave-awart ) to ct_leave.
            endif.
          endif.
        endloop.
      endloop.

    endloop.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->READ_PA2001
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_PERNR                       TYPE        ANY
* | [--->] IS_DATA                        TYPE        TY_LEAVE
* | [<---] ET_PA2001                      TYPE        TT_PA2001
* | [<---] ET_LEAVE                       TYPE        TT_LEAVE

  method calculate_day_intersection.

    constants lc_day_seconds type i value 86400.

    types:
      begin of ty_abs_result,
        beg_sec type i,
        end_sec type i,
        target  type c length 1,

      + conv i( lv_time_start1+4(2) ).
    data(lv_end1_sec) =
        conv i( lv_time_end1+0(2) ) * 3600
      + conv i( lv_time_end1+2(2) ) * 60
      + conv i( lv_time_end1+4(2) ).
    data(lv_start2_sec) =
        conv i( lv_time_start2+0(2) ) * 3600
      + conv i( lv_time_start2+2(2) ) * 60
      + conv i( lv_time_start2+4(2) ).
    data(lv_end2_sec) =
        conv i( lv_time_end2+0(2) ) * 3600
      + conv i( lv_time_end2+2(2) ) * 60
      + conv i( lv_time_end2+4(2) ).

    data lv_beg1_abs type i.
    data lv_end1_abs type i.
    data lv_beg2_abs type i.
    data lv_end2_abs type i.

    lv_beg1_abs = ( is_segment1-date - lv_base_date ) * lc_day_seconds + lv_start1_sec.
    lv_end1_abs = ( is_segment1-date - lv_base_date ) * lc_day_seconds + lv_end1_sec.
    lv_beg2_abs = ( is_segment2-date - lv_base_date ) * lc_day_seconds + lv_start2_sec.
    lv_end2_abs = ( is_segment2-date - lv_base_date ) * lc_day_seconds + lv_end2_sec.

    if lv_time_end1 < lv_time_start1.
      lv_end1_abs = lv_end1_abs + lc_day_seconds.
    endif.

    if lv_time_end2 < lv_time_start2.
      lv_end2_abs = lv_end2_abs + lc_day_seconds.
    endif.

    if lv_end1_abs <= lv_beg1_abs.
      return.
    endif.

    data(lv_intersect_beg) = cond i(
      when lv_beg1_abs > lv_beg2_abs then lv_beg1_abs
      else lv_beg2_abs ).
    data(lv_intersect_end) = cond i(
      when lv_end1_abs < lv_end2_abs then lv_end1_abs
      else lv_end2_abs ).

    if lv_intersect_beg < lv_intersect_end.
      append value #(
        beg_sec = lv_intersect_beg
        end_sec = lv_intersect_end
        target  = 'I' ) to lt_abs_result.

          beg_sec = lv_beg1_abs
          end_sec = lv_intersect_beg
          target  = 'N' ) to lt_abs_result.
      endif.

      if lv_intersect_end < lv_end1_abs.
        append value #(
          beg_sec = lv_intersect_end
          end_sec = lv_end1_abs
          target  = 'N' ) to lt_abs_result.
      endif.
    else.
      append value #(
        beg_sec = lv_beg1_abs
        end_sec = lv_end1_abs
        target  = 'N' ) to lt_abs_result.

        data(lv_day_end) = ( lv_day_index + 1 ) * lc_day_seconds.
        data(lv_part_end) = cond i(
          when ls_abs_result-end_sec < lv_day_end then ls_abs_result-end_sec
          else lv_day_end ).
        data(lv_result_date) = lv_base_date + lv_day_index.
        data(lv_start_rem) = lv_cursor mod lc_day_seconds.
        data(lv_end_rem) = lv_part_end mod lc_day_seconds.

        data(lv_start_hour) = lv_start_rem div 3600.
        data(lv_start_minute) = ( lv_start_rem mod 3600 ) div 60.
        data(lv_start_second) = lv_start_rem mod 60.
        data(lv_end_hour) = lv_end_rem div 3600.
        data(lv_end_minute) = ( lv_end_rem mod 3600 ) div 60.
        data(lv_end_second) = lv_end_rem mod 60.

        data(lv_start_time) = normalize_time(
          |{ lv_start_hour width = 2 pad = '0' }{ lv_start_minute width = 2 pad = '0' }{ lv_start_second width = 2 pad = '0' }| ).
        data(lv_end_time) = normalize_time(
          |{ lv_end_hour width = 2 pad = '0' }{ lv_end_minute width = 2 pad = '0' }{ lv_end_second width = 2 pad = '0' }| ).

        if lv_part_end = lv_day_end.
          lv_end_time = '240000'.
        endif.

        if ls_abs_result-target = 'I'.
          append value #(
            date  = lv_result_date
            start = lv_start_time
            end   = lv_end_time ) to et_intersect.
        else.
          append value #(
            date  = lv_result_date
            start = lv_start_time
            end   = lv_end_time ) to et_non_intersect.
        endif.

        lv_cursor = lv_part_end.
      endwhile.
    endloop.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_LEAVE_REQUEST->NORMALIZE_TIME
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_TIME                        TYPE        ANY
* | [<-()] RV_TIME                        TYPE        TIMS
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method normalize_time.

    data(lv_time) = conv string( iv_time ).
    condense lv_time no-gaps.
    replace all occurrences of ':' in lv_time with ''.

    if lv_time is initial.
      clear rv_time.
      return.
    endif.

    data(lv_len) = strlen( lv_time ).

    if lv_len > 6.
      lv_time = lv_time+0(6).
      lv_len = 6.
    endif.

    case lv_len.
      when 6.
        data(lv_hour6) = conv i( lv_time+0(2) ).
        if lv_hour6 > 24.
          lv_time = |0{ lv_time+0(5) }|.
        endif.

      when 5.
        lv_time = |0{ lv_time }|.

      when 4.
        data(lv_hour4) = conv i( lv_time+0(2) ).
        data(lv_min4) = conv i( lv_time+2(2) ).

        if lv_hour4 <= 24 and lv_min4 <= 59.
          lv_time = |{ lv_time }00|.
        else.
          lv_time = |0{ lv_time+0(3) }00|.
        endif.

      when 3.
        lv_time = |0{ lv_time }00|.

      when 2.
        lv_time = |{ lv_time }0000|.

      when 1.
        lv_time = |0{ lv_time }0000|.

      when others.
        while strlen( lv_time ) < 6.
          lv_time = |0{ lv_time }|.
        endwhile.
    endcase.

    rv_time = lv_time.

  endmethod.
ENDCLASS.

* | [--->] IV_CODE                        TYPE        ANY
* | [<-()] RT_RESULT                      TYPE        TT_MASSN
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method get_massn_for_flow.

    case iv_code.
      when gc_onboard_process.
        rt_result = value #( sign = 'I' option = 'EQ' ( low = 'ZA' ) ( low = 'ZY' ) ).
      when gc_exit_process.

      when others.
        rt_result = value #( sign = 'I' option = 'BT' ( low = '99' high = '11' )  ).
    endcase.

  endmethod.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_INFOTYPE_OPERATION->INIT_INFTY_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_INFTY                       TYPE        ANY
* | [--->] IV_DELETE                      TYPE        FLAG
* | [--->] IV_SECONDARY_INFTY             TYPE        ANY(optional)
* | [<-->] CS_DATA                        TYPE        TY_DATA
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method init_infty_data.

    "--------------------> 初始化数据源
    assign component |IT{ iv_infty }| of structure cs_data to field-symbol(<fs_data>).
    check <fs_data> is assigned and <fs_data> is not initial.

    append initial line to gt_write_infty assigning field-symbol(<fs_infty>).
    <fs_infty>-infty = iv_infty.
    <fs_infty>-second_infty = iv_secondary_infty.

    "--------------------> 获取子类型字段
    select single namst into @data(lv_namst) from t582a where infty = @iv_infty.
    if lv_namst is not initial.
      split lv_namst at '-' into data(lv_pxxxx) data(lv_subty).
    endif.

    "--------------------> 获取要删除的信息类型数据
    if iv_delete eq abap_true.
      data lt_pnnnn type ref to data.
      data(lv_tabname) = |P{ iv_infty }|.

      create data lt_pnnnn type table of (lv_tabname).
      assign lt_pnnnn->* to field-symbol(<fs_pnnnn_tab>).

      get_exists_data(
        exporting
          iv_infty = iv_infty
          iv_pernr = cs_data-pernr
        importing
          it_data  = <fs_pnnnn_tab> ).

      loop at <fs_pnnnn_tab> assigning field-symbol(<fs_pnnnn>).
        assign component 'INFTY' of structure <fs_pnnnn> to field-symbol(<fs_field>).
        if <fs_field> is assigned.
          <fs_field> = iv_infty.
        endif.

        if is_data_deleted( iv_infty = iv_infty is_data = <fs_pnnnn> ).
          append_data_to_pnnnn(
            exporting
              is_data = <fs_pnnnn>
            changing
              cs_data = cs_data-delete_tab ).
        endif.
      endloop.
    endif.

    "--------------------> 获取数据源类型
    if cl_abap_complexdescr=>get_data_type_kind( <fs_data> ) eq cl_abap_complexdescr=>typekind_table.
      <fs_infty>-tab_flag = abap_true.
      loop at <fs_data> assigning field-symbol(<fs_line>).
        conv_data_to_prelp(
          exporting
            iv_infty           = iv_infty
            iv_pernr           = cs_data-pernr
            iv_begda           = cs_data-begda
            iv_secondary_infty = iv_secondary_infty
            iv_subty           = lv_subty
          changing
            cs_line            = <fs_line>
            cs_data            = cs_data ).
      endloop.
    else.
      conv_data_to_prelp(
        exporting
          iv_infty           = iv_infty
          iv_pernr           = cs_data-pernr
          iv_begda           = cs_data-begda
          iv_secondary_infty = iv_secondary_infty
          iv_subty           = lv_subty
        changing
          cs_line            = <fs_data>
          cs_data            = cs_data ).
    endif.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_INFOTYPE_OPERATION->CONV_DATA_TO_PRELP
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_INFTY                       TYPE        ANY
* | [--->] IV_PERNR                       TYPE        ANY
* | [--->] IV_BEGDA                       TYPE        ANY
* | [--->] IV_SECONDARY_INFTY             TYPE        ANY(optional)
* | [--->] IV_SUBTY                       TYPE        ANY(optional)
* | [<-->] CS_LINE                        TYPE        ANY
* | [<-->] CS_DATA                        TYPE        TY_DATA
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method conv_data_to_prelp.

    data: ls_pskey type pskey.

    if iv_secondary_infty is not initial.
      assign component 'FIRST' of structure cs_line to field-symbol(<fs_first>).
      assign component 'SECOND' of structure cs_line to field-symbol(<fs_second>).
    else.
      assign cs_line to <fs_first>.
    endif.

    move-corresponding <fs_first> to ls_pskey.

    ls_pskey-infty = iv_infty.
    ls_pskey-pernr = iv_pernr.

    if iv_subty is not initial.
      assign component iv_subty of structure <fs_first> to field-symbol(<fs_subty>).
      ls_pskey-subty = <fs_subty>.
    endif.

    if ls_pskey-begda is initial.
      ls_pskey-begda = iv_begda.
    endif.

    if ls_pskey-endda is initial.
      ls_pskey-endda = '99991231'.
    endif.

    move-corresponding ls_pskey to <fs_first>.

    data: lo_struct_descr type ref to cl_abap_structdescr,
          ls_pnnnn        type ref to data,
          ls_spnnnn       type ref to data.

    data(lv_first) = |P{ iv_infty }|.
    data(lv_second) = |P{ iv_secondary_infty }|.

    field-symbols: <fs_pnnnn>  type any,
                   <fs_spnnnn> type any.

    lo_struct_descr ?= cl_abap_typedescr=>describe_by_name( lv_first ).
    create data ls_pnnnn type handle lo_struct_descr.
    assign ls_pnnnn->* to <fs_pnnnn>.

    move-corresponding <fs_first> to <fs_pnnnn>.

    if <fs_second> is assigned.

      lo_struct_descr ?= cl_abap_typedescr=>describe_by_name( lv_second ).
      create data ls_spnnnn type handle lo_struct_descr.
      assign ls_spnnnn->* to <fs_spnnnn>.

      ls_pskey-infty = iv_secondary_infty.
      move-corresponding ls_pskey to <fs_second>.
      move-corresponding <fs_second> to <fs_spnnnn>.

      append_data_to_pnnnn(
        exporting
          is_data        = <fs_pnnnn>
          is_second_data = <fs_spnnnn>
        changing
          cs_data        = cs_data-insert_tab ).
    else.
      append_data_to_pnnnn(
        exporting
          is_data = <fs_pnnnn>
        changing
          cs_data = cs_data-insert_tab ).
    endif.

  endmethod.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_INFOTYPE_OPERATION=>CHANGE_DATE_FOR_DTTYPE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_DAR01                       TYPE        ANY
* | [--->] IV_DAT01                       TYPE        ANY
* | [<-->] CS_DATA                        TYPE        ANY
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method change_date_for_dttype.

    data: ls_p0041           type p0041,
          lv_index           type n length 2,
          lv_field_dar_write type fieldname,
          lv_field_dat_write type fieldname,
          lv_field_dar       type fieldname,
          lv_field_dat       type fieldname,
          lv_field_val       type date.

    move-corresponding cs_data to ls_p0041.
    do 24 times.
      lv_index = sy-index.

      lv_field_dar = |DAR{ lv_index alpha = in }|.
      lv_field_dat = |DAT{ lv_index alpha = in }|.

      assign component lv_field_dar of structure ls_p0041 to field-symbol(<fs_dar_value>).
      if iv_dar01 = <fs_dar_value>.
        data(lv_exists) = abap_true.
        assign component lv_field_dat of structure ls_p0041 to field-symbol(<fs_dat_value>).
        <fs_dat_value> = iv_dat01.
        exit.
      endif.

      "--------------------> 记录一个未写值的字段
      if lv_field_dar_write is initial and <fs_dar_value> is initial.
        lv_field_dar_write = lv_field_dar.
        lv_field_dat_write = lv_field_dat.
      endif.

    enddo.

    "--------------------> 如果日期类型不存在，则写在新字段上
    if lv_exists = abap_false.
      assign component lv_field_dar_write of structure ls_p0041 to field-symbol(<fs_dar>).
      <fs_dar> = iv_dar01.

      assign component lv_field_dat_write of structure ls_p0041 to field-symbol(<fs_dat>).
      <fs_dat> = iv_dat01.
    endif.

    move-corresponding ls_p0041 to cs_data.

  endmethod.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_INFOTYPE_OPERATION->HIRE_EMPLOYEE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_DATA                        TYPE        TY_DATA
* | [--->] IT_HIRE_FIELDNAMES             TYPE        HRPAD_T_HIRE_FIELDNAMES
* | [<-->] CS_OUT                         TYPE        TY_RESULT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method hire_employee.

    data: lt_return_tab        type hrpad_return_tab,
          lt_bapipakey_tab     type hrpad_bapipakey_tab,
          lv_is_ok             type boole_d,
          lt_modified_keys_tab type hrpad_pskey_tab.

    check cs_out-code ne gc_error.

    call function 'HR_PAD_HIRE_EMPLOYEE'
      exporting
        employeenumber     = is_data-pernr
        hiringdate         = is_data-begda
        actiontype         = is_data-massn
        reasonforaction    = is_data-massg
        pnnnn_tab          = is_data-insert_tab
        nocommit           = is_data-nocommit
        it_hire_fieldnames = it_hire_fieldnames
      importing
        return_tab         = lt_return_tab
        bapipakey_tab      = lt_bapipakey_tab
        is_ok              = lv_is_ok
        modified_keys_tab  = lt_modified_keys_tab.

    if lv_is_ok is initial.
      cs_out-code = gc_error.
      cs_out-message = '创建失败'.

      loop at lt_return_tab into data(ls_return).
        if ls_return-id = 'RP' and  ls_return-number = '135'.
          ls_return-message = |薪资计算中，功能暂不可用，请稍后再试。This function is temporarily unavailable during payroll processing. Please try again later.|.
        endif.

        append initial line to cs_out-error_data assigning field-symbol(<fs_line>).
        <fs_line>-type = ls_return-type.
        <fs_line>-message = ls_return-message.
        loop at ls_return-field_list into data(ls_field).
          split ls_field at '-' into data(lv_table) data(lv_field).
          append initial line to <fs_line>-error_fields assigning field-symbol(<fs_field>).
          if lv_field is not initial.
            <fs_field> = |it{ lv_table+1(4) }-{ to_lower( lv_field ) }|.
          else.
            <fs_field> = |{ to_lower( lv_table ) }|.
          endif.
        endloop.
      endloop.
    endif.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_INFOTYPE_OPERATION->PROCESS_ACTION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_DATA                        TYPE        TY_DATA
* | [<-->] CS_OUT                         TYPE        TY_RESULT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method process_action.

    data: ls_return type bapireturn1.

    check cs_out-code ne gc_error.

    call function 'BAPI_EMPLOYEE_ENQUEUE'
      exporting
        number = is_data-pernr
      importing
        return = ls_return.

    if ls_return-type ca 'EAX'.
      cs_out-code = gc_error.
      cs_out-message = ls_return-message.
      return.
    endif.

    call function 'HR_INITIALIZE_BUFFER'.

    call function 'HR_PSBUFFER_INITIALIZE'.

    "--------------------> 删除数据
    loop at is_data-delete_tab into data(ls_delete).
      if delete_infotype(
         exporting
           is_data   = ls_delete
         changing
           cs_out    = cs_out
       ).
        exit.
      endif.
    endloop.

    "--------------------> 插入/更改数据
    loop at gt_write_infty into data(ls_write_infty).
      if insert_infotype(
           exporting
             is_infty = ls_write_infty
             is_data  = is_data
           changing
             cs_out   = cs_out ).
        exit.
      endif.
    endloop.

    if is_data-create_1001 eq abap_true.
      operation_infotyp_1001_a008(
        exporting
          is_data = is_data
        changing
          cs_out  = cs_out ).
    endif.

    call function 'BAPI_EMPLOYEE_DEQUEUE'
      exporting
        number = is_data-pernr.

    if cs_out-code is initial and is_data-nocommit is initial.

      call function 'BAPI_TRANSACTION_COMMIT'
        exporting
          wait = abap_true.
    else.
      call function 'BAPI_TRANSACTION_ROLLBACK'.
    endif.

  endmethod.

* <SIGNATURE>---------------------------------------------------------------------------------------+

  method insert_infotype.

    assign component |IT{ is_infty-infty }| of structure is_data to field-symbol(<fs_data>).

    if is_infty-tab_flag eq abap_true.
      loop at <fs_data> assigning field-symbol(<fs_line>).
        rv_result = get_line_of_infotype(
          exporting
            is_infty = is_infty
            is_data  = <fs_line>
          changing
            cs_out   = cs_out ).
        if rv_result eq abap_true.
          exit.
        endif.
      endloop.
    else.
      rv_result = get_line_of_infotype(
        exporting
          is_infty = is_infty
          is_data  = <fs_data>
        changing
          cs_out   = cs_out ).
    endif.

  endmethod.

* <SIGNATURE>---------------------------------------------------------------------------------------+

  method get_line_of_infotype.

    data: lv_operation type actio value 'INS',

      assign component 'FIRST' of structure is_data to field-symbol(<fs_first>).
      assign component 'SECOND' of structure is_data to field-symbol(<fs_second>).
      rv_result = infotype_operation(

          iv_operation   = lv_operation

        changing

            ls_data         type ref to data.

      data(lv_tabname) = |P{ is_infty-infty }|.
      lo_struct_descr ?= cl_abap_typedescr=>describe_by_name( lv_tabname ).

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZCL_INFOTYPE_OPERATION->INFOTYPE_OPERATION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_OPERATION                   TYPE        ACTIO
* | [--->] IS_PSKEY                       TYPE        PSKEY(optional)
* | [--->] IS_DATA                        TYPE        ANY
* | [--->] IS_SECOND_DATA                 TYPE        ANY(optional)
* | [<-->] CS_OUT                         TYPE        TY_RESULT
* | [<-()] RV_ERROR                       TYPE        BOOLE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method infotype_operation.

    data: ls_pskey  type pskey,
          ls_return type bapireturn1.

    if is_pskey is initial.
      move-corresponding is_data to ls_pskey.
    else.
      move-corresponding is_pskey to ls_pskey.
    endif.

    if is_second_data is initial.
      call function 'HR_INFOTYPE_OPERATION'
        exporting
          infty         = ls_pskey-infty
          number        = ls_pskey-pernr
          subtype       = ls_pskey-subty
          objectid      = ls_pskey-objps
          lockindicator = ls_pskey-sprps
          validityend   = ls_pskey-endda
          validitybegin = ls_pskey-begda
          recordnumber  = ls_pskey-seqnr
          operation     = iv_operation
          nocommit      = 'X'
          record        = is_data
        importing
          return        = ls_return.
    endif.

    if is_second_data is not initial.
      call function 'HR_INFOTYPE_OPERATION'
        exporting
          infty            = ls_pskey-infty
          number           = ls_pskey-pernr
          subtype          = ls_pskey-subty
          objectid         = ls_pskey-objps
          lockindicator    = ls_pskey-sprps
          validityend      = ls_pskey-endda
          validitybegin    = ls_pskey-begda
          recordnumber     = ls_pskey-seqnr
          operation        = iv_operation
          nocommit         = 'X'
          record           = is_data
          view_identifier  = '28'
          secondary_record = is_second_data
        importing
          return           = ls_return.
    endif.

    if ls_return-type ca 'EAX'.
