## 消息类定义

```abap
class zcl_message definition
  public
  final
  create private .

  public section.
    constants: error   type c value 'E',
               success type c value 'S',
               waring  type c value 'W',
               infor   type c value 'I'.

    constants: back     type char1 value '1',
               exit     type char1 value '2',
               cancel   type char1 value '3',
               continue type char1 value '4',
               external type char1 value '5',
               others   type char1 value '0'.

    types: tt_fields type standard table of fieldname with default key.

    class-methods factory
      importing
                iv_uuid          type any optional
      exporting
                ev_uuid          type sysuuid_c32
      returning value(ro_result) type ref to zcl_message.

    methods append_message
      importing
                iv_fieldname     type any optional
                iv_msgty         type any default 'E'
                iv_msgid         type any
                iv_msgno         type any
                iv_msgv1         type any optional
                iv_msgv2         type any optional
                iv_msgv3         type any optional
                iv_msgv4         type any optional
      returning value(rv_result) type string.

    methods is_exists_errordata
      returning value(rv_result) type boole.

    methods get_error_fields
      returning value(rt_result) type tt_fields.

    methods show_message
      importing
                iv_one_msg_direct type c optional
      returning value(rv_result)  type char1.

    methods refresh.


  protected section.
  private section.
    class-data: begin of gs_object,
                  uuid   type sysuuid_c32,
                  object type ref to zcl_message,
                end of gs_object.
    class-data: gt_object like hashed table of gs_object with unique key uuid.

    data: begin of gs_message,
            fieldname type fieldname,
            message   type rs_s_msg,
          end of gs_message.

    data: gt_message like table of gs_message.

endclass.
```

## 消息类实现

```abap
class zcl_message implementation.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Static Public Method ZCL_MESSAGE=>FACTORY
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_UUID                        TYPE        ANY(optional)
* | [<---] EV_UUID                        TYPE        SYSUUID_C32
* | [<-()] RO_RESULT                      TYPE REF TO ZCL_MESSAGE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method factory.

    data: lv_uuid type sysuuid_c32.

    lv_uuid = iv_uuid.
    if lv_uuid is initial.
      try.
          lv_uuid = cl_system_uuid=>if_system_uuid_static~create_uuid_c32( ).
        catch cx_root into data(lo_root).
          message e001(00) with lo_root->get_longtext( ).
      endtry.
    endif.

    read table gt_object into gs_object with key uuid = lv_uuid.
    if sy-subrc <> 0.
      gs_object-uuid = lv_uuid.
      gs_object-object = new zcl_message( ).
      insert gs_object into table gt_object.
    endif.

    ev_uuid   = lv_uuid.
    ro_result = gs_object-object.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_MESSAGE->APPEND_MESSAGE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_FIELDNAME                   TYPE        ANY(optional)
* | [--->] IV_MSGTY                       TYPE        ANY (default ='E')
* | [--->] IV_MSGID                       TYPE        ANY
* | [--->] IV_MSGNO                       TYPE        ANY
* | [--->] IV_MSGV1                       TYPE        ANY(optional)
* | [--->] IV_MSGV2                       TYPE        ANY(optional)
* | [--->] IV_MSGV3                       TYPE        ANY(optional)
* | [--->] IV_MSGV4                       TYPE        ANY(optional)
* | [<-()] RV_RESULT                      TYPE        STRING
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method append_message.

    data: lt_longtext type table of tline,
          lv_message  type syst_lisel.


    gt_message = value #(
      base gt_message ( fieldname = iv_fieldname
                        message   = value #( msgty = iv_msgty
                                             msgid = iv_msgid
                                             msgno = iv_msgno
                                             msgv1 = iv_msgv1
                                             msgv2 = iv_msgv2
                                             msgv3 = iv_msgv3
                                             msgv4 = iv_msgv4 )
      ) ).

    call function 'RPY_MESSAGE_COMPOSE'
      exporting
        message_id        = conv syst_msgid( iv_msgid )
        message_number    = conv syst_msgno( iv_msgno )
        message_var1      = conv syst_msgv( iv_msgv1 )
        message_var2      = conv syst_msgv( iv_msgv2 )
        message_var3      = conv syst_msgv( iv_msgv3 )
        message_var4      = conv syst_msgv( iv_msgv4 )
      importing
        message_text      = lv_message
      tables
        longtext          = lt_longtext
      exceptions
        message_not_found = 1
        others            = 2.

    rv_result = lv_message.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_MESSAGE->IS_EXISTS_ERRORDATA
* +-------------------------------------------------------------------------------------------------+
* | [<-()] RV_RESULT                      TYPE        BOOLE
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method is_exists_errordata.

    if line_exists( gt_message[ message-msgty = error ] ).
      rv_result = abap_true.
    else.
      rv_result = abap_false.
    endif.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_MESSAGE->GET_ERROR_FIELDS
* +-------------------------------------------------------------------------------------------------+
* | [<-()] RT_RESULT                      TYPE        TT_FIELDS
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method get_error_fields.

    data: lt_fields type table of fieldname.

    loop at gt_message into data(ls_message).
      lt_fields = value #( base lt_fields ( ls_message-fieldname ) ).
    endloop.

    rt_result = lt_fields.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_MESSAGE->SHOW_MESSAGE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_ONE_MSG_DIRECT              TYPE        C(optional)
* | [<-()] RV_RESULT                      TYPE        CHAR1
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method show_message.

    data: lv_one_msg_direct type flag,
          ls_exit_command   type bal_s_excm,
          lt_message        type rs_t_msg.

    loop at gt_message into data(ls_message).
      append ls_message-message to lt_message.
    endloop.

    if lt_message is not initial.
      lv_one_msg_direct = iv_one_msg_direct.
      cl_epic_ui_services=>show_messages_with_alog(
        exporting
          it_messages       = lt_message
          iv_one_msg_direct = lv_one_msg_direct
        importing
          es_exit_command   = ls_exit_command
      ).
    endif.

    case 'X'.
      when ls_exit_command-back.
        rv_result = back.
      when ls_exit_command-cancel.
        rv_result = cancel.
      when ls_exit_command-continue.
        rv_result = continue.
      when ls_exit_command-exit.
        rv_result = exit.
      when ls_exit_command-external.
        rv_result = external.
      when others.
        rv_result = others.
    endcase.

  endmethod.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_MESSAGE->REFRESH
* +-------------------------------------------------------------------------------------------------+
* +--------------------------------------------------------------------------------------</SIGNATURE>
  method refresh.

    refresh gt_message.

  endmethod.
endclass.
```
