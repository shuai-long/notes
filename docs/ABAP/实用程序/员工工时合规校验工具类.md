  method get_work_shift_data.

    rt_result = mt_shift.

    loop at mt_leave into data(ls_leave).
      read table rt_result assigning field-symbol(<ls_shift>)
        with key pernr = ls_leave-pernr.
      if sy-subrc <> 0.
        continue.
      endif.

      append value #(
        beg_date = ls_leave-beg_date
        beg_time = ls_leave-beg_time
        end_date = ls_leave-end_date
        end_time = ls_leave-end_time
      ) to <ls_shift>-leaves.
    endloop.

  endmethod.

  method get_daily_work_time_by_data.

    data(lt_leave_interval) = build_leave_intervals( it_shift ).

    loop at it_shift into data(ls_shift).
      data(lv_seconds) = get_shift_work_seconds(
        is_shift          = ls_shift
        it_leave_interval = lt_leave_interval ).

      read table rt_result assigning field-symbol(<ls_day>)
        with key pernr = ls_shift-pernr
                 work_date = ls_shift-work_date.
      if sy-subrc = 0.
        add lv_seconds to <ls_day>-work_seconds.
        <ls_day>-work_hours = seconds_to_hours( <ls_day>-work_seconds ).
      else.
        append value #(
          pernr        = ls_shift-pernr
          work_date    = ls_shift-work_date
          work_seconds = lv_seconds
          work_hours   = seconds_to_hours( lv_seconds )
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by pernr work_date.

  endmethod.

  method get_period_work_time_by_data.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        it_shift     = it_shift
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lt_day_work) = get_daily_work_time_by_data( it_shift ).

    loop at lt_day_work into data(ls_day_work)
      where work_date >= lv_date_from
        and work_date <= lv_date_to.

      if iv_pernr is not initial and ls_day_work-pernr <> iv_pernr.
        continue.
      endif.

      read table rt_result assigning field-symbol(<ls_period>)
        with key pernr = ls_day_work-pernr.
      if sy-subrc = 0.
        add ls_day_work-work_seconds to <ls_period>-work_seconds.
        <ls_period>-work_hours = seconds_to_hours( <ls_period>-work_seconds ).
      else.
        append value #(
          pernr        = ls_day_work-pernr
          date_from    = lv_date_from
          date_to      = lv_date_to
          work_seconds = ls_day_work-work_seconds
          work_hours   = seconds_to_hours( ls_day_work-work_seconds )
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by pernr.

  endmethod.

  method get_shift_rest_time_by_data.

    data lt_shift_abs type tt_abs_interval.

    loop at it_shift into data(ls_shift).
      get_shift_bounds(
        exporting
          is_shift   = ls_shift
        importing
          ev_beg_sec = data(lv_beg_sec)
          ev_end_sec = data(lv_end_sec) ).

      append value #(
        pernr     = ls_shift-pernr
        work_date = ls_shift-work_date
        beg_sec   = lv_beg_sec
        end_sec   = lv_end_sec
      ) to lt_shift_abs.
    endloop.

    sort lt_shift_abs by pernr beg_sec end_sec.

    data ls_prev type ty_abs_interval.
    data lv_has_prev type abap_bool.

    loop at lt_shift_abs into data(ls_next).
      if lv_has_prev = abap_true and ls_prev-pernr = ls_next-pernr.
        data(lv_rest_seconds) = ls_next-beg_sec - ls_prev-end_sec.

        seconds_to_datetime(
          exporting
            iv_seconds = ls_prev-end_sec
          importing
            ev_date    = data(lv_prev_end_date)
            ev_time    = data(lv_prev_end_time) ).

        seconds_to_datetime(
          exporting
            iv_seconds = ls_next-beg_sec
          importing
            ev_date    = data(lv_next_beg_date)
            ev_time    = data(lv_next_beg_time) ).

        append value #(
          pernr          = ls_next-pernr
          prev_work_date = ls_prev-work_date
          prev_end_date  = lv_prev_end_date
          prev_end_time  = lv_prev_end_time
          next_work_date = ls_next-work_date
          next_beg_date  = lv_next_beg_date
          next_beg_time  = lv_next_beg_time
          rest_seconds   = lv_rest_seconds
          rest_hours     = seconds_to_hours( lv_rest_seconds )
        ) to rt_result.
      endif.

      ls_prev = ls_next.
      lv_has_prev = abap_true.
    endloop.

  endmethod.

    data(lv_end_sec) = to_seconds( iv_date = lv_day_after_to iv_time = '000000' ).

      it_work_interval = lt_work_interval

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lv_daily_seconds) = hours_to_seconds( iv_daily_hours ).

    data(lt_day_work) = get_daily_work_time_by_data( it_shift ).
    loop at lt_day_work into data(ls_day_work)
      where work_date >= lv_date_from
        and work_date <= lv_date_to.

      if iv_pernr is not initial and ls_day_work-pernr <> iv_pernr.
        continue.
      endif.

      if ls_day_work-work_seconds > lv_daily_seconds.
        append value #(
          pernr           = ls_day_work-pernr
          rule_code       = 'DAILY_WORK'
          work_date       = ls_day_work-work_date
          actual_seconds  = ls_day_work-work_seconds
          actual_hours    = ls_day_work-work_hours
          limit_seconds   = lv_daily_seconds
          limit_hours     = iv_daily_hours
          message         = |员工 { ls_day_work-pernr } 日期 { ls_day_work-work_date } 工作时长超过每日上限|
        ) to rt_result.
      endif.
    endloop.

  endmethod.

  method validate_avg_weekly_work_by_data.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        it_shift     = it_shift
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lv_weekly_seconds) = hours_to_seconds( iv_weekly_hours ).

    data(lt_period_work) = get_period_work_time_by_data(
      it_shift     = it_shift
      iv_date_from = lv_date_from
      iv_date_to   = lv_date_to
      iv_pernr     = iv_pernr ).

    data(lv_days) = lv_date_to - lv_date_from + 1.
    if lv_days > 0.
      loop at lt_period_work into data(ls_period_work).
        data(lv_avg_week_seconds) = ls_period_work-work_seconds * 7 / lv_days.
        if lv_avg_week_seconds > lv_weekly_seconds.
          append value #(
            pernr           = ls_period_work-pernr
            rule_code       = 'AVG_WEEKLY_WORK'
            period_beg_date = lv_date_from
            period_end_date = lv_date_to
            actual_seconds  = lv_avg_week_seconds
            actual_hours    = seconds_to_hours( lv_avg_week_seconds )
            limit_seconds   = lv_weekly_seconds
            limit_hours     = iv_weekly_hours
            message         = |员工 { ls_period_work-pernr } 平均每周工作时长超过上限|
          ) to rt_result.
        endif.
      endloop.
    endif.

  endmethod.

  method validate_shift_rest_by_data.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        it_shift     = it_shift
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lv_min_shift_rest_seconds) = hours_to_seconds( iv_min_shift_rest_hours ).

    data(lt_shift_rest) = get_shift_rest_time_by_data( it_shift ).
    loop at lt_shift_rest into data(ls_shift_rest)
      where next_work_date >= lv_date_from
        and next_work_date <= lv_date_to.

      if iv_pernr is not initial and ls_shift_rest-pernr <> iv_pernr.
        continue.
      endif.

      if ls_shift_rest-rest_seconds < lv_min_shift_rest_seconds.
        append value #(
          pernr           = ls_shift_rest-pernr
          rule_code       = 'SHIFT_REST'
          work_date       = ls_shift_rest-next_work_date
          actual_seconds  = ls_shift_rest-rest_seconds
          actual_hours    = ls_shift_rest-rest_hours
          limit_seconds   = lv_min_shift_rest_seconds
          limit_hours     = iv_min_shift_rest_hours
          message         = |员工 { ls_shift_rest-pernr } 连续两个班次之间休息时间不足|
        ) to rt_result.
      endif.
    endloop.

  endmethod.

  method validate_rest_day_by_data.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        it_shift     = it_shift
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lv_rest_day_seconds) = hours_to_seconds( iv_rest_day_hours ).

    if iv_rest_window_days > 0.
      data(lt_work_interval) = build_work_intervals( it_shift ).
      data(lt_pernr) = get_pernr_keys(
        it_shift = it_shift
        iv_pernr = iv_pernr ).

      data(lv_total_days) = lv_date_to - lv_date_from + 1.
      if lv_total_days >= iv_rest_window_days.
        data lv_window_beg_date type d.
        data lv_last_window_beg_date type d.
        data lv_window_end_date type d.

        lv_window_beg_date = lv_date_from.
        lv_last_window_beg_date = lv_date_to - iv_rest_window_days + 1.

        while lv_window_beg_date <= lv_last_window_beg_date.
          lv_window_end_date = lv_window_beg_date + iv_rest_window_days.
          data(lv_window_beg_sec) = to_seconds( iv_date = lv_window_beg_date iv_time = '000000' ).
          data(lv_window_end_sec) = to_seconds( iv_date = lv_window_end_date iv_time = '000000' ).

          data(lt_longest_rest) = get_longest_rest_for_bounds(
            it_work_interval = lt_work_interval
            it_pernr         = lt_pernr
            iv_beg_sec       = lv_window_beg_sec
            iv_end_sec       = lv_window_end_sec ).

          loop at lt_longest_rest into data(ls_longest_rest).
            if ls_longest_rest-rest_seconds < lv_rest_day_seconds.
              append value #(
                pernr           = ls_longest_rest-pernr
                rule_code       = 'REST_DAY'
                period_beg_date = lv_window_beg_date
                period_end_date = lv_window_end_date - 1
                actual_seconds  = ls_longest_rest-rest_seconds
                actual_hours    = ls_longest_rest-rest_hours
                limit_seconds   = lv_rest_day_seconds
                limit_hours     = iv_rest_day_hours
                message         = |员工 { ls_longest_rest-pernr } 连续 { iv_rest_window_days } 日内没有满足要求的连续休息|
              ) to rt_result.
            endif.
          endloop.

          lv_window_beg_date = lv_window_beg_date + 1.
        endwhile.
      endif.
    endif.

  endmethod.

  method validate_by_data.

    data lt_violation type tt_violation.

    lt_violation = validate_daily_work_by_data(
      it_shift       = it_shift
      iv_date_from   = iv_date_from
      iv_date_to     = iv_date_to
      iv_pernr       = iv_pernr
      iv_daily_hours = iv_daily_hours ).
    append lines of lt_violation to rt_result.

    lt_violation = validate_avg_weekly_work_by_data(
      it_shift        = it_shift
      iv_date_from    = iv_date_from
      iv_date_to      = iv_date_to
      iv_pernr        = iv_pernr
      iv_weekly_hours = iv_weekly_hours ).
    append lines of lt_violation to rt_result.

    lt_violation = validate_shift_rest_by_data(
      it_shift                = it_shift
      iv_date_from            = iv_date_from
      iv_date_to              = iv_date_to
      iv_pernr                = iv_pernr
      iv_min_shift_rest_hours = iv_min_shift_rest_hours ).
    append lines of lt_violation to rt_result.

    lt_violation = validate_rest_day_by_data(
      it_shift            = it_shift
      iv_date_from        = iv_date_from
      iv_date_to          = iv_date_to
      iv_pernr            = iv_pernr
      iv_rest_window_days = iv_rest_window_days
      iv_rest_day_hours   = iv_rest_day_hours ).
    append lines of lt_violation to rt_result.

  endmethod.

  method get_period_dates.

    ev_date_from = iv_date_from.
    ev_date_to   = iv_date_to.

    if ev_date_from is initial or ev_date_to is initial.
      loop at it_shift into data(ls_shift).
        if ev_date_from is initial or ls_shift-work_date < ev_date_from.
          ev_date_from = ls_shift-work_date.
        endif.
        if ev_date_to is initial or ls_shift-work_date > ev_date_to.
          ev_date_to = ls_shift-work_date.
        endif.
      endloop.
    endif.

  endmethod.

  method get_pernr_keys.

    if iv_pernr is not initial.
      insert value #( pernr = iv_pernr ) into table rt_result.
      return.
    endif.

    loop at it_shift into data(ls_shift).
      insert value #( pernr = ls_shift-pernr ) into table rt_result.
    endloop.

  endmethod.

  method get_shift_bounds.

    data(lv_beg_date) = is_shift-work_date.
    data(lv_end_date) = is_shift-end_date.

    if lv_end_date is initial.
      lv_end_date = is_shift-work_date.
      if is_shift-end_time <= is_shift-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_shift-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_shift-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_interval_bounds.

    data: lv_shift_beg_sec type ty_seconds,
          lv_shift_end_sec type ty_seconds.

    get_shift_bounds(
      exporting
        is_shift   = is_shift
      importing
        ev_beg_sec = lv_shift_beg_sec
        ev_end_sec = lv_shift_end_sec ).

    data(lv_shift_cross_day) = abap_false.
    data lv_next_date type d.
    lv_next_date = is_shift-work_date + 1.
    if lv_shift_end_sec >= to_seconds( iv_date = lv_next_date iv_time = '000000' ).
      lv_shift_cross_day = abap_true.
    endif.

    data(lv_beg_date) = is_interval-beg_date.
    if lv_beg_date is initial.
      lv_beg_date = is_shift-work_date.
      if lv_shift_cross_day = abap_true and is_interval-beg_time < is_shift-beg_time.
        lv_beg_date = lv_beg_date + 1.
      endif.
    endif.

    data(lv_end_date) = is_interval-end_date.
    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
      if is_interval-end_time <= is_interval-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_interval-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_interval-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_leave_bounds.

    data(lv_beg_date) = is_leave-beg_date.
    if lv_beg_date is initial.
      lv_beg_date = is_shift-work_date.
    endif.

    data(lv_end_date) = is_leave-end_date.
    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
    endif.

    if lv_beg_date <> lv_end_date
      or ( is_leave-beg_time is initial and is_leave-end_time is initial ).

      ev_beg_sec = to_seconds(
        iv_date = lv_beg_date
        iv_time = '000000' ).

      data lv_day_after_end type d.
      lv_day_after_end = lv_end_date + 1.
      ev_end_sec = to_seconds(
        iv_date = lv_day_after_end
        iv_time = '000000' ).

      return.
    endif.

    get_interval_bounds(
      exporting
        is_shift    = is_shift
        is_interval = is_leave
      importing
        ev_beg_sec  = ev_beg_sec
        ev_end_sec  = ev_end_sec ).

  endmethod.

  method get_shift_work_seconds.

    data lt_segment type tt_segment.

    if it_leave_interval is supplied.
      lt_segment = get_shift_work_segments(
        is_shift          = is_shift
        it_leave_interval = it_leave_interval ).
    else.
      lt_segment = get_shift_work_segments( is_shift = is_shift ).
    endif.

    loop at lt_segment into data(ls_segment).
      add ls_segment-end_sec to rv_result.
      subtract ls_segment-beg_sec from rv_result.
    endloop.

  endmethod.

  method get_shift_work_segments.

    data: lv_shift_beg_sec type ty_seconds,
          lv_shift_end_sec type ty_seconds.

    get_shift_bounds(
      exporting
        is_shift   = is_shift
      importing
        ev_beg_sec = lv_shift_beg_sec
        ev_end_sec = lv_shift_end_sec ).

    data lt_work_segment type tt_segment.

    append value #(
      beg_sec = lv_shift_beg_sec
      end_sec = lv_shift_end_sec
    ) to lt_work_segment.

    data lt_exclude type tt_segment.

    loop at is_shift-breaks into data(ls_break).
      get_interval_bounds(
        exporting
          is_shift    = is_shift
          is_interval = ls_break
        importing
          ev_beg_sec  = data(lv_break_beg_sec)
          ev_end_sec  = data(lv_break_end_sec) ).

      if lv_break_end_sec > lv_shift_beg_sec and lv_break_beg_sec < lv_shift_end_sec.
        append value #(
          beg_sec = max_seconds( iv_a = lv_break_beg_sec iv_b = lv_shift_beg_sec )
          end_sec = min_seconds( iv_a = lv_break_end_sec iv_b = lv_shift_end_sec )
        ) to lt_exclude.
      endif.
    endloop.

    if it_leave_interval is supplied.
      loop at it_leave_interval into data(ls_leave_interval)
        where pernr = is_shift-pernr
          and end_sec > lv_shift_beg_sec
          and beg_sec < lv_shift_end_sec.

        append value #(
          beg_sec = max_seconds( iv_a = ls_leave_interval-beg_sec iv_b = lv_shift_beg_sec )
          end_sec = min_seconds( iv_a = ls_leave_interval-end_sec iv_b = lv_shift_end_sec )
        ) to lt_exclude.
      endloop.
    else.
      loop at is_shift-leaves into data(ls_leave).
        get_leave_bounds(
          exporting
            is_shift   = is_shift
            is_leave   = ls_leave
          importing
            ev_beg_sec = data(lv_leave_beg_sec)
            ev_end_sec = data(lv_leave_end_sec) ).

        if lv_leave_end_sec > lv_shift_beg_sec and lv_leave_beg_sec < lv_shift_end_sec.
          append value #(
            beg_sec = max_seconds( iv_a = lv_leave_beg_sec iv_b = lv_shift_beg_sec )
            end_sec = min_seconds( iv_a = lv_leave_end_sec iv_b = lv_shift_end_sec )
          ) to lt_exclude.
        endif.
      endloop.
    endif.

    sort lt_exclude by beg_sec end_sec.

    data lt_new_segment type tt_segment.
    data ls_segment type ty_segment.
    data lv_segment_count type i.

    loop at lt_exclude into data(ls_exclude).
      clear lt_new_segment.

      lv_segment_count = lines( lt_work_segment ).

      do lv_segment_count times.
        read table lt_work_segment into ls_segment index sy-index.
        if sy-subrc <> 0.
          continue.
        endif.

        if ls_exclude-end_sec <= ls_segment-beg_sec or ls_exclude-beg_sec >= ls_segment-end_sec.
          append ls_segment to lt_new_segment.
          continue.
        endif.

        if ls_exclude-beg_sec > ls_segment-beg_sec.
          append value #(
            beg_sec = ls_segment-beg_sec
            end_sec = min_seconds( iv_a = ls_exclude-beg_sec iv_b = ls_segment-end_sec )
          ) to lt_new_segment.
        endif.

        if ls_exclude-end_sec < ls_segment-end_sec.
          append value #(
            beg_sec = max_seconds( iv_a = ls_exclude-end_sec iv_b = ls_segment-beg_sec )
            end_sec = ls_segment-end_sec
          ) to lt_new_segment.
        endif.
      enddo.

      lt_work_segment = lt_new_segment.
    endloop.

    loop at lt_work_segment into data(ls_valid_segment).
      if ls_valid_segment-end_sec > ls_valid_segment-beg_sec.
        append ls_valid_segment to rt_result.
      endif.
    endloop.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method build_leave_intervals.

    loop at it_shift into data(ls_shift).
      loop at ls_shift-leaves into data(ls_leave).
        get_leave_bounds(
          exporting
            is_shift   = ls_shift
            is_leave   = ls_leave
          importing
            ev_beg_sec = data(lv_beg_sec)
            ev_end_sec = data(lv_end_sec) ).

        if lv_end_sec > lv_beg_sec.
          append value #(
            pernr     = ls_shift-pernr
            work_date = ls_shift-work_date
            beg_sec   = lv_beg_sec
            end_sec   = lv_end_sec
          ) to rt_result.
        endif.
      endloop.
    endloop.

    sort rt_result by pernr beg_sec end_sec.

  endmethod.

  method build_work_intervals.

    data(lt_leave_interval) = build_leave_intervals( it_shift ).

    loop at it_shift into data(ls_shift).
      data(lt_segment) = get_shift_work_segments(
        is_shift          = ls_shift
        it_leave_interval = lt_leave_interval ).

      loop at lt_segment into data(ls_segment).
        append value #(
          pernr     = ls_shift-pernr
          work_date = ls_shift-work_date
          beg_sec   = ls_segment-beg_sec
          end_sec   = ls_segment-end_sec
        ) to rt_result.
      endloop.
    endloop.

    sort rt_result by pernr beg_sec end_sec.

  endmethod.

  method get_longest_rest_for_bounds.

    loop at it_pernr into data(ls_pernr).
      data(lv_cursor_sec) = iv_beg_sec.
      data(lv_max_beg_sec) = iv_beg_sec.
      data(lv_max_end_sec) = iv_beg_sec.
      data lv_max_rest_sec type ty_seconds.
      data lv_rest_sec type ty_seconds.
      data lv_tail_rest_sec type ty_seconds.

      clear lv_max_rest_sec.

      loop at it_work_interval into data(ls_work_interval)
        where pernr = ls_pernr-pernr
          and end_sec > iv_beg_sec
          and beg_sec < iv_end_sec.

        data(lv_work_beg_sec) = max_seconds( iv_a = ls_work_interval-beg_sec iv_b = iv_beg_sec ).
        data(lv_work_end_sec) = min_seconds( iv_a = ls_work_interval-end_sec iv_b = iv_end_sec ).

        if lv_work_beg_sec > lv_cursor_sec.
          lv_rest_sec = lv_work_beg_sec - lv_cursor_sec.
          if lv_rest_sec > lv_max_rest_sec.
            lv_max_rest_sec = lv_rest_sec.
            lv_max_beg_sec = lv_cursor_sec.
            lv_max_end_sec = lv_work_beg_sec.
          endif.
        endif.

        if lv_work_end_sec > lv_cursor_sec.
          lv_cursor_sec = lv_work_end_sec.
        endif.
      endloop.

      if iv_end_sec > lv_cursor_sec.
        lv_tail_rest_sec = iv_end_sec - lv_cursor_sec.
        if lv_tail_rest_sec > lv_max_rest_sec.
          lv_max_rest_sec = lv_tail_rest_sec.
          lv_max_beg_sec = lv_cursor_sec.
          lv_max_end_sec = iv_end_sec.
        endif.
      endif.

      seconds_to_datetime(
        exporting
          iv_seconds = lv_max_beg_sec
        importing
          ev_date    = data(lv_beg_date)
          ev_time    = data(lv_beg_time) ).

      seconds_to_datetime(
        exporting
          iv_seconds = lv_max_end_sec
        importing
          ev_date    = data(lv_end_date)
          ev_time    = data(lv_end_time) ).

      append value #(
        pernr        = ls_pernr-pernr
        beg_date     = lv_beg_date
        beg_time     = lv_beg_time
        end_date     = lv_end_date
        end_time     = lv_end_time
        rest_seconds = lv_max_rest_sec
        rest_hours   = seconds_to_hours( lv_max_rest_sec )
      ) to rt_result.
    endloop.

  endmethod.

  method to_seconds.

    data(lv_time) = conv string( iv_time ).
    data(lv_hour) = conv i( lv_time+0(2) ).
    data(lv_minute) = conv i( lv_time+2(2) ).
    data(lv_second) = conv i( lv_time+4(2) ).

    rv_result = ( iv_date - gc_base_date ) * gc_day_seconds
              + lv_hour * 3600
              + lv_minute * 60
              + lv_second.

  endmethod.

  method seconds_to_datetime.

    data(lv_days) = iv_seconds div gc_day_seconds.
    data(lv_remain) = iv_seconds mod gc_day_seconds.

    if lv_remain < 0.
      lv_remain = lv_remain + gc_day_seconds.
      lv_days = lv_days - 1.
    endif.

    ev_date = gc_base_date + lv_days.

    data(lv_hour) = lv_remain div 3600.
    lv_remain = lv_remain mod 3600.
    data(lv_minute) = lv_remain div 60.
    data(lv_second) = lv_remain mod 60.

    ev_time = |{ lv_hour width = 2 pad = '0' align = right }{ lv_minute width = 2 pad = '0' align = right }{ lv_second width = 2 pad = '0' align = right }|.

  endmethod.

  method hours_to_seconds.
    rv_result = iv_hours * 3600.
  endmethod.

  method seconds_to_hours.
    rv_result = iv_seconds / 3600.
  endmethod.

  method max_seconds.
    rv_result = iv_a.
    if iv_b > rv_result.
      rv_result = iv_b.
    endif.
  endmethod.

  method min_seconds.
    rv_result = iv_a.
    if iv_b < rv_result.
      rv_result = iv_b.
    endif.
  endmethod.

endclass.

### 数据类型说明

#### 基础类型

| 类型 | ABAP 类型 | 说明 |
| --- | --- | --- |
| `ty_pernr` | `c length 20` | 员工号。可按实际系统改为 `pernr_d` 或自定义员工号类型。 |
| `ty_seconds` | `int8` | 秒数。内部所有时间跨度统一用秒计算，避免跨天时直接用 `T` 类型相减。 |
| `ty_hours` | `p length 15 decimals 2` | 小时数，保留 2 位小数。方法入参中的小时阈值均使用该类型。 |

#### `ty_interval` 时间区间

用于表示休息区间或请假区间。班内休息按普通时间段解释；请假区间有额外的整天规则。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `beg_date` | `d` | 建议填写 | 区间开始日期。跨天班次中的休息/请假建议明确传入，避免凌晨时间被当作班次归属日。 |
| `beg_time` | `t` | 休息必填 | 区间开始时间，格式为 `HHMMSS`。请假未填时为初始值 `000000`。 |
| `end_date` | `d` | 建议填写 | 区间结束日期。为空时默认等于 `beg_date`，若 `end_time <= beg_time` 则自动顺延一天。 |
| `end_time` | `t` | 休息必填 | 区间结束时间，格式为 `HHMMSS`。请假未填时为初始值 `000000`。 |

#### `ty_shift` 班次数据

每条记录表示某员工某个班次。员工一天只有一个班次时，一天传一条即可；如果实际数据中同一天误传多条，工具会按 `work_date` 累加工时。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `pernr` | `ty_pernr` | 是 | 员工号。 |
| `work_date` | `d` | 是 | 班次归属日期。每日工时和期间累计工时均按该日期归集。 |
| `beg_time` | `t` | 是 | 班次开始时间。开始日期固定按 `work_date` 处理。 |
| `end_date` | `d` | 否 | 班次结束日期。为空时，如果 `end_time <= beg_time`，自动认为班次跨到次日。 |
| `end_time` | `t` | 是 | 班次结束时间。 |
| `breaks` | `tt_interval` | 否 | 班内休息区间，可多段。会从班次中扣除，并裁剪到班次范围内。 |
| `leaves` | `tt_interval` | 否 | 兼容内嵌请假用法。建议留空，请假单独送入 `ty_leave`。 |

#### `ty_leave` 请假数据

用于实例接口中单独传入请假数据。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `pernr` | `ty_pernr` | 是 | 员工号。 |
| `beg_date` | `d` | 是 | 请假开始日期。 |
| `beg_time` | `t` | 否 | 请假开始时间。开始和结束时间都未填时，按整天请假处理。 |
| `end_date` | `d` | 是 | 请假结束日期。 |
| `end_time` | `t` | 否 | 请假结束时间。开始和结束日期不同，或开始/结束时间都未填时，按整天请假处理。 |

请假区间特殊规则：

- 如果 `beg_date` 和 `end_date` 不同，则该请假视为从 `beg_date 00:00:00` 到 `end_date + 1 00:00:00` 的整天请假，开始和结束时间会被忽略。
- 如果 `beg_time` 和 `end_time` 都为初始值，则该请假视为从开始日期到结束日期的整天请假；开始日期和结束日期相同时，即为当天一整天。
- 如果只有一端时间为 `000000`，仍按普通时间段处理。例如 `000000-090000` 表示当天 0 点到 9 点。
- ABAP `T` 类型中未填时间和显式 `000000` 都是初始值，因此整天判断使用“开始时间和结束时间都为初始值”作为未填时间的判断条件。
- 实例接口会先汇总对象中同一员工的全部 `ty_leave` 请假区间，再应用到该员工所有重叠班次；同一请假区间重复送入也不会重复扣减。
- 程序仍会从 `ty_shift-leaves` 汇总请假，作为内嵌请假用法兼容。

#### 返回结构

| 类型 | 用途 | 关键字段 |
| --- | --- | --- |
| `ty_day_work` | 员工每天实际工作时长 | `pernr`、`work_date`、`work_seconds`、`work_hours` |
| `ty_period_work` | 员工某段日期内累计工作时长 | `pernr`、`date_from`、`date_to`、`work_seconds`、`work_hours` |
| `ty_shift_rest` | 连续两个班次之间的休息时长 | 前一班结束时间、后一班开始时间、`rest_seconds`、`rest_hours` |
| `ty_longest_rest` | 某段时间内最长连续休息时长 | 休息开始、休息结束、`rest_seconds`、`rest_hours` |
| `ty_violation` | 合规校验结果 | `rule_code`、日期/期间、实际值、限制值、提示文本 |

#### `ty_violation-rule_code`

| 值 | 规则 | 说明 |
| --- | --- | --- |
| `DAILY_WORK` | 每日工作时长上限 | 员工某 `work_date` 的实际工作时长超过 `iv_daily_hours`。 |
| `AVG_WEEKLY_WORK` | 平均每周工作时长上限 | 所选期间累计工时按 `累计工时 / 天数 * 7` 折算后超过 `iv_weekly_hours`。 |
| `SHIFT_REST` | 连续班次之间最少休息 | 前一班次结束到后一班次开始的间隔少于 `iv_min_shift_rest_hours`。 |
| `REST_DAY` | 连续日历日内最少连续休息 | 任意 `iv_rest_window_days` 个连续日历日内，最长连续休息不足 `iv_rest_day_hours`。 |

实例计算口径：

- `ty_shift-leaves` 仍会被兼容处理，但推荐留空，统一通过 `set_leave_data` / `add_leave` 维护请假。
- 计算时会把对象中的 `mt_leave` 临时合并到同员工的班次数据中，仅供计算使用，不会改动 `get_shift_data` 返回的原始班次表。
- 如果某员工有请假但没有任何班次，该请假不会产生工时扣减，也不会单独生成结果。

#### `get_daily_work_time`

获取员工每天实际工作时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `rt_result` | Returning | `tt_day_work` | - | 每员工、每天一行结果，按 `pernr work_date` 排序。 |

计算口径：

- 统计日期使用 `ty_shift-work_date`，跨天班次不会拆分到次日。
- `breaks` 和 `leaves` 重叠时不会重复扣减。
- `breaks` / `leaves` 超出班次范围的部分会被忽略。
- 跨天或整天请假会按员工应用到所有重叠班次，不要求只放在被扣减班次当天。

#### `get_period_work_time`

获取员工某段日期内累计实际工作时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 否 | 查询开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 否 | 查询结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 否 | 员工号过滤。为空时返回所有员工。 |
| `rt_result` | Returning | `tt_period_work` | - | 每员工一行累计结果。 |

计算口径：

- 日期过滤依据 `work_date`，即班次归属日期。
- 实际工时复用 `get_daily_work_time` 的结果。
- 如果指定期间内没有工时，返回空表。

#### `get_shift_rest_time`

获取员工连续两个班次之间的休息时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `rt_result` | Returning | `tt_shift_rest` | - | 每员工相邻两个班次一行结果。 |

计算口径：

- 先按 `pernr`、班次开始绝对时间排序，再计算相邻班次间隔。
- 休息时长 = 后一班次开始时间 - 前一班次结束时间。
- 该方法使用原始班次开始/结束时间，不扣除班内休息和请假。
- 如果数据存在重叠班次，`rest_seconds` 可能为负数，校验时会被判为不满足最少休息要求。

#### `get_longest_rest_time`

获取员工某段日期内最长连续休息时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 否 | 查询开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 否 | 查询结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 否 | 员工号过滤。为空时返回所有员工。 |
| `rt_result` | Returning | `tt_longest_rest` | - | 每员工一行最长连续休息结果。 |

计算口径：

- 查询窗口为 `iv_date_from 00:00:00` 到 `iv_date_to + 1 00:00:00`。
- 先把班次扣除休息/请假后拆成实际工作区间，再在这些工作区间之间寻找最长非工作区间。
- 请假时间视为非工作时间，因此可能计入连续休息。

#### `validate_daily_work`

校验规则 1：员工每日工作时长不超过指定上限。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 校验开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 空 | 校验结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 空 | 员工号过滤。为空时校验所有员工。 |
| `iv_daily_hours` | Importing | `ty_hours` | `8.00` | 每日工作时长上限，单位小时。 |
| `rt_result` | Returning | `tt_violation` | - | 违规明细，`rule_code = DAILY_WORK`。 |

校验口径：

- 逐员工、逐 `work_date` 判断实际工作时长是否大于 `iv_daily_hours`。
- 实际工作时长会扣除班内休息和请假。

#### `validate_avg_weekly_work`

校验规则 2：员工平均每周工作时长不超过指定上限。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 校验开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 空 | 校验结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 空 | 员工号过滤。为空时校验所有员工。 |
| `iv_weekly_hours` | Importing | `ty_hours` | `40.00` | 平均每周工作时长上限，单位小时。 |
| `rt_result` | Returning | `tt_violation` | - | 违规明细，`rule_code = AVG_WEEKLY_WORK`。 |

校验口径：

- 在所选期间内按 `累计工时 / 期间天数 * 7` 折算平均每周工时。
- 期间累计工时按 `work_date` 归集，并扣除班内休息和请假。

#### `validate_shift_rest`

校验规则 3：连续两个班次之间必须满足最少连续休息时长。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 校验开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 空 | 校验结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 空 | 员工号过滤。为空时校验所有员工。 |
| `iv_min_shift_rest_hours` | Importing | `ty_hours` | `12.00` | 连续两个班次之间最少连续休息时长，单位小时。 |
| `rt_result` | Returning | `tt_violation` | - | 违规明细，`rule_code = SHIFT_REST`。 |

校验口径：

- 按原始班次结束到下一班次开始的间隔判断。
- 该规则不受班内休息和请假影响。
- 如果相邻班次重叠，休息秒数为负数，会被判为违规。

#### `validate_rest_day`

校验规则 4：每 N 个连续日历日内，必须存在至少 M 小时的连续休息。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 校验开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 空 | 校验结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 空 | 员工号过滤。为空时校验所有员工。 |
| `iv_rest_window_days` | Importing | `i` | `7` | 连续日历日窗口大小。例如传 `7` 表示每 7 个连续日历日。 |
| `iv_rest_day_hours` | Importing | `ty_hours` | `24.00` | 窗口内必须存在的最少连续休息时长，单位小时。 |
| `rt_result` | Returning | `tt_violation` | - | 违规明细，`rule_code = REST_DAY`。 |

校验口径：

- 对所选期间内每个连续日历日窗口逐个滑动检查，例如 `20260601-20260607`、`20260602-20260608`。
- 使用扣除休息/请假后的实际工作区间判断，请假时间视为非工作时间。
- `iv_rest_window_days <= 0` 时不做校验，直接返回空表。

#### `validate`

聚合校验入口，依次执行四个单项校验方法并合并返回结果。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 校验开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 空 | 校验结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `iv_pernr` | Importing | `ty_pernr` | 空 | 员工号过滤。为空时校验所有员工。 |
| `iv_daily_hours` | Importing | `ty_hours` | `8.00` | 传给 `validate_daily_work`。 |
| `iv_weekly_hours` | Importing | `ty_hours` | `40.00` | 传给 `validate_avg_weekly_work`。 |
| `iv_min_shift_rest_hours` | Importing | `ty_hours` | `12.00` | 传给 `validate_shift_rest`。 |
| `iv_rest_window_days` | Importing | `i` | `7` | 传给 `validate_rest_day`。 |
| `iv_rest_day_hours` | Importing | `ty_hours` | `24.00` | 传给 `validate_rest_day`。 |
| `rt_result` | Returning | `tt_violation` | - | 四个单项校验的违规明细合集。空表表示未发现违规。 |

### 说明

- `work_date` 是班次归属日期；日工作时长、区间累计工时按 `work_date` 统计。
- `end_date` 可不传；当 `end_time <= beg_time` 时自动按跨天班次处理。
- 实例方式下，班次和请假分开维护；计算时对象会临时合并数据，不会改动原始班次表。
- 班内休息、请假会从班次中扣除，并自动裁剪到班次范围内，重叠区间不会重复扣减。
- 跨天班次中，建议给 `breaks` 明确传入 `beg_date`、`end_date`，避免凌晨时间被误判到班次归属日。
- 请假开始和结束日期不同，或开始/结束时间都未填时，按整天请假处理。
- 跨日期请假会按员工应用到所有重叠班次，示例中的跨日期请假在存在对应日期班次时会自动扣减。
- `validate_avg_weekly_work` 中“平均每周工作时间”按所传期间折算：`期间累计工时 / 期间天数 * 7`。
- “每 N 个连续日历日应保证连续休息 M 小时”使用实际工作区间判断，请假时间视为非工作时间。
