```abap
class zcl_emp_work_time_tool implementation.

  method run_case_import.

    types tt_case_id type sorted table of string with unique key table_line.

    data lt_case_id type tt_case_id.

    loop at it_case_import into data(ls_import_key).
      if ls_import_key-case_id is initial.
        continue.
      endif.
      insert ls_import_key-case_id into table lt_case_id.
    endloop.

    loop at lt_case_id into data(lv_case_id).
      data lt_shift type tt_shift.
      data lt_leave type tt_leave.
      data lt_overtime type tt_overtime.
      data lt_no_shift_overtime_rest type tt_no_shift_overtime_rest.
      data lv_case_title type string.
      data lv_scenario_group type string.
      data lv_expected_result type string.
      data lv_expected_reason_codes type string.
      data lv_expected_reason type string.
      data lv_check_scope type string.
      data lv_leave_shift_rest_calc type abap_bool.
      data lv_full_day_leave_ot_calc type abap_bool.
      data lv_date_from type d.
      data lv_date_to type d.
      clear: lt_shift,
             lt_leave,
             lt_overtime,
             lt_no_shift_overtime_rest,
             lv_case_title,
             lv_scenario_group,
             lv_expected_result,
             lv_expected_reason_codes,
             lv_expected_reason,
             lv_check_scope,
             lv_leave_shift_rest_calc,
             lv_full_day_leave_ot_calc,
             lv_date_from,
             lv_date_to.

      loop at it_case_import into data(ls_case_row)
        where case_id = lv_case_id.

        if lv_case_title is initial and ls_case_row-case_title is not initial.
          lv_case_title = ls_case_row-case_title.
        endif.
        if lv_scenario_group is initial and ls_case_row-scenario_group is not initial.
          lv_scenario_group = ls_case_row-scenario_group.
        endif.
        if ls_case_row-leave_shift_rest_calc is not initial.
          lv_leave_shift_rest_calc = abap_true.
        endif.
        if ls_case_row-full_day_leave_ot_calc is not initial.
          lv_full_day_leave_ot_calc = abap_true.
        endif.

        if ls_case_row-work_date is not initial.
          if lv_date_from is initial or ls_case_row-work_date < lv_date_from.
            lv_date_from = ls_case_row-work_date.
          endif.
          if lv_date_to is initial or ls_case_row-work_date > lv_date_to.
            lv_date_to = ls_case_row-work_date.
          endif.
        endif.

        if ls_case_row-beg_date is not initial.
          if lv_date_from is initial or ls_case_row-beg_date < lv_date_from.
            lv_date_from = ls_case_row-beg_date.
          endif.
          if lv_date_to is initial or ls_case_row-beg_date > lv_date_to.
            lv_date_to = ls_case_row-beg_date.
          endif.
        endif.

        if ls_case_row-end_date is not initial
          and ( lv_date_to is initial or ls_case_row-end_date > lv_date_to ).
          lv_date_to = ls_case_row-end_date.
        endif.

        data(lv_row_type) = ls_case_row-row_type.
        data(lv_action_type) = ls_case_row-action_type.
        translate lv_row_type to upper case.
        translate lv_action_type to upper case.

        if lv_row_type = 'EXPECT'.
          lv_expected_result = ls_case_row-expected_result.
          lv_expected_reason_codes = ls_case_row-expected_reason_codes.
          lv_expected_reason = ls_case_row-expected_reason.
          lv_check_scope = ls_case_row-check_scope.
          continue.
        endif.

        if lv_action_type = 'CANCEL'.
          if lv_row_type = 'LEAVE'
            and ls_case_row-beg_date is not initial.

            data(lo_leave_cancel_helper) = new zcl_emp_work_time_tool( ).
            data(ls_leave_cancel_req) = value ty_leave(
              beg_date = ls_case_row-beg_date
              beg_time = ls_case_row-beg_time
              end_date = ls_case_row-end_date
              end_time = ls_case_row-end_time ).

            lo_leave_cancel_helper->get_leave_data_bounds(
              exporting
                is_leave   = ls_leave_cancel_req
              importing
                ev_beg_sec = data(lv_cancel_leave_beg_sec)
                ev_end_sec = data(lv_cancel_leave_end_sec) ).

            if lv_cancel_leave_end_sec > lv_cancel_leave_beg_sec.
              data lt_leave_after_cancel type tt_leave.
              clear lt_leave_after_cancel.

              loop at lt_leave into data(ls_leave_before_cancel).
                lo_leave_cancel_helper->get_leave_data_bounds(
                  exporting
                    is_leave   = ls_leave_before_cancel
                  importing
                    ev_beg_sec = data(lv_leave_before_beg_sec)
                    ev_end_sec = data(lv_leave_before_end_sec) ).

                if lv_leave_before_end_sec <= lv_cancel_leave_beg_sec
                  or lv_leave_before_beg_sec >= lv_cancel_leave_end_sec.
                  append ls_leave_before_cancel to lt_leave_after_cancel.
                  continue.
                endif.

                if lv_leave_before_beg_sec < lv_cancel_leave_beg_sec.
                  data ls_leave_left type ty_leave.
                  clear ls_leave_left.

                  lo_leave_cancel_helper->seconds_to_datetime(
                    exporting
                      iv_seconds = lv_leave_before_beg_sec
                    importing
                      ev_date    = ls_leave_left-beg_date
                      ev_time    = ls_leave_left-beg_time ).

                  lo_leave_cancel_helper->seconds_to_datetime(
                    exporting
                      iv_seconds = lv_cancel_leave_beg_sec
                    importing
                      ev_date    = ls_leave_left-end_date
                      ev_time    = ls_leave_left-end_time ).

                  append ls_leave_left to lt_leave_after_cancel.
                endif.

                if lv_cancel_leave_end_sec < lv_leave_before_end_sec.
                  data ls_leave_right type ty_leave.
                  clear ls_leave_right.

                  lo_leave_cancel_helper->seconds_to_datetime(
                    exporting
                      iv_seconds = lv_cancel_leave_end_sec
                    importing
                      ev_date    = ls_leave_right-beg_date
                      ev_time    = ls_leave_right-beg_time ).

                  lo_leave_cancel_helper->seconds_to_datetime(
                    exporting
                      iv_seconds = lv_leave_before_end_sec
                    importing
                      ev_date    = ls_leave_right-end_date
                      ev_time    = ls_leave_right-end_time ).

                  append ls_leave_right to lt_leave_after_cancel.
                endif.
              endloop.

              lt_leave = lt_leave_after_cancel.
            endif.
          endif.

          continue.
        endif.

        case lv_row_type.
          when 'SHIFT'.
            data(ls_shift_case) = value ty_shift(
              work_date = ls_case_row-work_date
              rest_flag = ls_case_row-rest_flag
              beg_time  = ls_case_row-beg_time
              end_date  = ls_case_row-end_date
              end_time  = ls_case_row-end_time ).

            if ls_case_row-break1_beg_time is not initial
              or ls_case_row-break1_end_time is not initial.
              append value #(
                beg_date = ls_case_row-work_date
                beg_time = ls_case_row-break1_beg_time
                end_date = ls_case_row-work_date
                end_time = ls_case_row-break1_end_time
              ) to ls_shift_case-breaks.
            endif.

            if ls_case_row-break2_beg_time is not initial
              or ls_case_row-break2_end_time is not initial.
              append value #(
                beg_date = ls_case_row-work_date
                beg_time = ls_case_row-break2_beg_time
                end_date = ls_case_row-work_date
                end_time = ls_case_row-break2_end_time
              ) to ls_shift_case-breaks.
            endif.

            append ls_shift_case to lt_shift.

          when 'LEAVE'.
            append value #(
              beg_date = ls_case_row-beg_date
              beg_time = ls_case_row-beg_time
              end_date = ls_case_row-end_date
              end_time = ls_case_row-end_time
            ) to lt_leave.

          when 'OVERTIME'.
            append value #(
              beg_date = ls_case_row-beg_date
              beg_time = ls_case_row-beg_time
              end_date = ls_case_row-end_date
              end_time = ls_case_row-end_time
            ) to lt_overtime.

          when 'NO_SHIFT_REST'.
            append value #(
              beg_date = ls_case_row-beg_date
              beg_time = ls_case_row-beg_time
              end_date = ls_case_row-end_date
              end_time = ls_case_row-end_time
            ) to lt_no_shift_overtime_rest.
        endcase.
      endloop.

      if lv_expected_result is initial.
        lv_expected_result = '合规'.
      endif.

      if lv_check_scope is initial.
        lv_check_scope = 'SHIFT_REST;OVERTIME'.
      endif.
      translate lv_check_scope to upper case.
      condense lv_check_scope no-gaps.
      data(lv_scope_match) = |;{ lv_check_scope };|.

      data(lo_case_tool) = new zcl_emp_work_time_tool(
        iv_leave_extend_calc      = abap_true
        iv_leave_shift_rest_calc  = lv_leave_shift_rest_calc
        iv_full_day_leave_ot_calc = lv_full_day_leave_ot_calc ).
      lo_case_tool->set_shift_data( lt_shift ).
      lo_case_tool->set_leave_data( lt_leave ).
      lo_case_tool->set_overtime_data( lt_overtime ).
      lo_case_tool->set_no_shift_ot_rest_data( lt_no_shift_overtime_rest ).

      data lv_actual_reason_codes type string.
      clear lv_actual_reason_codes.

      if lv_scope_match cs ';SHIFT_REST;'.
        data(lt_shift_rest_case) = lo_case_tool->get_shift_rest_time_v2( ).
        loop at lt_shift_rest_case into data(ls_shift_rest_case)
          where rest_seconds < gc_ot_rest_limit_sec.
          add_case_reason(
            exporting
              iv_reason = 'SHIFT_REST'
            changing
              cv_reason = lv_actual_reason_codes ).
        endloop.
      endif.

      if lv_scope_match cs ';OVERTIME;'.
        data(lt_overtime_compliance_case) = lo_case_tool->get_overtime_compliance(
          iv_actual_time = abap_true ).
        loop at lt_overtime_compliance_case into data(ls_overtime_compliance_case)
          where is_valid = abap_false.

          if ls_overtime_compliance_case-is_total_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_TOTAL'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.

          if ls_overtime_compliance_case-is_day_total_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_DAY_TOTAL'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.

          if ls_overtime_compliance_case-is_continuous_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_CONTINUOUS'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.

          if ls_overtime_compliance_case-is_prev_work_rest_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_PREV_REST'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.

          if ls_overtime_compliance_case-is_next_shift_rest_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_NEXT_REST'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.

          if ls_overtime_compliance_case-is_leave_overlap_valid = abap_false.
            add_case_reason(
              exporting
                iv_reason = 'OT_LEAVE_OVERLAP'
              changing
                cv_reason = lv_actual_reason_codes ).
          endif.
        endloop.
      endif.

      if lv_scope_match cs ';MONTH_OVERTIME;'
        and lv_date_from is not initial
        and lv_date_to is not initial.
        data lv_month_check_date type d.
        data lv_month_from_case type d.
        data lv_month_to_case type d.
        data lt_month_overtime_case type tt_period_overtime.
        data ls_month_overtime_case type ty_period_overtime.

        lv_month_check_date = lv_date_from.
        lv_month_check_date+6(2) = '01'.

        while lv_month_check_date <= lv_date_to.
          lo_case_tool->get_month_dates(
            exporting
              iv_base_date = lv_month_check_date
            importing
              ev_date_from = lv_month_from_case
              ev_date_to   = lv_month_to_case ).

          lt_month_overtime_case = lo_case_tool->get_base_overtime_time(
            iv_base_date    = lv_month_check_date
            iv_period_type  = 'M'
            iv_calendar_day = abap_true
            iv_actual_time  = abap_true ).

          read table lt_month_overtime_case into ls_month_overtime_case index 1.
          if sy-subrc = 0
            and ls_month_overtime_case-overtime_seconds > 36 * 3600.
            add_case_reason(
              exporting
                iv_reason = 'MONTH_OT_36'
              changing
                cv_reason = lv_actual_reason_codes ).
            exit.
          endif.

          lv_month_check_date = lv_month_to_case + 1.
        endwhile.
      endif.

      if lv_scope_match cs ';LONGEST_REST;'
        and lv_date_from is not initial
        and lv_date_to is not initial.
        data lv_long_rest_violation type abap_bool.
        data lv_long_rest_window_from type d.
        data lv_long_rest_last_from type d.
        data lv_long_rest_window_to type d.
        data lt_longest_rest_case type tt_longest_rest.
        data ls_longest_rest_case type ty_longest_rest.
        data lv_max_rest_seconds type ty_seconds.

        clear lv_long_rest_violation.

        if lv_date_to - lv_date_from >= 6.
          lv_long_rest_window_from = lv_date_from.
          lv_long_rest_last_from = lv_date_to - 6.

          while lv_long_rest_window_from <= lv_long_rest_last_from.
            lv_long_rest_window_to = lv_long_rest_window_from + 6.
            lt_longest_rest_case = lo_case_tool->get_longest_rest_time(
              iv_date_from = lv_long_rest_window_from
              iv_date_to   = lv_long_rest_window_to ).

            clear lv_max_rest_seconds.
            loop at lt_longest_rest_case into ls_longest_rest_case.
              if ls_longest_rest_case-rest_seconds > lv_max_rest_seconds.
                lv_max_rest_seconds = ls_longest_rest_case-rest_seconds.
              endif.
            endloop.

            if lv_max_rest_seconds < gc_long_ot_rest_sec.
              lv_long_rest_violation = abap_true.
              exit.
            endif.

            lv_long_rest_window_from = lv_long_rest_window_from + 1.
          endwhile.
        else.
          lt_longest_rest_case = lo_case_tool->get_longest_rest_time(
            iv_date_from = lv_date_from
            iv_date_to   = lv_date_to ).

          clear lv_max_rest_seconds.
          loop at lt_longest_rest_case into ls_longest_rest_case.
            if ls_longest_rest_case-rest_seconds > lv_max_rest_seconds.
              lv_max_rest_seconds = ls_longest_rest_case-rest_seconds.
            endif.
          endloop.

          if lv_max_rest_seconds < gc_long_ot_rest_sec.
            lv_long_rest_violation = abap_true.
          endif.
        endif.

        if lv_long_rest_violation = abap_true.
          add_case_reason(
            exporting
              iv_reason = 'LONG_REST_24'
            changing
              cv_reason = lv_actual_reason_codes ).
        endif.
      endif.

      data(lv_actual_result) = cond string(
        when lv_actual_reason_codes is initial then '合规'
        else '不合规' ).

      condense lv_expected_reason_codes no-gaps.
      condense lv_actual_reason_codes no-gaps.

      append value #(
        case_id               = lv_case_id
        case_title            = lv_case_title
        scenario_group        = lv_scenario_group
        expected_result       = lv_expected_result
        actual_result         = lv_actual_result
        expected_reason_codes = lv_expected_reason_codes
        actual_reason_codes   = lv_actual_reason_codes
        expected_reason       = lv_expected_reason
        actual_reason         = lv_actual_reason_codes
        result_match          = xsdbool( lv_expected_result = lv_actual_result )
        reason_match          = xsdbool( lv_expected_reason_codes = lv_actual_reason_codes )
        is_match              = xsdbool( lv_expected_result = lv_actual_result
                                      and lv_expected_reason_codes = lv_actual_reason_codes )
      ) to rt_result.
    endloop.

    sort rt_result by case_id.

  endmethod.

  method constructor.
    mv_leave_extend_calc = xsdbool( iv_leave_extend_calc is not initial ).
    mv_leave_shift_rest_calc = xsdbool( iv_leave_shift_rest_calc is not initial ).

  method get_daily_work_time.

    data(lt_leave_interval) = build_leave_intervals( ).

    loop at mt_shift into data(ls_shift).
      data(lv_seconds) = get_shift_work_seconds(
        is_shift          = ls_shift
        it_leave_interval = lt_leave_interval ).

      read table rt_result assigning field-symbol(<ls_day>)
        with key work_date = ls_shift-work_date.
      if sy-subrc = 0.
        add lv_seconds to <ls_day>-work_seconds.
        <ls_day>-work_hours = seconds_to_hours( <ls_day>-work_seconds ).
      else.
        append value #(
          work_date    = ls_shift-work_date
          work_seconds = lv_seconds
          work_hours   = seconds_to_hours( lv_seconds )
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by work_date.

  endmethod.

  method get_period_work_time.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lt_day_work) = get_daily_work_time( ).

    loop at lt_day_work into data(ls_day_work)
      where work_date >= lv_date_from
        and work_date <= lv_date_to.

      if rt_result is initial.
        append value #(
          date_from = lv_date_from
          date_to   = lv_date_to
        ) to rt_result.
      endif.

      read table rt_result assigning field-symbol(<ls_period>) index 1.
      add ls_day_work-work_seconds to <ls_period>-work_seconds.
      <ls_period>-work_hours = seconds_to_hours( <ls_period>-work_seconds ).
    endloop.

  endmethod.

  endmethod.

  method get_shift_rest_time_v2.

    data lt_shift_abs type tt_abs_interval.
    data lt_overtime_abs type tt_abs_interval.
    data lt_outside_overtime_abs type tt_abs_interval.
    data lt_leave_interval_v2 type tt_abs_interval.

    if mv_leave_shift_rest_calc = abap_true.
      lt_leave_interval_v2 = build_leave_intervals( ).
    endif.

    loop at mt_shift into data(ls_shift_v2).
      if ls_shift_v2-rest_flag is not initial.
        continue.
      endif.

      get_shift_bounds(
        exporting
          is_shift   = ls_shift_v2
        importing
          ev_beg_sec = data(lv_shift_beg_sec_v2)
          ev_end_sec = data(lv_shift_end_sec_v2) ).

      if lv_shift_end_sec_v2 <= lv_shift_beg_sec_v2.
        continue.
      endif.

      data(lv_work_beg_sec_v2) = lv_shift_beg_sec_v2.
      data(lv_work_end_sec_v2) = lv_shift_end_sec_v2.

      if mv_leave_shift_rest_calc = abap_true.
        data(lt_shift_segment_v2) = get_shift_work_segments(
          is_shift          = ls_shift_v2
          it_leave_interval = lt_leave_interval_v2 ).

        if lt_shift_segment_v2 is initial.
          continue.
        endif.

        clear: lv_work_beg_sec_v2,
               lv_work_end_sec_v2.

        loop at lt_shift_segment_v2 into data(ls_shift_segment_v2).
          if lv_work_beg_sec_v2 is initial
            or ls_shift_segment_v2-beg_sec < lv_work_beg_sec_v2.
            lv_work_beg_sec_v2 = ls_shift_segment_v2-beg_sec.
          endif.
          if ls_shift_segment_v2-end_sec > lv_work_end_sec_v2.
            lv_work_end_sec_v2 = ls_shift_segment_v2-end_sec.
          endif.
        endloop.
      endif.

      append value #(
        work_date     = ls_shift_v2-work_date
        work_type     = gc_work_type_shift
        beg_sec       = lv_work_beg_sec_v2
        end_sec       = lv_work_end_sec_v2
        shift_beg_sec = lv_shift_beg_sec_v2
        shift_end_sec = lv_shift_end_sec_v2
      ) to lt_shift_abs.
    endloop.

    sort lt_shift_abs by beg_sec end_sec.

    lt_overtime_abs = build_overtime_intervals( ).

    loop at lt_overtime_abs into data(ls_overtime_src_v2).
      data lt_candidate_segment type tt_segment.
      data lt_next_segment type tt_segment.
      clear lt_candidate_segment.

      append value #(
        beg_sec = ls_overtime_src_v2-beg_sec
        end_sec = ls_overtime_src_v2-end_sec
      ) to lt_candidate_segment.

      loop at lt_shift_abs into data(ls_shift_cover_v2)
        where end_sec > ls_overtime_src_v2-beg_sec
          and beg_sec < ls_overtime_src_v2-end_sec.

        clear lt_next_segment.

        loop at lt_candidate_segment into data(ls_candidate_segment).
          if ls_shift_cover_v2-end_sec <= ls_candidate_segment-beg_sec
            or ls_shift_cover_v2-beg_sec >= ls_candidate_segment-end_sec.
            append ls_candidate_segment to lt_next_segment.
            continue.
          endif.

          if ls_shift_cover_v2-beg_sec > ls_candidate_segment-beg_sec.
            append value #(
              beg_sec = ls_candidate_segment-beg_sec
              end_sec = min_seconds(
                iv_a = ls_shift_cover_v2-beg_sec
                iv_b = ls_candidate_segment-end_sec )
            ) to lt_next_segment.
          endif.

          if ls_shift_cover_v2-end_sec < ls_candidate_segment-end_sec.
            append value #(
              beg_sec = max_seconds(
                iv_a = ls_shift_cover_v2-end_sec
                iv_b = ls_candidate_segment-beg_sec )
              end_sec = ls_candidate_segment-end_sec
            ) to lt_next_segment.
          endif.
        endloop.

        lt_candidate_segment = lt_next_segment.
      endloop.

      loop at lt_candidate_segment into data(ls_outside_segment).
        if ls_outside_segment-end_sec <= ls_outside_segment-beg_sec.
          continue.
        endif.

        append value #(
          work_date     = ls_overtime_src_v2-work_date
          work_type     = gc_work_type_ot
          beg_sec       = ls_outside_segment-beg_sec
          end_sec       = ls_outside_segment-end_sec
          shift_beg_sec = ls_outside_segment-beg_sec
          shift_end_sec = ls_outside_segment-end_sec
        ) to lt_outside_overtime_abs.
      endloop.
    endloop.

    sort lt_outside_overtime_abs by beg_sec end_sec.

    data ls_prev_v2 type ty_abs_interval.
    data lv_has_prev_v2 type abap_bool.

    loop at lt_shift_abs into data(ls_next_v2).
      if lv_has_prev_v2 = abap_true.
        data(lv_prev_work_end_sec) = ls_prev_v2-end_sec.
        data(lv_rest_beg_sec_v2) = ls_prev_v2-end_sec.
        data(lv_rest_end_sec_v2) = ls_next_v2-beg_sec.
        data(lv_stop_overtime_loop) = abap_false.
        data ls_result_ot_v2 type ty_abs_interval.
        data lv_has_result_ot_v2 type abap_bool.
        clear: ls_result_ot_v2,
               lv_has_result_ot_v2.

        loop at lt_outside_overtime_abs into data(ls_outside_overtime)
          where end_sec > ls_prev_v2-end_sec
            and beg_sec < ls_next_v2-beg_sec.

          if lv_stop_overtime_loop = abap_true.
            exit.
          endif.

          data(lv_outside_ot_beg_sec) = max_seconds(
            iv_a = ls_outside_overtime-beg_sec
            iv_b = ls_prev_v2-end_sec ).
          data(lv_outside_ot_end_sec) = min_seconds(
            iv_a = ls_outside_overtime-end_sec
            iv_b = ls_next_v2-beg_sec ).

          if lv_outside_ot_end_sec <= lv_outside_ot_beg_sec.
            continue.
          endif.

          if lv_outside_ot_beg_sec - lv_prev_work_end_sec < gc_shift_ot_rest_sec.
            lv_prev_work_end_sec = max_seconds(
              iv_a = lv_prev_work_end_sec
              iv_b = lv_outside_ot_end_sec ).
            lv_rest_beg_sec_v2 = lv_prev_work_end_sec.
            lv_rest_end_sec_v2 = ls_next_v2-beg_sec.
            ls_result_ot_v2 = ls_outside_overtime.
            ls_result_ot_v2-beg_sec = lv_outside_ot_beg_sec.
            ls_result_ot_v2-end_sec = lv_outside_ot_end_sec.
            lv_has_result_ot_v2 = abap_true.
          else.
            lv_rest_beg_sec_v2 = lv_prev_work_end_sec.
            lv_rest_end_sec_v2 = lv_outside_ot_beg_sec.
            ls_result_ot_v2 = ls_outside_overtime.
            ls_result_ot_v2-beg_sec = lv_outside_ot_beg_sec.
            ls_result_ot_v2-end_sec = lv_outside_ot_end_sec.
            lv_has_result_ot_v2 = abap_true.
            lv_stop_overtime_loop = abap_true.
          endif.
        endloop.

        data(lv_rest_seconds_v2) = lv_rest_end_sec_v2 - lv_rest_beg_sec_v2.

        seconds_to_datetime(
          exporting
            iv_seconds = ls_prev_v2-end_sec
          importing
            ev_date    = data(lv_prev_end_date_v2)
            ev_time    = data(lv_prev_end_time_v2) ).

        seconds_to_datetime(
          exporting
            iv_seconds = ls_next_v2-beg_sec
          importing
            ev_date    = data(lv_next_beg_date_v2)
            ev_time    = data(lv_next_beg_time_v2) ).

        seconds_to_datetime(
          exporting
            iv_seconds = lv_rest_beg_sec_v2
          importing
            ev_date    = data(lv_rest_beg_date_v2)
            ev_time    = data(lv_rest_beg_time_v2) ).

        seconds_to_datetime(
          exporting
            iv_seconds = lv_rest_end_sec_v2
          importing
            ev_date    = data(lv_rest_end_date_v2)
            ev_time    = data(lv_rest_end_time_v2) ).

        data: lv_ot_beg_date_v2 type d,
              lv_ot_beg_time_v2 type t,
              lv_ot_end_date_v2 type d,
              lv_ot_end_time_v2 type t.
        clear: lv_ot_beg_date_v2,
               lv_ot_beg_time_v2,
               lv_ot_end_date_v2,
               lv_ot_end_time_v2.

        if lv_has_result_ot_v2 = abap_true.
          seconds_to_datetime(
            exporting
              iv_seconds = ls_result_ot_v2-beg_sec
            importing
              ev_date    = lv_ot_beg_date_v2
              ev_time    = lv_ot_beg_time_v2 ).

          seconds_to_datetime(
            exporting
              iv_seconds = ls_result_ot_v2-end_sec
            importing
              ev_date    = lv_ot_end_date_v2
              ev_time    = lv_ot_end_time_v2 ).
        endif.

        append value #(
          prev_work_date = ls_prev_v2-work_date
          prev_end_date  = lv_prev_end_date_v2
          prev_end_time  = lv_prev_end_time_v2
          next_work_date = ls_next_v2-work_date
          next_beg_date  = lv_next_beg_date_v2
          next_beg_time  = lv_next_beg_time_v2
          rest_beg_date  = lv_rest_beg_date_v2
          rest_beg_time  = lv_rest_beg_time_v2
          rest_end_date  = lv_rest_end_date_v2
          rest_end_time  = lv_rest_end_time_v2
          ot_beg_date    = lv_ot_beg_date_v2
          ot_beg_time    = lv_ot_beg_time_v2
          ot_end_date    = lv_ot_end_date_v2
          ot_end_time    = lv_ot_end_time_v2
          rest_seconds   = lv_rest_seconds_v2
          rest_hours     = seconds_to_hours( lv_rest_seconds_v2 )
        ) to rt_result.
      endif.

      ls_prev_v2 = ls_next_v2.
      lv_has_prev_v2 = abap_true.
    endloop.

  endmethod.

  method get_longest_rest_time.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_period_dates(
      exporting
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lt_work_interval) = build_work_intervals( ).

    data(lv_beg_sec) = to_seconds( iv_date = lv_date_from iv_time = '000000' ).
    data lv_day_after_to type d.
    lv_day_after_to = lv_date_to + 1.
    data(lv_end_sec) = to_seconds( iv_date = lv_day_after_to iv_time = '000000' ).

    rt_result = get_rest_intervals_for_bounds(
      it_work_interval = lt_work_interval
      iv_beg_sec       = lv_beg_sec
      iv_end_sec       = lv_end_sec ).

    data lv_max_rest_seconds type ty_seconds.
    loop at rt_result assigning field-symbol(<ls_rest>).
      clear <ls_rest>-is_longest.
      if <ls_rest>-rest_seconds > lv_max_rest_seconds.
        lv_max_rest_seconds = <ls_rest>-rest_seconds.
      endif.
    endloop.

    loop at rt_result assigning <ls_rest>
      where rest_seconds = lv_max_rest_seconds.
      <ls_rest>-is_longest = abap_true.
    endloop.

    sort rt_result by beg_date beg_time end_date end_time.

  endmethod.

  method get_base_longest_rest_time.

    data: lv_date_from type d,
          lv_date_to   type d.

    get_around_dates(
      exporting
        iv_base_date = iv_base_date
        iv_days      = iv_calc_days
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    field-symbols <ls_base_rest> type ty_longest_rest.

    data(lt_base_rest) = get_longest_rest_time(
      iv_date_from = lv_date_from
      iv_date_to   = iv_base_date ).

    loop at lt_base_rest assigning <ls_base_rest>.
      <ls_base_rest>-base_date = iv_base_date.
      <ls_base_rest>-base_side = gc_base_side_prev.
      <ls_base_rest>-base_date_from = lv_date_from.
      <ls_base_rest>-base_date_to = iv_base_date.
    endloop.
    append lines of lt_base_rest to rt_result.

    lt_base_rest = get_longest_rest_time(
      iv_date_from = iv_base_date
      iv_date_to   = lv_date_to ).

    loop at lt_base_rest assigning <ls_base_rest>.
      <ls_base_rest>-base_date = iv_base_date.
      <ls_base_rest>-base_side = gc_base_side_next.
      <ls_base_rest>-base_date_from = iv_base_date.
      <ls_base_rest>-base_date_to = lv_date_to.
    endloop.
    append lines of lt_base_rest to rt_result.

  endmethod.

  method get_overtime_time.

    data lt_leave_interval_ot_time type tt_abs_interval.
    data lt_shift_abs_ot_time type tt_abs_interval.
    data lt_ns_rest_itv_ot_time type tt_abs_interval.

    if mv_full_day_leave_ot_calc = abap_true.
      lt_leave_interval_ot_time = build_leave_intervals( ).
    endif.

    loop at mt_shift into data(ls_overtime_shift_src).
      if ls_overtime_shift_src-rest_flag is not initial.
        continue.
      endif.

      get_shift_bounds(
        exporting
          is_shift   = ls_overtime_shift_src
        importing
          ev_beg_sec = data(lv_overtime_shift_beg_sec)
          ev_end_sec = data(lv_overtime_shift_end_sec) ).

      if lv_overtime_shift_end_sec <= lv_overtime_shift_beg_sec.
        continue.
      endif.

      if mv_full_day_leave_ot_calc = abap_true.
        data(lt_overtime_shift_segment) = get_shift_work_segments(
          is_shift          = ls_overtime_shift_src
          it_leave_interval = lt_leave_interval_ot_time ).
        if lt_overtime_shift_segment is initial.
          continue.
        endif.
      endif.

      append value #(
        work_date     = ls_overtime_shift_src-work_date
        work_type     = gc_work_type_shift
        beg_sec       = lv_overtime_shift_beg_sec
        end_sec       = lv_overtime_shift_end_sec
        shift_beg_sec = lv_overtime_shift_beg_sec
        shift_end_sec = lv_overtime_shift_end_sec
      ) to lt_shift_abs_ot_time.
    endloop.

    sort lt_shift_abs_ot_time by beg_sec end_sec.

    lt_ns_rest_itv_ot_time =
      build_no_shift_ot_rest_itv( ).

    loop at mt_overtime into data(ls_overtime).
      get_overtime_bounds(
        exporting
          is_overtime = ls_overtime
        importing
          ev_beg_sec  = data(lv_beg_sec)
          ev_end_sec  = data(lv_end_sec) ).

      if lv_end_sec <= lv_beg_sec.
        continue.
      endif.

      data(lv_overtime_seconds) = lv_end_sec - lv_beg_sec.

      data(lv_has_shift_context_ot_time) = has_shift_overtime_context(
        it_shift_interval = lt_shift_abs_ot_time
        iv_beg_sec        = lv_beg_sec
        iv_end_sec        = lv_end_sec ).

      if iv_actual_time is not initial.
        loop at lt_shift_abs_ot_time into data(ls_overtime_shift).
          data(lv_overtime_shift_overlap_sec) = min_seconds(
            iv_a = lv_end_sec
            iv_b = ls_overtime_shift-end_sec )
            - max_seconds(
                iv_a = lv_beg_sec
                iv_b = ls_overtime_shift-beg_sec ).

          if lv_overtime_shift_overlap_sec > 0.
            subtract lv_overtime_shift_overlap_sec from lv_overtime_seconds.
          endif.
        endloop.
      endif.

      if lv_has_shift_context_ot_time is initial.
        data(lv_ns_rest_secs_ot_time) = get_interval_overlap_seconds(
          it_interval = lt_ns_rest_itv_ot_time
          iv_beg_sec  = lv_beg_sec
          iv_end_sec  = lv_end_sec ).
        subtract lv_ns_rest_secs_ot_time from lv_overtime_seconds.
      endif.

      if lv_overtime_seconds < 0.
        clear lv_overtime_seconds.
      endif.

      seconds_to_datetime(
        exporting
          iv_seconds = lv_beg_sec
        importing
          ev_date    = data(lv_beg_date)
          ev_time    = data(lv_beg_time) ).

      seconds_to_datetime(
        exporting
          iv_seconds = lv_end_sec
        importing
          ev_date    = data(lv_end_date)
          ev_time    = data(lv_end_time) ).

      append value #(
        beg_date         = lv_beg_date
        beg_time         = lv_beg_time
        end_date         = lv_end_date
        end_time         = lv_end_time
        overtime_seconds = lv_overtime_seconds
        overtime_hours   = seconds_to_hours( lv_overtime_seconds )
      ) to rt_result.
    endloop.

    sort rt_result by beg_date beg_time end_date end_time.

  endmethod.

  method get_period_overtime_time.

    data: lv_date_from type d,
          lv_date_to   type d.
    data lt_leave_interval_ot_period type tt_abs_interval.
    data lt_shift_abs_ot_period type tt_abs_interval.
    data lt_ns_rest_itv_ot_period type tt_abs_interval.

    if mv_full_day_leave_ot_calc = abap_true.
      lt_leave_interval_ot_period = build_leave_intervals( ).
    endif.

    loop at mt_shift into data(ls_period_overtime_shift_src).
      if ls_period_overtime_shift_src-rest_flag is not initial.
        continue.
      endif.

      get_shift_bounds(
        exporting
          is_shift   = ls_period_overtime_shift_src
        importing
          ev_beg_sec = data(lv_period_shift_beg_sec)
          ev_end_sec = data(lv_period_shift_end_sec) ).

      if lv_period_shift_end_sec <= lv_period_shift_beg_sec.
        continue.
      endif.

      if mv_full_day_leave_ot_calc = abap_true.
        data(lt_period_shift_segment) = get_shift_work_segments(
          is_shift          = ls_period_overtime_shift_src
          it_leave_interval = lt_leave_interval_ot_period ).
        if lt_period_shift_segment is initial.
          continue.
        endif.
      endif.

      append value #(
        work_date     = ls_period_overtime_shift_src-work_date
        work_type     = gc_work_type_shift
        beg_sec       = lv_period_shift_beg_sec
        end_sec       = lv_period_shift_end_sec
        shift_beg_sec = lv_period_shift_beg_sec
        shift_end_sec = lv_period_shift_end_sec
      ) to lt_shift_abs_ot_period.
    endloop.

    sort lt_shift_abs_ot_period by beg_sec end_sec.

    lt_ns_rest_itv_ot_period =
      build_no_shift_ot_rest_itv( ).

    get_period_dates(
      exporting
        iv_date_from = iv_date_from
        iv_date_to   = iv_date_to
      importing
        ev_date_from = lv_date_from
        ev_date_to   = lv_date_to ).

    if lv_date_from is initial or lv_date_to is initial.
      return.
    endif.

    data(lv_period_beg_sec) = to_seconds(
      iv_date = lv_date_from
      iv_time = '000000' ).

    data lv_day_after_period type d.
    lv_day_after_period = lv_date_to + 1.

    data(lv_period_end_sec) = to_seconds(
      iv_date = lv_day_after_period
      iv_time = '000000' ).

    loop at mt_overtime into data(ls_period_overtime).
      get_overtime_bounds(
        exporting
          is_overtime = ls_period_overtime
        importing
          ev_beg_sec  = data(lv_period_ot_beg_sec)
          ev_end_sec  = data(lv_period_ot_end_sec) ).

      data(lv_count_beg_sec) = max_seconds(
        iv_a = lv_period_ot_beg_sec
        iv_b = lv_period_beg_sec ).
      data(lv_count_end_sec) = min_seconds(
        iv_a = lv_period_ot_end_sec
        iv_b = lv_period_end_sec ).

      if lv_count_end_sec <= lv_count_beg_sec.
        continue.
      endif.

      data(lv_count_overtime_seconds) = lv_count_end_sec - lv_count_beg_sec.

      data(lv_has_shift_context_ot_period) = has_shift_overtime_context(
        it_shift_interval = lt_shift_abs_ot_period
        iv_beg_sec        = lv_count_beg_sec
        iv_end_sec        = lv_count_end_sec ).

      if iv_actual_time is not initial.
        loop at lt_shift_abs_ot_period into data(ls_period_overtime_shift).
          data(lv_period_shift_overlap_sec) = min_seconds(
            iv_a = lv_count_end_sec
            iv_b = ls_period_overtime_shift-end_sec )
            - max_seconds(
                iv_a = lv_count_beg_sec
                iv_b = ls_period_overtime_shift-beg_sec ).

          if lv_period_shift_overlap_sec > 0.
            subtract lv_period_shift_overlap_sec from lv_count_overtime_seconds.
          endif.
        endloop.
      endif.

      if lv_has_shift_context_ot_period is initial.
        data(lv_ns_rest_secs_ot_period) = get_interval_overlap_seconds(
          it_interval = lt_ns_rest_itv_ot_period
          iv_beg_sec  = lv_count_beg_sec
          iv_end_sec  = lv_count_end_sec ).
        subtract lv_ns_rest_secs_ot_period from lv_count_overtime_seconds.
      endif.

      if lv_count_overtime_seconds < 0.
        clear lv_count_overtime_seconds.
      endif.

      if lv_count_overtime_seconds <= 0.
        continue.
      endif.

      if rt_result is initial.
        append value #(
          date_from = lv_date_from
          date_to   = lv_date_to
        ) to rt_result.
      endif.

      read table rt_result assigning field-symbol(<ls_period>) index 1.
      add lv_count_overtime_seconds to <ls_period>-overtime_seconds.
      <ls_period>-overtime_hours = seconds_to_hours( <ls_period>-overtime_seconds ).
    endloop.

  endmethod.

    data: lv_date_from type d,
          lv_date_to   type d.

    if iv_period_type = 'M' or iv_period_type = 'm'.

        get_around_dates(
          exporting
            iv_base_date = iv_base_date
            iv_days      = 30
          importing
            ev_date_from = lv_date_from
            ev_date_to   = lv_date_to ).
      else.
        get_month_dates(
          exporting
            iv_base_date = iv_base_date
          importing
            ev_date_from = lv_date_from
            ev_date_to   = lv_date_to ).
      endif.
    else.
      if iv_calendar_day is initial.
        get_around_dates(
          exporting
            iv_base_date = iv_base_date
            iv_days      = 7
          importing
            ev_date_from = lv_date_from
            ev_date_to   = lv_date_to ).

            iv_base_date = iv_base_date
          importing

      iv_date_from   = lv_date_from

      iv_actual_time = iv_actual_time ).

  endmethod.

  method get_overtime_compliance.

    types:
      begin of ty_overtime_group,
        beg_sec type ty_seconds,
        end_sec type ty_seconds,
      end of ty_overtime_group,
      tt_overtime_group type standard table of ty_overtime_group with default key,
      begin of ty_no_shift_day,
        work_date        type d,
        overtime_seconds type ty_seconds,
      end of ty_no_shift_day,
      tt_no_shift_day type standard table of ty_no_shift_day with default key.

    data lt_shift_abs type tt_abs_interval.
    data lt_overtime_abs type tt_abs_interval.
    data lt_actual_overtime_abs type tt_abs_interval.
    data lt_no_shift_abs type tt_abs_interval.
    data lt_no_shift_group type tt_overtime_group.
    data lt_no_shift_day type tt_no_shift_day.
    data lt_leave_interval_rule type tt_abs_interval.
    data lt_no_shift_rest_interval_rule type tt_abs_interval.

    lt_leave_interval_rule = build_leave_intervals( ).

    loop at mt_shift into data(ls_shift_rule).
      if ls_shift_rule-rest_flag is not initial.
        continue.
      endif.

      get_shift_bounds(
        exporting
          is_shift   = ls_shift_rule
        importing
          ev_beg_sec = data(lv_shift_rule_beg_sec)
          ev_end_sec = data(lv_shift_rule_end_sec) ).

      if lv_shift_rule_end_sec <= lv_shift_rule_beg_sec.
        continue.
      endif.

      if mv_full_day_leave_ot_calc = abap_true.
        data(lt_shift_rule_segment) = get_shift_work_segments(
          is_shift          = ls_shift_rule
          it_leave_interval = lt_leave_interval_rule ).
        if lt_shift_rule_segment is initial.
          continue.
        endif.
      endif.

      append value #(
        work_date     = ls_shift_rule-work_date
        work_type     = gc_work_type_shift
        beg_sec       = lv_shift_rule_beg_sec
        end_sec       = lv_shift_rule_end_sec
        shift_beg_sec = lv_shift_rule_beg_sec
        shift_end_sec = lv_shift_rule_end_sec
      ) to lt_shift_abs.
    endloop.

    sort lt_shift_abs by beg_sec end_sec.

    lt_overtime_abs = build_overtime_intervals( ).
    sort lt_overtime_abs by beg_sec end_sec.

    lt_no_shift_rest_interval_rule =
      build_no_shift_ot_rest_itv( ).

    loop at lt_overtime_abs into data(ls_actual_src).
      data lt_actual_candidate_segment type tt_segment.
      data lt_actual_next_segment type tt_segment.
      clear lt_actual_candidate_segment.

      append value #(
        beg_sec = ls_actual_src-beg_sec
        end_sec = ls_actual_src-end_sec
      ) to lt_actual_candidate_segment.

      loop at lt_shift_abs into data(ls_actual_shift_cover)
        where end_sec > ls_actual_src-beg_sec
          and beg_sec < ls_actual_src-end_sec.

        clear lt_actual_next_segment.

        loop at lt_actual_candidate_segment into data(ls_actual_candidate_segment).
          if ls_actual_shift_cover-end_sec <= ls_actual_candidate_segment-beg_sec
            or ls_actual_shift_cover-beg_sec >= ls_actual_candidate_segment-end_sec.
            append ls_actual_candidate_segment to lt_actual_next_segment.
            continue.
          endif.

          if ls_actual_shift_cover-beg_sec > ls_actual_candidate_segment-beg_sec.
            append value #(
              beg_sec = ls_actual_candidate_segment-beg_sec
              end_sec = min_seconds(
                iv_a = ls_actual_shift_cover-beg_sec
                iv_b = ls_actual_candidate_segment-end_sec )
            ) to lt_actual_next_segment.
          endif.

          if ls_actual_shift_cover-end_sec < ls_actual_candidate_segment-end_sec.
            append value #(
              beg_sec = max_seconds(
                iv_a = ls_actual_shift_cover-end_sec
                iv_b = ls_actual_candidate_segment-beg_sec )
              end_sec = ls_actual_candidate_segment-end_sec
            ) to lt_actual_next_segment.
          endif.
        endloop.

        lt_actual_candidate_segment = lt_actual_next_segment.
      endloop.

      loop at lt_actual_candidate_segment into data(ls_actual_segment_src).
        if ls_actual_segment_src-end_sec <= ls_actual_segment_src-beg_sec.
          continue.
        endif.

        append value #(
          work_date     = ls_actual_src-work_date
          work_type     = gc_work_type_ot
          beg_sec       = ls_actual_segment_src-beg_sec
          end_sec       = ls_actual_segment_src-end_sec
          shift_beg_sec = ls_actual_segment_src-beg_sec
          shift_end_sec = ls_actual_segment_src-end_sec
        ) to lt_actual_overtime_abs.
      endloop.
    endloop.

    sort lt_actual_overtime_abs by beg_sec end_sec.

    loop at lt_overtime_abs into data(ls_no_shift_src).
      data(lv_has_shift_context) = abap_false.

      loop at lt_shift_abs into data(ls_no_shift_cover).
        data(lv_no_shift_cover_beg_sec) =
          ls_no_shift_cover-beg_sec - gc_ot_shift_window_sec.
        data(lv_no_shift_cover_end_sec) =
          ls_no_shift_cover-end_sec + gc_ot_shift_window_sec.

        if lv_no_shift_cover_end_sec > ls_no_shift_src-beg_sec
          and lv_no_shift_cover_beg_sec < ls_no_shift_src-end_sec.
          lv_has_shift_context = abap_true.
          exit.
        endif.
      endloop.

      if lv_has_shift_context = abap_true.
        continue.
      endif.

      append ls_no_shift_src to lt_no_shift_abs.
    endloop.

    sort lt_no_shift_abs by beg_sec end_sec.

    data ls_no_shift_current_group type ty_overtime_group.
    data lv_has_no_shift_group type abap_bool.

    loop at lt_no_shift_abs into data(ls_no_shift_group_src).
      if lv_has_no_shift_group is initial.
        ls_no_shift_current_group-beg_sec = ls_no_shift_group_src-beg_sec.
        ls_no_shift_current_group-end_sec = ls_no_shift_group_src-end_sec.
        lv_has_no_shift_group = abap_true.
        continue.
      endif.

      if ls_no_shift_group_src-beg_sec > ls_no_shift_current_group-end_sec.
        append ls_no_shift_current_group to lt_no_shift_group.
        ls_no_shift_current_group-beg_sec = ls_no_shift_group_src-beg_sec.
        ls_no_shift_current_group-end_sec = ls_no_shift_group_src-end_sec.
      else.
        ls_no_shift_current_group-end_sec = max_seconds(
          iv_a = ls_no_shift_current_group-end_sec
          iv_b = ls_no_shift_group_src-end_sec ).
      endif.
    endloop.

    if lv_has_no_shift_group = abap_true.
      append ls_no_shift_current_group to lt_no_shift_group.
    endif.

    loop at lt_no_shift_abs into data(ls_no_shift_day_src).
      data(lv_no_shift_day_cursor_sec) = ls_no_shift_day_src-beg_sec.

      while lv_no_shift_day_cursor_sec < ls_no_shift_day_src-end_sec.
        seconds_to_datetime(
          exporting
            iv_seconds = lv_no_shift_day_cursor_sec
          importing
            ev_date    = data(lv_no_shift_day_date)
            ev_time    = data(lv_no_shift_day_time) ).

        data lv_no_shift_next_date type d.
        lv_no_shift_next_date = lv_no_shift_day_date + 1.

        data(lv_no_shift_day_end_sec) = min_seconds(
          iv_a = ls_no_shift_day_src-end_sec
          iv_b = to_seconds(
            iv_date = lv_no_shift_next_date
            iv_time = '000000' ) ).

        read table lt_no_shift_day assigning field-symbol(<ls_no_shift_day>)
          with key work_date = lv_no_shift_day_date.
        if sy-subrc <> 0.
          append value #(
            work_date = lv_no_shift_day_date
          ) to lt_no_shift_day assigning <ls_no_shift_day>.
        endif.

        data(lv_no_shift_day_seconds) =
          lv_no_shift_day_end_sec - lv_no_shift_day_cursor_sec.

        data(lv_no_shift_day_rest_seconds) = get_interval_overlap_seconds(
          it_interval = lt_no_shift_rest_interval_rule
          iv_beg_sec  = lv_no_shift_day_cursor_sec
          iv_end_sec  = lv_no_shift_day_end_sec ).
        subtract lv_no_shift_day_rest_seconds from lv_no_shift_day_seconds.

        if lv_no_shift_day_seconds < 0.
          clear lv_no_shift_day_seconds.
        endif.

        add lv_no_shift_day_seconds to <ls_no_shift_day>-overtime_seconds.

        lv_no_shift_day_cursor_sec = lv_no_shift_day_end_sec.
      endwhile.
    endloop.

    loop at lt_overtime_abs into data(ls_overtime_rule).
      data(lv_overtime_seconds_rule) = ls_overtime_rule-end_sec - ls_overtime_rule-beg_sec.
      if lv_overtime_seconds_rule <= 0.
        continue.
      endif.

      data ls_best_shift type ty_abs_interval.
      data lv_best_overlap_sec type ty_seconds.
      data lv_best_window_overlap_sec type ty_seconds.
      data lv_has_shift_rule type abap_bool.
      clear: ls_best_shift,
             lv_best_overlap_sec,
             lv_best_window_overlap_sec,
             lv_has_shift_rule.

      data lv_shift_overlap_seconds_rule type ty_seconds.
      clear lv_shift_overlap_seconds_rule.

      loop at lt_shift_abs into data(ls_shift_match).
        data(lv_shift_overlap_sec) = min_seconds(
          iv_a = ls_overtime_rule-end_sec
          iv_b = ls_shift_match-end_sec )
          - max_seconds(
              iv_a = ls_overtime_rule-beg_sec
              iv_b = ls_shift_match-beg_sec ).
        data(lv_match_shift_sec) = lv_shift_overlap_sec.
        if lv_match_shift_sec < 0.
          clear lv_match_shift_sec.
        endif.
        if lv_match_shift_sec > 0.
          add lv_match_shift_sec to lv_shift_overlap_seconds_rule.
        endif.

        data(lv_match_before_beg_sec) = ls_shift_match-beg_sec - gc_ot_shift_window_sec.
        data(lv_match_before_end_sec) = ls_shift_match-beg_sec.
        data(lv_match_after_beg_sec) = ls_shift_match-end_sec.
        data(lv_match_after_end_sec) = ls_shift_match-end_sec + gc_ot_shift_window_sec.

        data(lv_match_before_sec) = min_seconds(
          iv_a = ls_overtime_rule-end_sec
          iv_b = lv_match_before_end_sec )
          - max_seconds(
              iv_a = ls_overtime_rule-beg_sec
              iv_b = lv_match_before_beg_sec ).
        if lv_match_before_sec < 0.
          clear lv_match_before_sec.
        endif.

        data(lv_match_after_sec) = min_seconds(
          iv_a = ls_overtime_rule-end_sec
          iv_b = lv_match_after_end_sec )
          - max_seconds(
              iv_a = ls_overtime_rule-beg_sec
              iv_b = lv_match_after_beg_sec ).
        if lv_match_after_sec < 0.
          clear lv_match_after_sec.
        endif.

        data(lv_match_window_overlap_sec) =
          lv_match_before_sec + lv_match_after_sec.
        data(lv_match_overlap_sec) =
          lv_match_window_overlap_sec + lv_match_shift_sec.

        if lv_match_overlap_sec > lv_best_overlap_sec.
          lv_best_overlap_sec = lv_match_overlap_sec.
          lv_best_window_overlap_sec = lv_match_window_overlap_sec.
          ls_best_shift = ls_shift_match.
          lv_has_shift_rule = abap_true.
        endif.
      endloop.

      data(lv_actual_ot_seconds) =
        lv_overtime_seconds_rule - lv_shift_overlap_seconds_rule.
      if lv_has_shift_rule is initial.
        data(lv_no_shift_rest_seconds_rule) = get_interval_overlap_seconds(
          it_interval = lt_no_shift_rest_interval_rule
          iv_beg_sec  = ls_overtime_rule-beg_sec
          iv_end_sec  = ls_overtime_rule-end_sec ).
        subtract lv_no_shift_rest_seconds_rule from lv_actual_ot_seconds.
      endif.
      if lv_actual_ot_seconds < 0.
        clear lv_actual_ot_seconds.
      endif.

      data ls_result type ty_overtime_compliance.
      clear ls_result.

      seconds_to_datetime(
        exporting
          iv_seconds = ls_overtime_rule-beg_sec
        importing
          ev_date    = ls_result-beg_date
          ev_time    = ls_result-beg_time ).

      seconds_to_datetime(
        exporting
          iv_seconds = ls_overtime_rule-end_sec
        importing
          ev_date    = ls_result-end_date
          ev_time    = ls_result-end_time ).

      ls_result-overtime_seconds = lv_actual_ot_seconds.
      ls_result-overtime_hours =
        seconds_to_hours( lv_actual_ot_seconds ).
      ls_result-leave_overlap_seconds = get_interval_overlap_seconds(
        it_interval = lt_leave_interval_rule
        iv_beg_sec  = ls_overtime_rule-beg_sec
        iv_end_sec  = ls_overtime_rule-end_sec ).
      ls_result-leave_overlap_hours =
        seconds_to_hours( ls_result-leave_overlap_seconds ).

      ls_result-is_window_valid = abap_true.
      ls_result-is_total_valid = abap_true.
      ls_result-is_day_total_valid = abap_true.
      ls_result-is_continuous_valid = abap_true.
      ls_result-is_prev_work_rest_valid = abap_true.
      ls_result-is_next_shift_rest_valid = abap_true.
      ls_result-is_leave_overlap_valid = abap_true.

      if ls_result-leave_overlap_seconds > 0.
        ls_result-is_leave_overlap_valid = abap_false.
      endif.

      if lv_has_shift_rule = abap_true.
        ls_result-has_shift = abap_true.
        ls_result-shift_work_date = ls_best_shift-work_date.

        seconds_to_datetime(
          exporting
            iv_seconds = ls_best_shift-beg_sec
          importing
            ev_date    = ls_result-shift_beg_date
            ev_time    = ls_result-shift_beg_time ).

        seconds_to_datetime(
          exporting
            iv_seconds = ls_best_shift-end_sec
          importing
            ev_date    = ls_result-shift_end_date
            ev_time    = ls_result-shift_end_time ).

        data(lv_window_beg_sec) = ls_best_shift-beg_sec - gc_ot_shift_window_sec.
        data(lv_before_window_end_sec) = ls_best_shift-beg_sec.
        data(lv_after_window_beg_sec) = ls_best_shift-end_sec.
        data(lv_window_end_sec) = ls_best_shift-end_sec + gc_ot_shift_window_sec.

        seconds_to_datetime(
          exporting
            iv_seconds = lv_window_beg_sec
          importing
            ev_date    = ls_result-window_beg_date
            ev_time    = ls_result-window_beg_time ).

        seconds_to_datetime(
          exporting
            iv_seconds = lv_window_end_sec
          importing
            ev_date    = ls_result-window_end_date
            ev_time    = ls_result-window_end_time ).

        if lv_best_window_overlap_sec > 0.
          data lt_shift_overtime_group type tt_overtime_group.
          data lt_shift_window_segment type tt_segment.
          data ls_shift_current_group type ty_overtime_group.
          data lv_has_shift_group type abap_bool.
          clear: lt_shift_overtime_group,
                 lt_shift_window_segment,
                 ls_shift_current_group,
                 lv_has_shift_group.

          loop at lt_overtime_abs into data(ls_shift_group_src)
            where end_sec > lv_window_beg_sec
              and beg_sec < lv_window_end_sec.

            data(lv_group_before_seconds) = min_seconds(
              iv_a = ls_shift_group_src-end_sec
              iv_b = lv_before_window_end_sec )
              - max_seconds(
                  iv_a = ls_shift_group_src-beg_sec
                  iv_b = lv_window_beg_sec ).
            if lv_group_before_seconds < 0.
              clear lv_group_before_seconds.
            endif.

            data(lv_group_after_seconds) = min_seconds(
              iv_a = ls_shift_group_src-end_sec
              iv_b = lv_window_end_sec )
              - max_seconds(
                  iv_a = ls_shift_group_src-beg_sec
                  iv_b = lv_after_window_beg_sec ).
            if lv_group_after_seconds < 0.
              clear lv_group_after_seconds.
            endif.

            data(lv_group_window_seconds) =
              lv_group_before_seconds + lv_group_after_seconds.
            if lv_group_window_seconds <= 0.
              continue.
            endif.

            if lv_group_before_seconds > 0.
              append value #(
                beg_sec = max_seconds(
                  iv_a = ls_shift_group_src-beg_sec
                  iv_b = lv_window_beg_sec )
                end_sec = min_seconds(
                  iv_a = ls_shift_group_src-end_sec
                  iv_b = lv_before_window_end_sec )
              ) to lt_shift_window_segment.
            endif.

            if lv_group_after_seconds > 0.
              append value #(
                beg_sec = max_seconds(
                  iv_a = ls_shift_group_src-beg_sec
                  iv_b = lv_after_window_beg_sec )
                end_sec = min_seconds(
                  iv_a = ls_shift_group_src-end_sec
                  iv_b = lv_window_end_sec )
              ) to lt_shift_window_segment.
            endif.
          endloop.

          sort lt_shift_window_segment by beg_sec end_sec.

          loop at lt_shift_window_segment into data(ls_shift_window_segment).
            if lv_has_shift_group is initial.
              ls_shift_current_group-beg_sec = ls_shift_window_segment-beg_sec.
              ls_shift_current_group-end_sec = ls_shift_window_segment-end_sec.
              lv_has_shift_group = abap_true.
              continue.
            endif.

            if ls_shift_window_segment-beg_sec > ls_shift_current_group-end_sec.
              append ls_shift_current_group to lt_shift_overtime_group.
              ls_shift_current_group-beg_sec = ls_shift_window_segment-beg_sec.
              ls_shift_current_group-end_sec = ls_shift_window_segment-end_sec.
            else.
              ls_shift_current_group-end_sec = max_seconds(
                iv_a = ls_shift_current_group-end_sec
                iv_b = ls_shift_window_segment-end_sec ).
            endif.
          endloop.

          if lv_has_shift_group = abap_true.
            append ls_shift_current_group to lt_shift_overtime_group.
          endif.

          loop at lt_shift_overtime_group into data(ls_shift_group_match)
            where end_sec > ls_overtime_rule-beg_sec
              and beg_sec < ls_overtime_rule-end_sec.

            ls_result-continuous_seconds =
              ls_shift_group_match-end_sec - ls_shift_group_match-beg_sec.
            ls_result-continuous_hours =
              seconds_to_hours( ls_result-continuous_seconds ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_shift_group_match-beg_sec
              importing
                ev_date    = ls_result-continuous_beg_date
                ev_time    = ls_result-continuous_beg_time ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_shift_group_match-end_sec
              importing
                ev_date    = ls_result-continuous_end_date
                ev_time    = ls_result-continuous_end_time ).
            exit.
          endloop.
        endif.

        data(lv_self_before_seconds) = min_seconds(
          iv_a = ls_overtime_rule-end_sec
          iv_b = lv_before_window_end_sec )
          - max_seconds(
              iv_a = ls_overtime_rule-beg_sec
              iv_b = lv_window_beg_sec ).
        if lv_self_before_seconds < 0.
          clear lv_self_before_seconds.
        endif.

        data(lv_self_after_seconds) = min_seconds(
          iv_a = ls_overtime_rule-end_sec
          iv_b = lv_window_end_sec )
          - max_seconds(
              iv_a = ls_overtime_rule-beg_sec
              iv_b = lv_after_window_beg_sec ).
        if lv_self_after_seconds < 0.
          clear lv_self_after_seconds.
        endif.

        data(lv_self_window_seconds) = lv_self_before_seconds + lv_self_after_seconds.

        ls_result-outside_window_seconds =
          lv_actual_ot_seconds - lv_self_window_seconds.
        if ls_result-outside_window_seconds < 0.
          clear ls_result-outside_window_seconds.
        endif.
        ls_result-outside_window_hours =
          seconds_to_hours( ls_result-outside_window_seconds ).

        data lv_shift_limit_total_seconds type ty_seconds.
        clear lv_shift_limit_total_seconds.

        loop at lt_overtime_abs into data(ls_window_overtime)
          where end_sec > lv_window_beg_sec
            and beg_sec < lv_window_end_sec.

          data(lv_sum_before_seconds) = min_seconds(
            iv_a = ls_window_overtime-end_sec
            iv_b = lv_before_window_end_sec )
            - max_seconds(
                iv_a = ls_window_overtime-beg_sec
                iv_b = lv_window_beg_sec ).
          if lv_sum_before_seconds < 0.
            clear lv_sum_before_seconds.
          endif.

          data(lv_sum_after_seconds) = min_seconds(
            iv_a = ls_window_overtime-end_sec
            iv_b = lv_window_end_sec )
            - max_seconds(
                iv_a = ls_window_overtime-beg_sec
                iv_b = lv_after_window_beg_sec ).
          if lv_sum_after_seconds < 0.
            clear lv_sum_after_seconds.
          endif.

          data(lv_sum_window_seconds) = lv_sum_before_seconds + lv_sum_after_seconds.

          if lv_sum_window_seconds > 0.
            add lv_sum_window_seconds to ls_result-window_overtime_seconds.
          endif.

          data(lv_sum_context_overlap_seconds) = min_seconds(
            iv_a = ls_window_overtime-end_sec
            iv_b = lv_window_end_sec )
            - max_seconds(
                iv_a = ls_window_overtime-beg_sec
                iv_b = lv_window_beg_sec ).
          if lv_sum_context_overlap_seconds < 0.
            clear lv_sum_context_overlap_seconds.
          endif.

          if lv_sum_context_overlap_seconds > 0.
            data lv_sum_shift_overlap_seconds type ty_seconds.
            clear lv_sum_shift_overlap_seconds.

            loop at lt_shift_abs into data(ls_sum_shift_cover)
              where end_sec > ls_window_overtime-beg_sec
                and beg_sec < ls_window_overtime-end_sec.

              data(lv_sum_shift_overlap_part) = min_seconds(
                iv_a = ls_window_overtime-end_sec
                iv_b = ls_sum_shift_cover-end_sec )
                - max_seconds(
                    iv_a = ls_window_overtime-beg_sec
                    iv_b = ls_sum_shift_cover-beg_sec ).
              if lv_sum_shift_overlap_part > 0.
                add lv_sum_shift_overlap_part to lv_sum_shift_overlap_seconds.
              endif.
            endloop.

            data(lv_sum_actual_overtime_seconds) =
              ls_window_overtime-end_sec
              - ls_window_overtime-beg_sec
              - lv_sum_shift_overlap_seconds.
            if lv_sum_actual_overtime_seconds < 0.
              clear lv_sum_actual_overtime_seconds.
            endif.

            add lv_sum_actual_overtime_seconds to lv_shift_limit_total_seconds.
          endif.
        endloop.

        ls_result-window_overtime_hours =
          seconds_to_hours( ls_result-window_overtime_seconds ).

        if lv_shift_limit_total_seconds > gc_ot_shift_limit_sec.
          ls_result-is_total_valid = abap_false.
        endif.

        if ls_result-continuous_seconds > gc_ot_shift_limit_sec.
          ls_result-is_continuous_valid = abap_false.
        endif.
      endif.

      data lv_has_no_shift_part type abap_bool.
      clear lv_has_no_shift_part.

      loop at lt_no_shift_abs into data(ls_no_shift_rule_part)
        where end_sec > ls_overtime_rule-beg_sec
          and beg_sec < ls_overtime_rule-end_sec.

        lv_has_no_shift_part = abap_true.

        data(lv_no_shift_prev_ref_beg) = ls_no_shift_rule_part-beg_sec.

        loop at lt_no_shift_group into data(ls_no_shift_group_match)
          where end_sec > ls_no_shift_rule_part-beg_sec
            and beg_sec < ls_no_shift_rule_part-end_sec.

          if ls_no_shift_group_match-beg_sec < lv_no_shift_prev_ref_beg.
            lv_no_shift_prev_ref_beg = ls_no_shift_group_match-beg_sec.
          endif.

          data(lv_no_shift_continuous_sec) =
            ls_no_shift_group_match-end_sec - ls_no_shift_group_match-beg_sec.
          data(lv_no_shift_group_rest_seconds) = get_interval_overlap_seconds(
            it_interval = lt_no_shift_rest_interval_rule
            iv_beg_sec  = ls_no_shift_group_match-beg_sec
            iv_end_sec  = ls_no_shift_group_match-end_sec ).
          subtract lv_no_shift_group_rest_seconds from lv_no_shift_continuous_sec.
          if lv_no_shift_continuous_sec < 0.
            clear lv_no_shift_continuous_sec.
          endif.

          if lv_no_shift_continuous_sec > ls_result-continuous_seconds.
            ls_result-continuous_seconds = lv_no_shift_continuous_sec.
            ls_result-continuous_hours =
              seconds_to_hours( ls_result-continuous_seconds ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_no_shift_group_match-beg_sec
              importing
                ev_date    = ls_result-continuous_beg_date
                ev_time    = ls_result-continuous_beg_time ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_no_shift_group_match-end_sec
              importing
                ev_date    = ls_result-continuous_end_date
                ev_time    = ls_result-continuous_end_time ).
          endif.
        endloop.

        data lv_prev_work_found type abap_bool.
        data ls_prev_work_rule type ty_abs_interval.
        clear: lv_prev_work_found,
               ls_prev_work_rule.

        loop at lt_shift_abs into data(ls_prev_shift_candidate)
          where end_sec <= lv_no_shift_prev_ref_beg.
          if lv_prev_work_found is initial
            or ls_prev_shift_candidate-end_sec > ls_prev_work_rule-end_sec.
            ls_prev_work_rule = ls_prev_shift_candidate.
            lv_prev_work_found = abap_true.
          endif.
        endloop.

        loop at lt_actual_overtime_abs into data(ls_prev_ot_candidate)
          where end_sec <= lv_no_shift_prev_ref_beg.
          if lv_prev_work_found is initial
            or ls_prev_ot_candidate-end_sec > ls_prev_work_rule-end_sec.
            ls_prev_work_rule = ls_prev_ot_candidate.
            lv_prev_work_found = abap_true.
          endif.
        endloop.

        if lv_prev_work_found = abap_true.
          data(lv_prev_rest_seconds) =
            lv_no_shift_prev_ref_beg - ls_prev_work_rule-end_sec.

          if ls_result-prev_work_end_date is initial
            or lv_prev_rest_seconds < ls_result-prev_work_rest_seconds.

            ls_result-prev_work_date = ls_prev_work_rule-work_date.
            ls_result-prev_work_rest_seconds = lv_prev_rest_seconds.
            ls_result-prev_work_rest_hours =
              seconds_to_hours( lv_prev_rest_seconds ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_prev_work_rule-end_sec
              importing
                ev_date    = ls_result-prev_work_end_date
                ev_time    = ls_result-prev_work_end_time ).
          endif.

          if lv_prev_rest_seconds < gc_ot_rest_limit_sec.
            ls_result-is_prev_work_rest_valid = abap_false.
          endif.
        endif.

        data lv_next_shift_found type abap_bool.
        data ls_next_shift_rule type ty_abs_interval.
        clear: lv_next_shift_found,
               ls_next_shift_rule.

        loop at lt_shift_abs into data(ls_next_shift_candidate)
          where beg_sec >= ls_no_shift_rule_part-end_sec.
          ls_next_shift_rule = ls_next_shift_candidate.
          lv_next_shift_found = abap_true.
          exit.
        endloop.

        if lv_next_shift_found = abap_true.
          data(lv_next_rest_seconds) =
            ls_next_shift_rule-beg_sec - ls_no_shift_rule_part-end_sec.

          if ls_result-next_shift_beg_date is initial
            or lv_next_rest_seconds < ls_result-next_shift_rest_seconds.

            ls_result-next_shift_work_date = ls_next_shift_rule-work_date.
            ls_result-next_shift_rest_seconds = lv_next_rest_seconds.
            ls_result-next_shift_rest_hours =
              seconds_to_hours( lv_next_rest_seconds ).

            seconds_to_datetime(
              exporting
                iv_seconds = ls_next_shift_rule-beg_sec
              importing
                ev_date    = ls_result-next_shift_beg_date
                ev_time    = ls_result-next_shift_beg_time ).
          endif.

          if lv_next_rest_seconds < gc_ot_rest_limit_sec.
            ls_result-is_next_shift_rest_valid = abap_false.
          endif.
        endif.

        data(lv_no_shift_part_cursor_sec) = ls_no_shift_rule_part-beg_sec.

        while lv_no_shift_part_cursor_sec < ls_no_shift_rule_part-end_sec.
          seconds_to_datetime(
            exporting
              iv_seconds = lv_no_shift_part_cursor_sec
            importing
              ev_date    = data(lv_no_shift_part_date)
              ev_time    = data(lv_no_shift_part_time) ).

          read table lt_no_shift_day into data(ls_no_shift_day_result)
            with key work_date = lv_no_shift_part_date.
          if sy-subrc = 0
            and ls_no_shift_day_result-overtime_seconds >
                ls_result-no_shift_day_overtime_seconds.
            ls_result-no_shift_day_overtime_seconds =
              ls_no_shift_day_result-overtime_seconds.
            ls_result-no_shift_day_overtime_hours =
              seconds_to_hours( ls_result-no_shift_day_overtime_seconds ).
          endif.

          data lv_no_shift_part_next_date type d.
          lv_no_shift_part_next_date = lv_no_shift_part_date + 1.

          lv_no_shift_part_cursor_sec = min_seconds(
            iv_a = ls_no_shift_rule_part-end_sec
            iv_b = to_seconds(
              iv_date = lv_no_shift_part_next_date
              iv_time = '000000' ) ).
        endwhile.
      endloop.

      if lv_has_no_shift_part = abap_true.
        if ls_result-continuous_seconds > gc_no_shift_ot_limit_sec.
          ls_result-is_continuous_valid = abap_false.
        endif.

        if ls_result-no_shift_day_overtime_seconds > gc_no_shift_ot_limit_sec.
          ls_result-is_day_total_valid = abap_false.
        endif.
      endif.

      if ls_result-is_window_valid = abap_true
        and ls_result-is_total_valid = abap_true
        and ls_result-is_day_total_valid = abap_true
        and ls_result-is_continuous_valid = abap_true
        and ls_result-is_prev_work_rest_valid = abap_true
        and ls_result-is_leave_overlap_valid = abap_true.
        if ls_result-is_next_shift_rest_valid = abap_true.
          ls_result-is_valid = abap_true.
        else.
          ls_result-is_valid = abap_false.
        endif.
      else.
        ls_result-is_valid = abap_false.
      endif.

      append ls_result to rt_result.
    endloop.

    sort rt_result by beg_date beg_time end_date end_time.

  endmethod.

  method get_period_dates.

    ev_date_from = iv_date_from.
    ev_date_to   = iv_date_to.

    if iv_date_from is initial or iv_date_to is initial.
      loop at mt_shift into data(ls_shift).
        if iv_date_from is initial
          and ( ev_date_from is initial or ls_shift-work_date < ev_date_from ).
          ev_date_from = ls_shift-work_date.
        endif.
        if iv_date_to is initial
          and ( ev_date_to is initial or ls_shift-work_date > ev_date_to ).
          ev_date_to = ls_shift-work_date.
        endif.
      endloop.

      loop at mt_overtime into data(ls_overtime_date).
        if ls_overtime_date-beg_date is initial.
          continue.
        endif.

        if iv_date_from is initial
          and ( ev_date_from is initial or ls_overtime_date-beg_date < ev_date_from ).
          ev_date_from = ls_overtime_date-beg_date.
        endif.
        if iv_date_to is initial
          and ( ev_date_to is initial or ls_overtime_date-beg_date > ev_date_to ).
          ev_date_to = ls_overtime_date-beg_date.
        endif.
      endloop.
    endif.

  endmethod.

  method get_week_dates.

    if iv_base_date is initial.
      return.
    endif.

    data(lv_week_offset) = ( iv_base_date - gc_monday_base ) mod 7.
    if lv_week_offset < 0.
      lv_week_offset = lv_week_offset + 7.
    endif.

    ev_date_from = iv_base_date - lv_week_offset.
    ev_date_to = ev_date_from + 6.

  endmethod.

  method get_month_dates.

    if iv_base_date is initial.
      return.
    endif.

    ev_date_from = iv_base_date.
    ev_date_from+6(2) = '01'.

    data(lv_year) = conv i( ev_date_from+0(4) ).
    data(lv_month) = conv i( ev_date_from+4(2) ).

    if lv_month = 12.
      lv_year = lv_year + 1.
      lv_month = 1.
    else.
      lv_month = lv_month + 1.
    endif.

    data lv_next_month_beg type d.
    lv_next_month_beg = |{ lv_year width = 4 pad = '0' align = right }{ lv_month width = 2 pad = '0' align = right }01|.
    ev_date_to = lv_next_month_beg - 1.

  endmethod.

  method get_around_dates.

    if iv_base_date is initial.
      return.
    endif.

    data(lv_days) = iv_days.
    if lv_days < 0.
      lv_days = lv_days * -1.
    endif.

    ev_date_from = iv_base_date - lv_days.
    ev_date_to = iv_base_date + lv_days.

  endmethod.

  method get_shift_bounds.

    data(lv_beg_date) = is_shift-work_date.
    data(lv_end_date) = is_shift-end_date.

    if lv_end_date is initial.
      lv_end_date = is_shift-work_date.
      if is_shift-end_time <= is_shift-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_shift-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_shift-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_interval_bounds.

    data: lv_shift_beg_sec type ty_seconds,
          lv_shift_end_sec type ty_seconds.

    get_shift_bounds(
      exporting
        is_shift   = is_shift
      importing
        ev_beg_sec = lv_shift_beg_sec
        ev_end_sec = lv_shift_end_sec ).

    data(lv_shift_cross_day) = abap_false.
    data lv_next_date type d.
    lv_next_date = is_shift-work_date + 1.
    if lv_shift_end_sec >= to_seconds( iv_date = lv_next_date iv_time = '000000' ).
      lv_shift_cross_day = abap_true.
    endif.

    data(lv_beg_date) = is_interval-beg_date.
    if lv_beg_date is initial.
      lv_beg_date = is_shift-work_date.
      if lv_shift_cross_day = abap_true and is_interval-beg_time < is_shift-beg_time.
        lv_beg_date = lv_beg_date + 1.
      endif.
    endif.

    data(lv_end_date) = is_interval-end_date.
    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
      if is_interval-end_time <= is_interval-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_interval-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_interval-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_leave_data_bounds.

    data(lv_beg_date) = is_leave-beg_date.
    data(lv_end_date) = is_leave-end_date.

    if lv_beg_date is initial.
      return.
    endif.

    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
    endif.

    if lv_beg_date <> lv_end_date
      or ( is_leave-beg_time is initial and is_leave-end_time is initial ).

      ev_beg_sec = to_seconds(
        iv_date = lv_beg_date
        iv_time = '000000' ).

      data lv_day_after_end type d.
      lv_day_after_end = lv_end_date + 1.
      ev_end_sec = to_seconds(
        iv_date = lv_day_after_end
        iv_time = '000000' ).

      return.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_leave-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_leave-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_overtime_bounds.

    data(lv_beg_date) = is_overtime-beg_date.
    data(lv_end_date) = is_overtime-end_date.

    if lv_beg_date is initial.
      return.
    endif.

    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
      if is_overtime-end_time <= is_overtime-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_overtime-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_overtime-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method get_no_shift_ot_rest_bounds.

    data(lv_beg_date) = is_rest-beg_date.
    data(lv_end_date) = is_rest-end_date.

    if lv_beg_date is initial.
      return.
    endif.

    if lv_end_date is initial.
      lv_end_date = lv_beg_date.
      if is_rest-end_time <= is_rest-beg_time.
        lv_end_date = lv_end_date + 1.
      endif.
    endif.

    ev_beg_sec = to_seconds(
      iv_date = lv_beg_date
      iv_time = is_rest-beg_time ).

    ev_end_sec = to_seconds(
      iv_date = lv_end_date
      iv_time = is_rest-end_time ).

    if ev_end_sec <= ev_beg_sec.
      ev_end_sec = ev_end_sec + gc_day_seconds.
    endif.

  endmethod.

  method overtime_has_shift_overlap.

    loop at mt_shift into data(ls_shift_overlap).
      if ls_shift_overlap-rest_flag is not initial.
        continue.
      endif.

      get_shift_bounds(
        exporting
          is_shift   = ls_shift_overlap
        importing
          ev_beg_sec = data(lv_shift_beg_sec)
          ev_end_sec = data(lv_shift_end_sec) ).

      if lv_shift_end_sec > iv_beg_sec and lv_shift_beg_sec < iv_end_sec.
        rv_result = abap_true.
        return.
      endif.
    endloop.

  endmethod.

  method has_shift_overtime_context.

    loop at it_shift_interval into data(ls_shift_context).
      data(lv_context_beg_sec) =
        ls_shift_context-beg_sec - gc_ot_shift_window_sec.
      data(lv_context_end_sec) =
        ls_shift_context-end_sec + gc_ot_shift_window_sec.

      if lv_context_end_sec > iv_beg_sec
        and lv_context_beg_sec < iv_end_sec.
        rv_result = abap_true.
        return.
      endif.
    endloop.

  endmethod.

  method get_shift_work_seconds.

    data lt_segment type tt_segment.

    if it_leave_interval is supplied.
      lt_segment = get_shift_work_segments(
        is_shift          = is_shift
        it_leave_interval = it_leave_interval ).
    else.
      lt_segment = get_shift_work_segments( is_shift = is_shift ).
    endif.

    loop at lt_segment into data(ls_segment).
      add ls_segment-end_sec to rv_result.
      subtract ls_segment-beg_sec from rv_result.
    endloop.

  endmethod.

  method get_shift_work_segments.

    if is_shift-rest_flag is not initial.
      return.
    endif.

    data: lv_shift_beg_sec type ty_seconds,
          lv_shift_end_sec type ty_seconds.

    get_shift_bounds(
      exporting
        is_shift   = is_shift
      importing
        ev_beg_sec = lv_shift_beg_sec
        ev_end_sec = lv_shift_end_sec ).

    data lt_work_segment type tt_segment.

    append value #(
      beg_sec = lv_shift_beg_sec
      end_sec = lv_shift_end_sec
    ) to lt_work_segment.

    data lt_exclude type tt_segment.

    loop at is_shift-breaks into data(ls_break).
      get_interval_bounds(
        exporting
          is_shift    = is_shift
          is_interval = ls_break
        importing
          ev_beg_sec  = data(lv_break_beg_sec)
          ev_end_sec  = data(lv_break_end_sec) ).

      if lv_break_end_sec > lv_shift_beg_sec and lv_break_beg_sec < lv_shift_end_sec.
        append value #(
          beg_sec = max_seconds( iv_a = lv_break_beg_sec iv_b = lv_shift_beg_sec )
          end_sec = min_seconds( iv_a = lv_break_end_sec iv_b = lv_shift_end_sec )
        ) to lt_exclude.
      endif.
    endloop.

    if it_leave_interval is supplied.
      loop at it_leave_interval into data(ls_leave_interval)
        where end_sec > lv_shift_beg_sec
          and beg_sec < lv_shift_end_sec.

        append value #(
          beg_sec = max_seconds( iv_a = ls_leave_interval-beg_sec iv_b = lv_shift_beg_sec )
          end_sec = min_seconds( iv_a = ls_leave_interval-end_sec iv_b = lv_shift_end_sec )
        ) to lt_exclude.
      endloop.
    endif.

    sort lt_exclude by beg_sec end_sec.

    data lt_new_segment type tt_segment.
    data ls_segment type ty_segment.
    data lv_segment_count type i.

    loop at lt_exclude into data(ls_exclude).
      clear lt_new_segment.

      lv_segment_count = lines( lt_work_segment ).

      do lv_segment_count times.
        read table lt_work_segment into ls_segment index sy-index.
        if sy-subrc <> 0.
          continue.
        endif.

        if ls_exclude-end_sec <= ls_segment-beg_sec or ls_exclude-beg_sec >= ls_segment-end_sec.
          append ls_segment to lt_new_segment.
          continue.
        endif.

        if ls_exclude-beg_sec > ls_segment-beg_sec.
          append value #(
            beg_sec = ls_segment-beg_sec
            end_sec = min_seconds( iv_a = ls_exclude-beg_sec iv_b = ls_segment-end_sec )
          ) to lt_new_segment.
        endif.

        if ls_exclude-end_sec < ls_segment-end_sec.
          append value #(
            beg_sec = max_seconds( iv_a = ls_exclude-end_sec iv_b = ls_segment-beg_sec )
            end_sec = ls_segment-end_sec
          ) to lt_new_segment.
        endif.
      enddo.

      lt_work_segment = lt_new_segment.
    endloop.

    loop at lt_work_segment into data(ls_valid_segment).
      if ls_valid_segment-end_sec > ls_valid_segment-beg_sec.
        append ls_valid_segment to rt_result.
      endif.
    endloop.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method build_leave_intervals.

    loop at mt_leave into data(ls_leave_data).
      data(lv_leave_end_date) = ls_leave_data-end_date.
      if lv_leave_end_date is initial.
        lv_leave_end_date = ls_leave_data-beg_date.
      endif.

      get_leave_data_bounds(
        exporting
          is_leave   = ls_leave_data
        importing
          ev_beg_sec = data(lv_leave_beg_sec)
          ev_end_sec = data(lv_leave_end_sec) ).

      if mv_leave_extend_calc is not initial
        and ls_leave_data-beg_date is not initial
        and ( ls_leave_data-beg_date <> lv_leave_end_date
              or ( ls_leave_data-beg_time is initial
                   and ls_leave_data-end_time is initial ) ).

        loop at mt_shift into data(ls_leave_shift)
          where work_date >= ls_leave_data-beg_date
            and work_date <= lv_leave_end_date.

          if ls_leave_shift-rest_flag is not initial.
            continue.
          endif.

          get_shift_bounds(
            exporting
              is_shift   = ls_leave_shift
            importing
              ev_beg_sec = data(lv_shift_beg_sec)
              ev_end_sec = data(lv_shift_end_sec) ).

          if lv_shift_beg_sec >= lv_leave_beg_sec
            and lv_shift_beg_sec < lv_leave_end_sec
            and lv_shift_end_sec > lv_leave_end_sec.
            lv_leave_end_sec = lv_shift_end_sec.
          endif.
        endloop.
      endif.

      if lv_leave_end_sec > lv_leave_beg_sec.
        append value #(
          work_date = ls_leave_data-beg_date
          beg_sec   = lv_leave_beg_sec
          end_sec   = lv_leave_end_sec
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method build_work_intervals.

    data(lt_leave_interval) = build_leave_intervals( ).

    loop at mt_shift into data(ls_shift).
      get_shift_bounds(
        exporting
          is_shift   = ls_shift
        importing
          ev_beg_sec = data(lv_shift_beg_sec)
          ev_end_sec = data(lv_shift_end_sec) ).

      data(lt_segment) = get_shift_work_segments(
        is_shift          = ls_shift
        it_leave_interval = lt_leave_interval ).

      loop at lt_segment into data(ls_segment).
        append value #(
          work_date     = ls_shift-work_date
          work_type     = gc_work_type_shift
          beg_sec       = ls_segment-beg_sec
          end_sec       = ls_segment-end_sec
          shift_beg_sec = lv_shift_beg_sec
          shift_end_sec = lv_shift_end_sec
        ) to rt_result.
      endloop.
    endloop.

    data(lt_overtime_interval) = build_overtime_intervals( ).
    append lines of lt_overtime_interval to rt_result.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method build_overtime_intervals.

    loop at mt_overtime into data(ls_overtime).
      get_overtime_bounds(
        exporting
          is_overtime = ls_overtime
        importing
          ev_beg_sec  = data(lv_beg_sec)
          ev_end_sec  = data(lv_end_sec) ).

      if lv_end_sec > lv_beg_sec.
        append value #(
          work_date     = ls_overtime-beg_date
          work_type     = gc_work_type_ot
          beg_sec       = lv_beg_sec
          end_sec       = lv_end_sec
          shift_beg_sec = lv_beg_sec
          shift_end_sec = lv_end_sec
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method build_no_shift_ot_rest_itv.

      get_no_shift_ot_rest_bounds(
        exporting
          is_rest    = ls_rest
        importing
          ev_beg_sec = data(lv_beg_sec)
          ev_end_sec = data(lv_end_sec) ).

      if lv_end_sec > lv_beg_sec.
        append value #(
          beg_sec = lv_beg_sec
          end_sec = lv_end_sec
        ) to rt_result.
      endif.
    endloop.

    sort rt_result by beg_sec end_sec.

  endmethod.

  method get_interval_overlap_seconds.

    data lt_overlap type tt_segment.

    loop at it_interval into data(ls_interval)
      where end_sec > iv_beg_sec
        and beg_sec < iv_end_sec.

      append value #(
        beg_sec = max_seconds(
          iv_a = ls_interval-beg_sec
          iv_b = iv_beg_sec )
        end_sec = min_seconds(
          iv_a = ls_interval-end_sec
          iv_b = iv_end_sec )
      ) to lt_overlap.
    endloop.

    sort lt_overlap by beg_sec end_sec.

    data lv_has_current type abap_bool.
    data lv_current_beg_sec type ty_seconds.
    data lv_current_end_sec type ty_seconds.

    loop at lt_overlap into data(ls_overlap).
      if ls_overlap-end_sec <= ls_overlap-beg_sec.
        continue.
      endif.

      if lv_has_current is initial.
        lv_current_beg_sec = ls_overlap-beg_sec.
        lv_current_end_sec = ls_overlap-end_sec.
        lv_has_current = abap_true.
        continue.
      endif.

      if ls_overlap-beg_sec > lv_current_end_sec.
        add lv_current_end_sec to rv_result.
        subtract lv_current_beg_sec from rv_result.
        lv_current_beg_sec = ls_overlap-beg_sec.
        lv_current_end_sec = ls_overlap-end_sec.
      else.
        lv_current_end_sec = max_seconds(
          iv_a = lv_current_end_sec
          iv_b = ls_overlap-end_sec ).
      endif.
    endloop.

    if lv_has_current = abap_true.
      add lv_current_end_sec to rv_result.
      subtract lv_current_beg_sec from rv_result.
    endif.

  endmethod.

  method get_rest_intervals_for_bounds.

    data(lv_cursor_sec) = iv_beg_sec.
    data ls_prev_work type ty_abs_interval.
    data lv_has_prev_work type abap_bool.
    data lt_work_interval type tt_abs_interval.

    lt_work_interval = it_work_interval.
    sort lt_work_interval by beg_sec end_sec.

    loop at lt_work_interval into data(ls_work_interval)
      where end_sec > iv_beg_sec
        and beg_sec < iv_end_sec.

      data(ls_curr_work) = ls_work_interval.
      ls_curr_work-beg_sec = max_seconds( iv_a = ls_work_interval-beg_sec iv_b = iv_beg_sec ).
      ls_curr_work-end_sec = min_seconds( iv_a = ls_work_interval-end_sec iv_b = iv_end_sec ).

      if ls_curr_work-beg_sec > lv_cursor_sec.
        if lv_has_prev_work = abap_true.
          data(lv_skip_rest) = abap_false.
          if ls_prev_work-work_type = gc_work_type_shift
            and ls_curr_work-work_type = gc_work_type_ot
            and ls_curr_work-beg_sec - ls_prev_work-end_sec < gc_long_ot_rest_sec.
            lv_skip_rest = abap_true.
          endif.

          if lv_skip_rest = abap_false.
            append_rest_result(
              exporting
                iv_beg_sec   = lv_cursor_sec
                iv_end_sec   = ls_curr_work-beg_sec
                is_prev_work = ls_prev_work
                is_next_work = ls_curr_work
              changing
                ct_result    = rt_result ).
          endif.
        else.
          append_rest_result(
            exporting
              iv_beg_sec   = lv_cursor_sec
              iv_end_sec   = ls_curr_work-beg_sec
              is_next_work = ls_curr_work
            changing
              ct_result    = rt_result ).
        endif.
      endif.

      if ls_curr_work-end_sec > lv_cursor_sec.
        lv_cursor_sec = ls_curr_work-end_sec.
        ls_prev_work = ls_curr_work.
        lv_has_prev_work = abap_true.
      endif.
    endloop.

    if iv_end_sec > lv_cursor_sec.
      if lv_has_prev_work = abap_true.
        append_rest_result(
          exporting
            iv_beg_sec   = lv_cursor_sec
            iv_end_sec   = iv_end_sec
            is_prev_work = ls_prev_work
          changing
            ct_result    = rt_result ).
      else.
        append_rest_result(
          exporting
            iv_beg_sec = lv_cursor_sec
            iv_end_sec = iv_end_sec
          changing
            ct_result  = rt_result ).
      endif.
    endif.

  endmethod.

  method append_rest_result.

    if iv_end_sec <= iv_beg_sec.
      return.
    endif.

    seconds_to_datetime(
      exporting
        iv_seconds = iv_beg_sec
      importing
        ev_date    = data(lv_beg_date)
        ev_time    = data(lv_beg_time) ).

    seconds_to_datetime(
      exporting
        iv_seconds = iv_end_sec
      importing
        ev_date    = data(lv_end_date)
        ev_time    = data(lv_end_time) ).

    data ls_result type ty_longest_rest.
    ls_result-rest_date = lv_beg_date.
    ls_result-beg_date = lv_beg_date.
    ls_result-beg_time = lv_beg_time.
    ls_result-end_date = lv_end_date.
    ls_result-end_time = lv_end_time.
    ls_result-rest_seconds = iv_end_sec - iv_beg_sec.
    ls_result-rest_hours = seconds_to_hours( ls_result-rest_seconds ).

    if is_prev_work is supplied.
      ls_result-prev_work_type = is_prev_work-work_type.
      ls_result-prev_shift_work_date = is_prev_work-work_date.
      seconds_to_datetime(
        exporting
          iv_seconds = is_prev_work-shift_beg_sec
        importing
          ev_date    = ls_result-prev_shift_beg_date
          ev_time    = ls_result-prev_shift_beg_time ).
      seconds_to_datetime(
        exporting
          iv_seconds = is_prev_work-shift_end_sec
        importing
          ev_date    = ls_result-prev_shift_end_date
          ev_time    = ls_result-prev_shift_end_time ).
    endif.

    if is_next_work is supplied.
      ls_result-next_work_type = is_next_work-work_type.
      ls_result-next_shift_work_date = is_next_work-work_date.
      seconds_to_datetime(
        exporting
          iv_seconds = is_next_work-shift_beg_sec
        importing
          ev_date    = ls_result-next_shift_beg_date
          ev_time    = ls_result-next_shift_beg_time ).
      seconds_to_datetime(
        exporting
          iv_seconds = is_next_work-shift_end_sec
        importing
          ev_date    = ls_result-next_shift_end_date
          ev_time    = ls_result-next_shift_end_time ).
    endif.

    append ls_result to ct_result.

  endmethod.

  method to_seconds.

    data(lv_time) = conv string( iv_time ).
    data(lv_hour) = conv i( lv_time+0(2) ).
    data(lv_minute) = conv i( lv_time+2(2) ).
    data(lv_second) = conv i( lv_time+4(2) ).

    rv_result = ( iv_date - gc_base_date ) * gc_day_seconds
              + lv_hour * 3600
              + lv_minute * 60
              + lv_second.

  endmethod.

  method seconds_to_datetime.

    data(lv_days) = iv_seconds div gc_day_seconds.
    data(lv_remain) = iv_seconds mod gc_day_seconds.

    if lv_remain < 0.
      lv_remain = lv_remain + gc_day_seconds.
      lv_days = lv_days - 1.
    endif.

    ev_date = gc_base_date + lv_days.

    data(lv_hour) = lv_remain div 3600.
    lv_remain = lv_remain mod 3600.
    data(lv_minute) = lv_remain div 60.
    data(lv_second) = lv_remain mod 60.

    ev_time = |{ lv_hour width = 2 pad = '0' align = right }{ lv_minute width = 2 pad = '0' align = right }{ lv_second width = 2 pad = '0' align = right }|.

  endmethod.

  method seconds_to_hours.
    rv_result = iv_seconds / 3600.
  endmethod.

  method max_seconds.
    rv_result = iv_a.
    if iv_b > rv_result.
      rv_result = iv_b.
    endif.
  endmethod.

  method min_seconds.
    rv_result = iv_a.
    if iv_b < rv_result.
      rv_result = iv_b.
    endif.
  endmethod.

  method add_case_reason.
    if iv_reason is initial.
      return.
    endif.

    data lt_reason type standard table of string with default key.
    split cv_reason at ';' into table lt_reason.

    read table lt_reason with key table_line = iv_reason transporting no fields.
    if sy-subrc = 0.
      return.
    endif.

    if cv_reason is initial.
      cv_reason = iv_reason.
    else.
      cv_reason = |{ cv_reason };{ iv_reason }|.
    endif.
  endmethod.

endclass.
```

### 数据类型说明

#### 基础类型

| 类型 | ABAP 类型 | 说明 |
| --- | --- | --- |
| `ty_seconds` | `int8` | 秒数。内部所有时间跨度统一用秒计算，避免跨天时直接用 `T` 类型相减。 |
| `ty_hours` | `p length 15 decimals 2` | 小时数，保留 2 位小数。方法入参中的小时阈值均使用该类型。 |
| `ty_flag` | `c length 1` | 单字符标识类型，用于 `rest_flag` 等字段。 |
| `ty_work_type` | `c length 1` | 工作区间类型，`S` 表示班次，`O` 表示加班。 |
| `ty_period_type` | `c length 1` | 期间类型，`W` 表示周，`M` 表示月。 |
| `ty_base_side` | `c length 1` | 基准日期前后窗口标识，`P` 表示前侧窗口，`N` 表示后侧窗口。 |

#### `ty_interval` 时间区间

用于表示班内休息区间。请假请使用单独的 `ty_leave` 数据，加班请使用单独的 `ty_overtime` 数据。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `beg_date` | `d` | 建议填写 | 区间开始日期。跨天班次中的休息建议明确传入，避免凌晨时间被当作班次归属日。 |
| `beg_time` | `t` | 是 | 区间开始时间，格式为 `HHMMSS`。 |
| `end_date` | `d` | 建议填写 | 区间结束日期。为空时默认等于 `beg_date`，若 `end_time <= beg_time` 则自动顺延一天。 |
| `end_time` | `t` | 是 | 区间结束时间，格式为 `HHMMSS`。 |

#### `ty_shift` 班次数据

每条记录表示某员工某个班次。员工一天只有一个班次时，一天传一条即可；如果实际数据中同一天误传多条，工具会按 `work_date` 累加工时。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `work_date` | `d` | 是 | 班次归属日期。每日工时和期间累计工时均按该日期归集。 |
| `rest_flag` | `ty_flag` | 否 | 休息标识。非空时表示该日期不上班，不需要填写上下班时间，也不会生成工作区间。 |
| `beg_time` | `t` | `rest_flag` 为空时必填 | 班次开始时间。开始日期固定按 `work_date` 处理。 |
| `end_date` | `d` | 否 | 班次结束日期。为空时，如果 `end_time <= beg_time`，自动认为班次跨到次日。 |
| `end_time` | `t` | `rest_flag` 为空时必填 | 班次结束时间。 |
| `breaks` | `tt_interval` | 否 | 班内休息区间，可多段。会从班次中扣除，并裁剪到班次范围内。 |

#### `ty_leave` 请假数据

用于实例接口中单独传入请假数据。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `beg_date` | `d` | 是 | 请假开始日期。 |
| `beg_time` | `t` | 否 | 请假开始时间。开始和结束时间都未填时，按整天请假处理。 |
| `end_date` | `d` | 是 | 请假结束日期。 |
| `end_time` | `t` | 否 | 请假结束时间。开始和结束日期不同，或开始/结束时间都未填时，按整天请假处理。 |

请假区间特殊规则：

- 如果 `beg_date` 和 `end_date` 不同，则该请假视为从 `beg_date 00:00:00` 到 `end_date + 1 00:00:00` 的整天请假，开始和结束时间会被忽略。
- 如果 `beg_time` 和 `end_time` 都为初始值，则该请假视为从开始日期到结束日期的整天请假；开始日期和结束日期相同时，即为当天一整天。
- 如果只有一端时间为 `000000`，仍按普通时间段处理。例如 `000000-090000` 表示当天 0 点到 9 点。
- ABAP `T` 类型中未填时间和显式 `000000` 都是初始值，因此整天判断使用“开始时间和结束时间都为初始值”作为未填时间的判断条件。
- 构造参数 `iv_leave_extend_calc` 非空时，整天请假会扩展到被该请假日期覆盖的跨天班次结束时间。例如 `20260604` 整天请假且 `20260604` 班次为 `163000-011500`，请假会扣减到 `20260605 011500`；该参数为空时只按整天请假的自然日区间扣减到 `20260605 000000`。
- 实例接口会先汇总对象中的全部 `ty_leave` 请假区间，再应用到所有重叠班次；同一请假区间重复送入也不会重复扣减。

#### `ty_overtime` 加班数据

用于实例接口中单独传入加班数据。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `beg_date` | `d` | 是 | 加班开始日期。 |
| `beg_time` | `t` | 是 | 加班开始时间。 |
| `end_date` | `d` | 否 | 加班结束日期。为空时，如果 `end_time <= beg_time`，自动认为加班跨到次日。 |
| `end_time` | `t` | 是 | 加班结束时间。 |

#### `ty_no_shift_overtime_rest` 无班次加班休息数据

用于维护无班次归属加班中的休息区间。无班次归属加班只要与这些休息区间重叠，就会固定从加班时长、当日累计和连续累计中扣除，不再受 `iv_actual_time` 影响。

| 字段 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- |
| `beg_date` | `d` | 是 | 休息开始日期。 |
| `beg_time` | `t` | 是 | 休息开始时间。 |
| `end_date` | `d` | 否 | 休息结束日期。为空时，如果 `end_time <= beg_time`，自动认为休息跨到次日。 |
| `end_time` | `t` | 是 | 休息结束时间。 |

### 公共方法说明

#### 静态用例导入方法

| 方法 | 说明 |
| --- | --- |
| `run_case_import` | 按 `ty_case_import` 扁平行导入 case，内部创建实例、设置班次/请假/加班数据，执行现有实例校验并返回与预期的比对结果。 |

`run_case_import` 不直接读取 xlsx 文件；调用方可先用 Excel 上传工具把“导入数据”页签整理成 `tt_case_import`。`row_type = SHIFT` 生成班次，`LEAVE` 生成请假，`OVERTIME` 生成加班，`NO_SHIFT_REST` 生成无班次加班休息，`EXPECT` 保存预期；`action_type = CANCEL` 仅支持 `row_type = LEAVE`，会从当前 case 已累积的请假数据中扣除取消区间，并且取消行本身不进入最终实例数据。取消区间可只覆盖原请假的一部分，例如原请假 `10:00-12:00`、取消 `10:30-11:30` 后，最终请假会拆成 `10:00-10:30` 和 `11:30-12:00` 两段。`NO_SHIFT_REST` 使用 `beg_date/beg_time/end_date/end_time` 维护休息区间，参与无班次归属加班的时长、当日累计和连续累计扣减。`check_scope` 使用分号分隔精确范围，默认 `SHIFT_REST;OVERTIME`，可追加或单独使用 `MONTH_OVERTIME`、`LONGEST_REST`。`leave_shift_rest_calc` 和 `full_day_leave_ot_calc` 可在任一行传入 `X`，方法会按 case 传给构造函数；为兼容只读取业务数据行的 Excel 上传程序，建议在同一 case 的 `SHIFT`/`LEAVE`/`OVERTIME` 行同步填写这些开关字段。

当前静态比对方法输出的原因码包括：`SHIFT_REST`、`OT_TOTAL`、`OT_DAY_TOTAL`、`OT_CONTINUOUS`、`OT_PREV_REST`、`OT_NEXT_REST`、`OT_LEAVE_OVERLAP`、`MONTH_OT_36`、`LONG_REST_24`。其中月度 36 小时和滚动 7 日连续 24 小时休息是 `run_case_import` 基于现有查询方法追加的阈值比对，不是 `get_overtime_compliance` 自身直接返回的标识。

#### 实例计算方法

这些方法直接使用对象中已经设置好的班次、请假和加班数据，不再经过额外的数据转调层。

| 方法 | 说明 |
| --- | --- |
| `get_daily_work_time` | 获取员工每天实际工作时长。 |
| `get_period_work_time` | 获取员工某段日期内累计实际工作时长。 |
| `get_week_work_time` | 根据基准日期获取所在周的累计实际工作时长。 |
| `get_shift_rest_time_v2` | 获取员工连续两个班次之间的休息时长，班次内加班归属班次，班次外加班按距上一段工作 12 小时规则处理。 |
| `get_longest_rest_time` | 获取员工某段日期内连续休息明细，并标记最长连续休息。 |
| `get_base_longest_rest_time` | 根据基准日期和前后计算天数分别获取前侧/后侧连续休息明细，并标记各自最长连续休息。 |
| `get_overtime_time` | 获取每段加班时长。 |
| `get_period_overtime_time` | 获取某段日期内累计加班时长。 |
| `get_base_overtime_time` | 根据基准日期、期间类型和日历日标识获取累计加班时长。 |
| `get_overtime_compliance` | 获取每段加班的合规校验结果，可直接过滤违规申请。 |

实例计算口径：

- 请假统一通过 `set_leave_data` / `add_leave` 维护，班次结构中不再兼容内嵌请假字段。
- 计算时会直接汇总 `mt_leave` 的请假区间，不会改动 `get_shift_data` 返回的原始班次表。
- 如果只有请假但没有任何班次，该请假不会产生工时扣减，也不会单独生成结果。
- `rest_flag` 非空的班次视为休息日：每日工时返回 0，班次间休息和实际工作区间计算会跳过该记录。
- 加班统一通过 `set_overtime_data` / `add_overtime` 维护。加班不会计入每日工时和期间工时，但会影响班次间休息和连续休息。
- 无班次加班休息统一通过 `set_no_shift_ot_rest_data` / `add_no_shift_ot_rest` 维护；无班次归属加班与休息重叠时固定扣除，不受 `iv_actual_time` 影响。

#### `get_daily_work_time`

获取员工每天实际工作时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `rt_result` | Returning | `tt_day_work` | - | 每天一行结果，按 `work_date` 排序。 |

计算口径：

- 统计日期使用 `ty_shift-work_date`，跨天班次不会拆分到次日。
- `breaks` 和请假重叠时不会重复扣减。
- `breaks` / 请假超出班次范围的部分会被忽略。
- 跨天或整天请假会应用到所有重叠班次，不要求只放在被扣减班次当天。

#### `get_period_work_time`

获取员工某段日期内累计实际工作时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 否 | 查询开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 否 | 查询结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `rt_result` | Returning | `tt_period_work` | - | 一行累计结果。 |

计算口径：

- 日期过滤依据 `work_date`，即班次归属日期。
- 实际工时复用 `get_daily_work_time` 的结果。
- 如果指定期间内没有工时，返回空表。

#### `get_week_work_time`

根据基准日期获取该周累计实际工作时长。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_base_date` | Importing | `d` | 是 | 基准日期。方法会自动取该日期所在周的周一和周日。 |
| `rt_result` | Returning | `tt_period_work` | - | 一行累计结果，`date_from` 为周一，`date_to` 为周日。 |

#### `get_shift_rest_time_v2`

获取员工连续两个班次之间的休息时长，使用班内/班外加班规则。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `rt_result` | Returning | `tt_shift_rest` | - | 相邻两个班次一行结果。 |

计算口径：

- 先按班次开始绝对时间排序，再计算相邻班次间隔。
- 与任一非休息班次重叠的加班时间先归属于班次，不再单独作为班次外加班打断休息；如果一段加班跨出班次，只保留班次外的部分继续参与判断。
- 班次外加班按时间顺序处理，并与上一段工作结束时间比较。上一段工作结束时间初始为上一班次结束；如果班次外加班距离上一段工作结束不足 12 小时，则该加班视为连续工作，休息开始点顺延到该加班结束。
- 如果某段班次外加班距离上一段工作结束已达到 12 小时，则返回上一段工作结束到该加班开始之间的休息时长。
- 如果两个班次之间没有班次外加班，休息时长 = 后一班次开始时间 - 前一班次结束时间。
- 默认使用原始班次开始/结束时间，不扣除班内休息和请假；如果构造参数 `iv_leave_shift_rest_calc` 非空，则按扣除请假后的实际工作首尾计算班次间休息，全班请假班次会被跳过。
- 如果数据存在重叠班次，`rest_seconds` 可能为负数。

#### `get_longest_rest_time`

获取员工某段日期内连续休息明细，并标记其中最长的一段。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 否 | 查询开始日期。为空时取对象班次数据中最小 `work_date`。 |
| `iv_date_to` | Importing | `d` | 否 | 查询结束日期。为空时取对象班次数据中最大 `work_date`。 |
| `rt_result` | Returning | `tt_longest_rest` | - | 连续休息明细。按休息开始时间升序返回，`is_longest = abap_true` 的记录为期间内最长连续休息；最长并列时会标记多行。 |

计算口径：

- 查询窗口为 `iv_date_from 00:00:00` 到 `iv_date_to + 1 00:00:00`。
- 根据输入的 `iv_date_from`、`iv_date_to` 限定查询窗口；为空时使用对象班次中的最小/最大 `work_date`。
- 先把班次扣除休息/请假后拆成实际工作区间，再把加班作为工作区间接入，然后在这些工作区间之间寻找连续非工作区间。
- 如果休息段是班次结束后到加班开始前，且该段不足常量 `gc_long_ot_rest_sec = 86400`（24 小时），则该段不作为连续休息结果；后续会以加班结束后的非工作时间作为休息段。
- 请假时间视为非工作时间，因此可能计入连续休息。
- 每行结果包含该段休息所在日期 `rest_date`、休息开始/结束时间、休息秒数/小时数，以及休息前后相邻工作区间的类型和开始/结束时间；如果休息位于查询窗口边界，前一工作区间或后一工作区间字段可能为空。

#### `get_base_longest_rest_time`

根据基准日期和计算天数获取连续休息明细，并分别标记基准日前侧和后侧窗口中的最长休息。

| 参数 | 方向 | 类型 | 必填 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_base_date` | Importing | `d` | 是 | 基准日期。 |
| `iv_calc_days` | Importing | `i` | 是 | 前后计算天数。负数会按绝对值处理。 |
| `rt_result` | Returning | `tt_longest_rest` | - | 连续休息明细。`base_side = P` 表示 `iv_base_date - iv_calc_days` 到 `iv_base_date` 的前侧窗口，`base_side = N` 表示 `iv_base_date` 到 `iv_base_date + iv_calc_days` 的后侧窗口；`base_date_from/base_date_to` 返回该行所属窗口，`is_longest` 按各自窗口单独标记。 |

#### `get_overtime_time`

获取每段加班时长。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_actual_time` | Importing | `abap_bool` | `abap_false` | 实际加班时长标识。为空时不扣班次重叠；非空时扣除与非休息班次重叠的部分。无班次归属加班与已设置休息重叠时固定扣除休息，不受该参数影响。 |
| `rt_result` | Returning | `tt_overtime_work` | - | 每段加班一行结果，按加班开始时间排序。 |

构造参数 `iv_full_day_leave_ot_calc` 非空时，被全天请假覆盖的班次不参与重叠扣减，因此全天请假期间的加班会按班次外加班统计。

#### `get_period_overtime_time`

获取某段日期内累计加班时长。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_date_from` | Importing | `d` | 空 | 查询开始日期。为空时取对象班次或加班数据中的最小日期。 |
| `iv_date_to` | Importing | `d` | 空 | 查询结束日期。为空时取对象班次或加班数据中的最大日期。 |
| `iv_actual_time` | Importing | `abap_bool` | `abap_true` | 实际加班时长标识。默认按实际加班累计并扣除与非休息班次重叠的部分；显式传空时不扣班次重叠。无班次归属加班与已设置休息重叠时固定扣除休息，不受该参数影响。 |
| `rt_result` | Returning | `tt_period_overtime` | - | 一行累计结果。统计依据为加班实际发生时间与查询期间的交集。 |

计算口径：

- 查询窗口为 `iv_date_from 00:00:00` 到 `iv_date_to + 1 00:00:00`。
- 跨月或跨查询边界的加班只累计落在窗口内的实际时长，不再按 `beg_date` 把整段加班归入开始日期。
- 默认按实际加班累计，扣除查询窗口内与非休息班次重叠的部分；只有显式传 `iv_actual_time = abap_false` 时，才按申请时长累计班次重叠部分。
- 无班次归属加班始终扣除查询窗口内与 `mt_no_shift_overtime_rest` 重叠的休息。
- 构造参数 `iv_full_day_leave_ot_calc` 非空时，被全天请假覆盖的班次不参与重叠扣减。

#### `get_base_overtime_time`

根据基准日期、期间类型和日历日标识获取累计加班时长。

| 参数 | 方向 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- | --- |
| `iv_base_date` | Importing | `d` | - | 基准日期。 |
| `iv_period_type` | Importing | `ty_period_type` | `W` | 期间类型。`W` 表示周，`M` 表示月；其他值按 `W` 处理。 |
| `iv_calendar_day` | Importing | `abap_bool` | `abap_false` | 日历日标识。非空时，`W` 取基准日期所在周周一至周日，`M` 取月初至月末；为空时，`W` 取基准日前后各 7 天，`M` 取基准日前后各 30 天。 |
| `iv_actual_time` | Importing | `abap_bool` | `abap_true` | 实际加班时长标识，传给 `get_period_overtime_time`。默认按实际加班累计；显式传空时才按申请时长累计班次重叠部分。 |
| `rt_result` | Returning | `tt_period_overtime` | - | 一行累计结果。 |

#### `get_overtime_compliance`

获取每段加班的合规校验结果。

| 参数 | 方向 | 类型 | 说明 |
| --- | --- | --- | --- |
| `iv_actual_time` | Importing | `abap_bool` | 保留兼容参数。无班次归属加班的 `overtime_seconds`、当日累计和连续累计固定扣除已设置的无班次加班休息，不受该参数影响。 |
| `rt_result` | Returning | `tt_overtime_compliance` | 每段加班一行结果，按加班开始时间排序。 |

计算口径：

- 与班次开始前 3 小时到班次开始、班次本身、或班次结束到班次结束后 3 小时有交集的加班，会关联到该班次；不再因为 `beg_date = work_date` 自动回退归属同日班次。
- `overtime_seconds/overtime_hours` 返回扣除班次重叠后的加班时长；加班落入正常班次时段或班内休息时段时不直接判违规，重叠部分不计入有效加班。无班次归属加班与已设置休息重叠时固定扣除休息。
- `leave_overlap_seconds/leave_overlap_hours` 返回加班申请与最终请假区间的重叠时长；只要重叠大于 0，`is_leave_overlap_valid = abap_false`，原因码为 `OT_LEAVE_OVERLAP`。该校验不受 `iv_full_day_leave_ot_calc` 放行，全天请假期间申报加班仍按请假重叠违规处理。
- 构造参数 `iv_full_day_leave_ot_calc` 非空时，被全天请假覆盖的班次不再用于加班归属和重叠扣减；该期间申报加班会按无班次归属加班继续校验。
- `outside_window_seconds > 0` 表示该段关联班次的加班有部分实际加班跨出班前/班后窗口；该跨出部分会和窗口内实际加班合并比较 3 小时上限，不再拆成无班次归属加班。
- `window_beg_date/window_beg_time` 到 `window_end_date/window_end_time` 返回的是该班次允许加班范围的最早开始和最晚结束；中间正常班次时段不计入窗口累计。
- `window_overtime_seconds` 是关联班次班前和班后窗口内全部加班申请的累计时长；若关联该班次的实际加班总量（窗口内实际加班 + 跨出窗口的实际加班）超过 3 小时，`is_total_valid = abap_false`。
- 有关联班次时，连续加班按同一关联班次的班前/班后窗口内实际片段合并；首尾相接或重叠视为连续，超过 3 小时时 `is_continuous_valid = abap_false`。
- 完全不与班次或班前/班后窗口相交的加班按无班次归属加班单独校验：距上一段工作结束必须至少 12 小时，未满足时 `is_prev_work_rest_valid = abap_false`；上一段工作包含前一班次，也包含前一段有效加班；连续加班不得超过 11 小时；按实际自然日切分后的当日累计不得超过 11 小时，超过时 `is_day_total_valid = abap_false`；如果后续存在下一班次，加班结束到下一班次开始必须至少 12 小时，未满足时 `is_next_shift_rest_valid = abap_false`。连续加班和当日累计固定扣除无班次加班休息。
- 无班次归属加班按绝对开始/结束时间排序，首尾相接或重叠视为同一段连续加班；因此跨天非整天加班拆成 `08:00-24:00` 和 `00:00-08:00` 两条时，会按一段连续加班校验，且上一段工作休息按连续组开始时间判断，不会把同组前一条拆分加班当作上一段工作重复报 `OT_PREV_REST`。
- `no_shift_day_overtime_seconds/no_shift_day_overtime_hours` 返回该加班涉及的无班次归属自然日最大累计值；`prev_work_*` 返回最近且最短的一段“上一段工作结束到无班次归属连续组开始”的休息信息；`next_shift_*` 返回最近且最短的一段“无班次归属加班结束到下一班次开始”的休息信息。
- `is_valid` 汇总请假重叠、班次窗口累计、无班次当日累计、连续申请、上一段工作休息和下一班次休息等标识；调用方可直接筛选 `is_valid = abap_false` 的记录。
- 关联班次优先取与加班申请在班前 3 小时、班次本身或班后 3 小时窗口有交集的班次；因此跨天班次的班后加班会归属于前一日班次。无班次/窗口交集时按无班次归属加班处理。

### 说明

- `work_date` 是班次归属日期；日工作时长、区间累计工时按 `work_date` 统计。
- `end_date` 可不传；当 `end_time <= beg_time` 时自动按跨天班次处理。
- 实例方式下，班次、请假、加班和无班次加班休息分开维护；计算时对象会汇总请假、加班和休息区间，不会改动原始数据表。
- 班内休息、请假会从班次中扣除，并自动裁剪到班次范围内，重叠区间不会重复扣减。
- 加班不会计入每日工时和期间工时，但会影响班次间休息和连续休息；加班时长请使用 `get_overtime_time` 或 `get_period_overtime_time` 获取。
- 加班时长方法中 `iv_actual_time` 为空时不扣班次重叠，非空时扣除班次重叠；累计类方法 `get_period_overtime_time` 和 `get_base_overtime_time` 默认按实际加班累计。无班次归属加班始终会扣除 `mt_no_shift_overtime_rest` 中维护的休息。
- 加班合规校验请使用 `get_overtime_compliance`，可通过 `is_window_valid`、`is_total_valid`、`is_day_total_valid`、`is_continuous_valid`、`is_prev_work_rest_valid`、`is_next_shift_rest_valid`、`is_leave_overlap_valid` 和 `is_valid` 判断结果。
- 跨天班次中，建议给 `breaks` 明确传入 `beg_date`、`end_date`，避免凌晨时间被误判到班次归属日。
- 请假开始和结束日期不同，或开始/结束时间都未填时，按整天请假处理。
- `iv_leave_extend_calc` 非空时，整天请假会扩展扣减到请假日期覆盖的跨天班次结束；为空时按自然日整天请假区间扣减。
- `iv_leave_shift_rest_calc` 非空时，班次间休息按扣除请假后的实际工作首尾计算；例如夜班到 01:15，次日白班 08:00-14:00 请假时，可按 14:00 作为下一段实际工作开始。
- `iv_full_day_leave_ot_calc` 非空时，全天请假覆盖的班次不会抵扣或归属加班；该期间加班按无班次归属加班继续校验。
- 跨日期请假会应用到所有重叠班次，示例中的跨日期请假在存在对应日期班次时会自动扣减。
