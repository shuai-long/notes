```abap
methods calculate_training_penalty
  importing
    iv_service_start type d
    iv_service_end   type d
    iv_actual_end    type d
    iv_training_fee  type decfloat34
  returning
    value(rv_penalty) type decfloat34.

methods calculate_service_months
  importing
    iv_start_date type d
    iv_end_date   type d
  returning
    value(rv_months) type decfloat34.
```

```abap
method calculate_training_penalty.
  data: lv_required_months type decfloat34,
        lv_served_months   type decfloat34,
        lv_unserved_months type decfloat34,
        lv_served_end      type d.

  rv_penalty = 0.

  if iv_training_fee <= 0
     or iv_service_start is initial
     or iv_service_end is initial
     or iv_service_end < iv_service_start.
    return.
  endif.

  lv_required_months = me->calculate_service_months(
    iv_start_date = iv_service_start
    iv_end_date   = iv_service_end ).

  if lv_required_months <= 0.
    return.
  endif.

  if iv_actual_end is initial or iv_actual_end < iv_service_start.
    lv_served_months = 0.
  else.
    if iv_actual_end > iv_service_end.
      lv_served_end = iv_service_end.
    else.
      lv_served_end = iv_actual_end.
    endif.

    lv_served_months = me->calculate_service_months(
      iv_start_date = iv_service_start
      iv_end_date   = lv_served_end ).
  endif.

  if lv_served_months >= lv_required_months.
    return.
  endif.

  lv_unserved_months = lv_required_months - lv_served_months.
  rv_penalty = iv_training_fee * lv_unserved_months / lv_required_months.
  rv_penalty = round( val = rv_penalty dec = 2 ).
endmethod.

method calculate_service_months.
  data: lv_start_year     type i,
        lv_start_month    type i,
        lv_start_day      type i,
        lv_end_year       type i,
        lv_end_month      type i,
        lv_end_day        type i,
        lv_full_months    type i,
        lv_add_years      type i,
        lv_add_months     type i,
        lv_anchor_date    type d,
        lv_remaining_days type i.

  rv_months = 0.

  if iv_start_date is initial or iv_end_date is initial or iv_end_date < iv_start_date.
    return.
  endif.

  lv_start_year  = iv_start_date+0(4).
  lv_start_month = iv_start_date+4(2).
  lv_start_day   = iv_start_date+6(2).
  lv_end_year    = iv_end_date+0(4).
  lv_end_month   = iv_end_date+4(2).
  lv_end_day     = iv_end_date+6(2).

  lv_full_months = ( lv_end_year - lv_start_year ) * 12 + lv_end_month - lv_start_month.

  if lv_end_day < lv_start_day.
    lv_full_months = lv_full_months - 1.
  endif.

  if lv_full_months < 0.
    lv_full_months = 0.
  endif.

  lv_add_years  = lv_full_months div 12.
  lv_add_months = lv_full_months mod 12.

  call function 'RP_CALC_DATE_IN_INTERVAL'
    exporting
      date      = iv_start_date
      days      = 0
      months    = lv_add_months
      signum    = '+'
      years     = lv_add_years
    importing
      calc_date = lv_anchor_date.

  lv_remaining_days = iv_end_date - lv_anchor_date.

  if lv_remaining_days > 0.
    lv_remaining_days = lv_remaining_days + 1.
  endif.

  rv_months = conv decfloat34( lv_full_months ).

  if lv_remaining_days > 0 and lv_remaining_days < 15.
    rv_months = rv_months + conv decfloat34( '0.5' ).
  elseif lv_remaining_days >= 15.
    rv_months = rv_months + 1.
  endif.
endmethod.
```
