       exporting
         retfield        = lv_retfield
         dynpprog        = sy-repid
         dynpnr          = sy-dynnr
         dynprofield     = lv_dynfield
         value_org       = 'S'
       tables
         value_tab       = it_vtable

         return_tab      = lt_return_tab
         dynpfld_mapping = lt_dynpfld_mapping
       exceptions

     if sy-subrc <> 0.

         with sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
     endif.
   
   endmethod.

FORM FRM_GET_STEXT TABLES  RECORD_TAB  STRUCTURE SEAHLPRES

DATA: LT_ZMMS_INV_TKPR LIKE TABLE OF LS_ZMMS_INV_TKPR,

                            ( FLDNAME = 'F0002' DYFLDNAME = 'ZMMS_INV_TKPR-ZITMSN' ) ).

    VALUE_TAB       = LT_ZMMS_INV_TKPR

  EXCEPTIONS
    PARAMETER_ERROR = 1
    NO_VALUES_FOUND = 2
